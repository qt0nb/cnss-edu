   217→   217→  if (!port || !port.linkId) return null;
   218→   218→  const link = t.links.find((l) => l.id === port.linkId);
   219→   219→  if (!link) return null;
   220→   220→  const aDev = findDevice(t, link.a.deviceId);
   221→   221→  const bDev = findDevice(t, link.b.deviceId);

   223→   223→  const isA = link.a.deviceId === (findDeviceByPort(t, port) ?? aDev).id;
   224→   224→  const other = isA ? bDev : aDev;
   225→   225→  const otherPort = findPort(other, isA ? link.b.portId : link.a.portId)!;
   226→   226→  return { link, otherDev: other, otherPort };

   234→   234→export function resolveLink(
   235→   235→  t: Topology,
   236→   236→  deviceId: string,
   237→   237→  portId: string
   238→   238→): { link: Link; otherDevId: string; otherPortId: string } | null {
   239→   239→  const dev = findDevice(t, deviceId);
   240→   240→  const port = dev && findPort(dev, portId);
   241→   241→  if (!dev || !port || !port.linkId) return null;
   242→   242→  const link = t.links.find((l) => l.id === port.linkId);
   243→   243→  if (!link) return null;
   244→   244→  if (link.a.deviceId === deviceId && link.a.portId === portId)
   245→   245→    return { link, otherDevId: link.b.deviceId, otherPortId: link.b.portId };
   246→   246→  if (link.b.deviceId === deviceId && link.b.portId === portId)
   247→   247→    return { link, otherDevId: link.a.deviceId, otherPortId: link.a.portId };
   248→   248→  return null;

   272→   272→  aPort.linkId = link.id;

   278→   278→export function autoCable(
   279→   279→  t: Topology,
   280→   280→  aDevId: string,
   281→   281→  bDevId: string
   282→   282→): { aPortId: string; bPortId: string; kind: LinkKind } | null {
   283→   283→  const a = findDevice(t, aDevId);
   284→   284→  const b = findDevice(t, bDevId);
   285→   285→  if (!a || !b || a.id === b.id) return null;
   286→   286→  // wireless: both have free wireless ports + same ssid
   287→   287→  const aw = a.ports.find((p) => p.kind === "wireless" && !p.linkId && p.adminUp);
   288→   288→  const bw = b.ports.find((p) => p.kind === "wireless" && !p.linkId && p.adminUp);
   289→   289→  const aWirelessOnly = a.kind === "smartphone";
   290→   290→  const bWirelessOnly = b.kind === "smartphone";
   291→   291→  if ((aw && bw) && (aWirelessOnly || bWirelessOnly || a.kind === "ap" || b.kind === "ap" || a.kind === "wirelessRouter" || b.kind === "wirelessRouter")) {
   292→   292→    return { aPortId: aw.id, bPortId: bw.id, kind: "wireless" };
   293→   293→  }
   294→   294→  const serial = (d: Device) => d.ports.find((p) => p.kind === "serial" && !p.linkId && p.adminUp);
   295→   295→  if (a.kind === "router" && b.kind === "router") {
   296→   296→    const sa = serial(a);
   297→   297→    const sb = serial(b);
   298→   298→    if (sa && sb) return { aPortId: sa.id, bPortId: sb.id, kind: "serial" };
   299→   299→  }
   300→   300→  const eth = (d: Device) =>
   301→   301→    d.ports.find((p) => (p.kind === "ethernet" || (p.kind === "internet" && d.kind === "wirelessRouter")) && !p.linkId && p.adminUp);
   302→   302→  const ae = eth(a);
   303→   303→  const be = eth(b);
   304→   304→  if (!ae || !be) return null;
   305→   305→  const aIsSwitchy = SWITCHY.includes(a.kind) || (a.kind === "wirelessRouter" && ae.id !== "g0/0");
   306→   306→  const bIsSwitchy = SWITCHY.includes(b.kind) || (b.kind === "wirelessRouter" && be.id !== "g0/0");
   307→   307→  const kind: LinkKind = aIsSwitchy !== bIsSwitchy ? "copper" : "crossover";
   308→   308→  return { aPortId: ae.id, bPortId: be.id, kind };
   309→   309→}

   314→   314→function mkLayers(p: Omit<Packet, "layers" | "id">, vlan?: number | null): PduLayer[] {
   315→   315→  const layers: PduLayer[] = [];
   316→   316→  if (p.proto === "DHCP") {
   317→   317→    layers.push({ n: 7, title: { ar: "طبقة التطبيق — DHCP", en: "Application — DHCP" }, fields: [ { k: { ar: "الرسالة", en: "Message" }, v: p.kind }, { k: { ar: "المحتوى", en: "Payload" }, v: p.payload ? `${p.payload.en}` : "-" } ] });
   318→   318→  }
   319→   319→  if (p.proto === "DNS") {
   320→   320→    layers.push({ n: 7, title: { ar: "طبقة التطبيق — DNS", en: "Application — DNS" }, fields: [ { k: { ar: "الاستعلام", en: "Query" }, v: p.kind }, { k: { ar: "الاسم", en: "Name" }, v: p.payload ? p.payload.en : "-" } ] });
   321→   321→  }
   322→   322→  if (p.proto === "HTTP") {
   323→   323→    layers.push({ n: 7, title: { ar: "طبقة التطبيق — HTTP", en: "Application — HTTP" }, fields: [ { k: { ar: "الرسالة", en: "Message" }, v: p.kind }, { k: { ar: "المحتوى", en: "Content" }, v: p.payload ? p.payload.en : "-" } ] });
   324→   324→  }
   325→   325→  if (p.srcPort !== undefined) {
   326→   326→    layers.push({ n: 4, title: { ar: "طبقة النقل", en: "Transport Layer" }, fields: [ { k: { ar: "البروتوكول", en: "Protocol" }, v: p.proto === "TCP" || p.proto === "HTTP" ? "TCP" : "UDP" }, { k: { ar: "المنفذ المصدر", en: "Src Port" }, v: String(p.srcPort) }, { k: { ar: "المنفذ الهدف", en: "Dst Port" }, v: String(p.dstPort) } ] });
   327→   327→  } else if (p.proto === "ICMP") {
   328→   328→    layers.push({ n: 4, title: { ar: "طبقة النقل — ICMP", en: "Transport — ICMP" }, fields: [ { k: { ar: "النوع", en: "Type" }, v: p.kind === "Echo Request" ? "8 (Echo Request)" : "0 (Echo Reply)" } ] });
   329→   329→  }
   330→   330→  layers.push({ n: 3, title: { ar: "طبقة الشبكة — IP", en: "Network — IP" }, fields: [ { k: { ar: "عنوان المصدر", en: "Source IP" }, v: p.srcIp }, { k: { ar: "عنوان الهدف", en: "Dest IP" }, v: p.dstIp }, { k: { ar: "TTL", en: "TTL" }, v: String(p.ttl) }, ...(p.proto === "ICMP" ? [] : [{ k: { ar: "البروتوكول", en: "Protocol" }, v: p.proto === "ARP" ? "-" : p.srcPort !== undefined ? (p.proto === "TCP" || p.proto === "HTTP" ? "TCP" : "UDP") : "ICMP" }]) ] });
   331→   331→  layers.push({ n: 2, title: { ar: "طبقة وصلة البيانات — إيثرنت", en: "Data Link — Ethernet" }, fields: [ { k: { ar: "MAC المصدر", en: "Src MAC" }, v: p.srcMac }, { k: { ar: "MAC الهدف", en: "Dst MAC" }, v: p.dstMac }, { k: { ar: "EtherType", en: "EtherType" }, v: p.proto === "ARP" ? ETHERTYPE.ARP : ETHERTYPE.IP }, ...(vlan ? [ { k: { ar: "وسم 802.1Q", en: "802.1Q Tag" }, v: `VLAN ${vlan} (+4 bytes)` } ] : []) ] });
   332→   332→  layers.push({ n: 1, title: { ar: "الطبقة الفيزيائية", en: "Physical Layer" }, fields: [ { k: { ar: "الوسيط", en: "Medium" }, v: "bits — إشارات كهربائية/ضوئية" } ] });
   333→   333→  return layers.reverse().slice().sort((a, b) => b.n - a.n); // L7 on top like PT
   334→   334→}

   352→   352→  warn: string | null; // bilingual note at end
   353→   353→  seen: Set<string>; // loop guard
   354→   354→  maxSteps: number;
   355→   355→  /** per-source flow stats (flood/scan detection at IDS) */
   356→   356→  flow: Map<string, FlowStat>;
   357→   357→  /** IP→MAC binding DB for ARP-spoof detection */
   358→   358→  ipMac: Map<string, string>;
   359→   359→  /** per-target hit counters (flood saturation on victims) */
   360→   360→  hit: Map<string, number>;
   361→   361→}

   363→   363→function step(
   364→   364→  ctx: Ctx,
   365→   365→  dev: Device,
   366→   366→  inPort: string | null,
   367→   367→  outPort: string | null,
   368→   368→  toDeviceId: string | null,
   369→   369→  linkId: string | null,
   370→   370→  packet: Packet,
   371→   371→  info: { ar: string; en: string },
   372→   372→  dropped = false
   373→   373→) {
   374→   374→  ctx.steps.push({
   375→   375→    id: ++ctx.sid,
   376→   376→    deviceId: dev.id,
   377→   377→    deviceName: dev.name,
   378→   378→    inPort,
   379→   379→    outPort,
   380→   380→    linkId,
   381→   381→    toDeviceId,
   382→   382→    packet,
   383→   383→    dropped,
   384→   384→    info,
   385→   385→  });
   386→   386→}
   387→   387→
   388→   388→const info = (ar: string, en: string) => ({ ar, en });

   395→   395→  return d.ports.filter((p) => p.id !== "g0/0");

   398→   398→function learnMac(ctx: Ctx, d: Device, inPort: string, srcMac: string, vlan: number) {
   399→   399→  const exists = d.macTable.find((e) => e.mac === srcMac && e.vlan === vlan);
   400→   400→  if (!exists) d.macTable.push({ mac: srcMac, portId: inPort, vlan });
   401→   401→}

   403→   403→function egressPorts(ctx: Ctx, d: Device, inPortId: string, dstMac: string, ingressVlan: number): { port: Port; vlan: number }[] {
   404→   404→  if (d.kind === "hub") {
   405→   405→    return d.ports
   406→   406→      .filter((p) => p.id !== inPortId && p.linkId && p.adminUp)
   407→   407→      .map((p) => ({ port: p, vlan: ingressVlan }));
   408→   408→  }
   409→   409→  if (isBridge(d) || d.kind === "wirelessRouter") {
   410→   410→    const bridgePorts = d.kind === "wirelessRouter" ? wrLanPorts(d) : d.ports;
   411→   411→    const inPort = d.ports.find((p) => p.id === inPortId)!;
   412→   412→    const inIsTrunk = inPort.trunk;
   413→   413→    const up = (p: Port) => p.linkId && p.adminUp && p.id !== inPortId;
   414→   414→    if (dstMac === BROADCAST_MAC) {
   415→   415→      return bridgePorts
   416→   416→        .filter((p) => up(p) && (inIsTrunk || p.trunk || p.accessVlan === ingressVlan))
   417→   417→        .map((p) => ({ port: p, vlan: p.trunk ? ingressVlan : p.accessVlan === ingressVlan ? ingressVlan : ingressVlan }));
   418→   418→    }
   419→   419→    const entry = d.macTable.find((e) => e.mac === dstMac && (e.vlan === ingressVlan || inIsTrunk));
   420→   420→    if (entry) {
   421→   421→      const p = bridgePorts.find((pp) => pp.id === entry.portId);
   422→   422→      if (p && up(p)) {
   423→   423→        const dstIsTrunk = p.trunk;
   424→   424→        if (!dstIsTrunk && !inIsTrunk && p.accessVlan !== ingressVlan) {
   425→   425→          step(ctx, d, inPortId, p.id, null, null, mkPacket({ proto: "ARP", kind: "Dropped", kindBi: { ar: "إطار مُسقط", en: "Frame dropped" }, srcMac: dstMac, dstMac, srcIp: ZERO_IP, dstIp: ZERO_IP, ttl: 0 }), info("أُسقط الإطار: المنفذ في VLAN مختلف", "Frame dropped: port is in a different VLAN"), true);
   426→   426→          return [];
   427→   427→        }
   428→   428→        return [{ port: p, vlan: dstIsTrunk ? ingressVlan : ingressVlan }];
   429→   429→      }
   430→   430→      return [];
   431→   431→    }
   432→   432→    // unknown unicast → flood
   433→   433→    return bridgePorts
   434→   434→      .filter((p) => up(p) && (inIsTrunk || p.trunk || p.accessVlan === ingressVlan))
   435→   435→      .map((p) => ({ port: p, vlan: p.trunk ? ingressVlan : ingressVlan }));
   436→   436→  }
   437→   437→  return [];
   438→   438→}

   440→   440→// send a frame out of a device port onto the wire
   441→   441→function wireOut(
   442→   442→  ctx: Ctx,
   443→   443→  fromDev: Device,
   444→   444→  fromPortId: string,
   445→   445→  packet: Packet,
   446→   446→  depth = 0
   447→   447→): void {
   448→   448→  if (depth > 40 || ctx.steps.length > ctx.maxSteps) {
   449→   449→    step(ctx, fromDev, null, fromPortId, null, null, packet, info("تم إيقاف المحاكاة: حلقة أو عمق كبير — يحتاج STP", "Simulation stopped: loop or excessive depth — STP needed"), true);
   450→   450→    return;
   451→   451→  }
   452→   452→  const res = resolveLink(ctx.t, fromDev.id, fromPortId);
   453→   453→  const port = findPort(fromDev, fromPortId);
   454→   454→  if (!res || !port || !port.adminUp) {
   455→   455→    step(ctx, fromDev, null, fromPortId, null, null, packet, info("الحزمة سقطت: المنفذ غير متصل أو مغلق (shutdown)", "Packet dropped: port not connected or shut down"), true);
   456→   456→    return;
   457→   457→  }
   458→   458→  const outPacket = packet.vlan !== undefined && packet.vlan !== null ? packet : packet;
   459→   459→  step(ctx, fromDev, null, fromPortId, res.otherDevId, res.link.id, outPacket, info("", ""));
   460→   460→  const other = findDevice(ctx.t, res.otherDevId);
   461→   461→  if (!other) return;
   462→   462→  deliver(ctx, other, res.otherPortId, packet, depth + 1);
   463→   463→}

   465→   465→function ingressVlanOf(d: Device, portId: string): number {
   466→   466→  const p = findPort(d, portId);
   467→   467→  if (!p) return 1;
   468→   468→  return p.trunk ? -1 : p.accessVlan; // -1 = came tagged
   469→   469→}

   471→   471→function deliver(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
   472→   472→  const key = `${d.id}|${inPortId}|${packet.id}`;
   473→   473→  if (ctx.seen.has(key)) {
   474→   474→    step(ctx, d, inPortId, null, null, null, packet, info("حزمة تدور في حلقة (Loop) — في الواقع يمنعها STP", "Packet looping — real switches prevent this with STP"), true);
   475→   475→    return;
   476→   476→  }
   477→   477→  ctx.seen.add(key);
   478→   478→
   479→   479→  // VLAN ingress check on access ports (broadcast tagged?)
   480→   480→  const inPort = findPort(d, inPortId);
   481→   481→  if (!inPort) return;
   482→   482→
   483→   483→  // ── bridges (switch/hub/cloud/ap/l3switch/ids) and wireless-router LAN side
   484→   484→  if (isBridge(d) || (d.kind === "wirelessRouter" && inPortId !== "g0/0")) {
   485→   485→    // port-security: lock access ports to their first learned MAC (anti ARP-spoof mitigation)
   486→   486→    if ((d.kind === "switch" || d.kind === "l3switch") && inPort.kind === "ethernet" && inPort.secure) {
   487→   487→      if (inPort.stickyMac && inPort.stickyMac !== packet.srcMac) {
   488→   488→        d.statsDropped = (d.statsDropped ?? 0) + 1;
   489→   489→        pushAlert(d, "port-security", "critical", packet.srcIp, packet.dstIp,
   490→   490→          `انتهاك أمن المنفذ ${inPort.name}: MAC ${packet.srcMac} ≠ المثبّت ${inPort.stickyMac} — الإطار مرفوض`,
   491→   491→          `Port-security violation on ${inPort.name}: MAC ${packet.srcMac} ≠ sticky ${inPort.stickyMac} — frame rejected`);
   492→   492→        step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: أمن المنفذ أسقط إطاراً من ${packet.srcMac}`, `${d.name}: port-security dropped a frame from ${packet.srcMac}`), true);
   493→   493→        return;
   494→   494→      }
   495→   495→      if (!inPort.stickyMac) inPort.stickyMac = packet.srcMac;
   496→   496→    }
   497→   497→    // IDS sensor: inspect everything that traverses this bridge
   498→   498→    if (d.kind === "ids") idsInspect(ctx, d, inPortId, packet);
   499→   499→    // L3 switch: packets addressed to an SVI (its MAC/IP) are routed instead of switched
   500→   500→    if (d.kind === "l3switch") {
   501→   501→      const svi = d.ports.find((p) => p.kind === "svi" && (p.mac === packet.dstMac || (p.ip != null && p.ip === packet.dstIp)));
   502→   502→      if (svi || packet.proto === "ARP") {
   503→   503→        l3Route(ctx, d, inPortId, packet, depth);
   504→   504→        if (svi && packet.dstMac === svi.mac && packet.proto !== "ARP") return;
   505→   505→      }
   506→   506→    }
   507→   507→    const vlan = inPort.trunk ? (packet.vlan ?? 1) : inPort.accessVlan;
   508→   508→    if (d.kind !== "hub") learnMac(ctx, d, inPortId, packet.srcMac, vlan);
   509→   509→    const outs = egressPorts(ctx, d, inPortId, packet.dstMac, vlan);
   510→   510→    if (outs.length === 0) {
   511→   511→      // host attached? just no egress; record device entry info
   512→   512→      if (packet.dstMac === BROADCAST_MAC) {
   513→   513→        step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: بثّ الإطار ولا منافذ خروج أخرى`, `${d.name}: broadcast received, no other egress ports`));
   514→   514→      }
   515→   515→    }
   516→   516→    for (const { port, vlan: outVlan } of outs) {
   517→   517→      const tagged = port.trunk ? vlan : null;
   518→   518→      const pkt = tagged !== null ? { ...packet, vlan: tagged } : { ...packet, vlan: null };
   519→   519→      const localInfo =
   520→   520→        packet.dstMac === BROADCAST_MAC
   521→   521→          ? info(`${d.name}: غمر البث من ${inPort.name} إلى ${port.name}`, `${d.name}: flooding broadcast from ${inPort.name} to ${port.name}`)
   522→   522→          : info(`${d.name}: تعلم ${packet.srcMac} على ${inPort.name} ثم التوجيه إلى ${port.name}`, `${d.name}: learned ${packet.srcMac} on ${inPort.name}, forwarding out ${port.name}`);
   523→   523→      step(ctx, d, inPortId, port.id, null, null, pkt, localInfo);
   524→   524→      wireOut(ctx, d, port.id, pkt, depth);
   525→   525→    }
   526→   526→    // wireless-router LAN side: packets for the router itself (broadcast DHCP/DNS/ARP or unicast to its MAC/IP)
   527→   527→    // are also serviced locally — DHCP pool, DNS zone, ARP replies, ICMP, HTTP — like a real home router
   528→   528→    if (d.kind === "wirelessRouter") {
   529→   529→      const toMe =
   530→   530→        packet.dstMac === BROADCAST_MAC ||
   531→   531→        d.ports.some((p) => p.mac === packet.dstMac) ||
   532→   532→        d.ports.some((p) => p.ip && p.ip === packet.dstIp);
   533→   533→      const serviceable =
   534→   534→        packet.proto === "DHCP" ||
   535→   535→        packet.proto === "ARP" ||
   536→   536→        (packet.proto === "DNS" && packet.kind === "DNS Query") ||
   537→   537→        packet.proto === "ICMP" ||
   538→   538→        packet.proto === "HTTP" ||
   539→   539→        packet.proto === "TCP";
   540→   540→      if (toMe && serviceable) {
   541→   541→        routerHandle(ctx, d, inPortId, packet, depth);
   542→   542→      }
   543→   543→    }
   544→   544→    // if a host/server is directly the target of this bridge's delivery, it arrives via wireOut→deliver on the host
   545→   545→    return;
   546→   546→  }
   547→   547→
   548→   548→  // ── router / firewall / wireless-router WAN
   549→   549→  if (d.kind === "router" || d.kind === "firewall" || d.kind === "wirelessRouter") {
   550→   550→    routerHandle(ctx, d, inPortId, packet, depth);
   551→   551→    return;
   552→   552→  }
   553→   553→
   554→   554→  // ── hosts (pc/server/laptop/smartphone/attacker)
   555→   555→  hostHandle(ctx, d, inPortId, packet, depth);
   556→   556→}

   558→   558→function addArp(d: Device, ip: string, mac: string, portId: string) {
   559→   559→  if (!d.arp.find((e) => e.ip === ip)) d.arp.push({ ip, mac, portId });
   560→   560→}

   562→   562→export function hostIp(d: Device): { ip: string | null; mask: string | null; port: Port | null } {
   563→   563→  if (d.kind === "pc" || d.kind === "server") {
   564→   564→    const p = d.ports[0];
   565→   565→    return { ip: p.ip ?? d.staticIp, mask: p.mask ?? d.staticMask, port: p };
   566→   566→  }
   567→   567→  // laptop/smartphone: first port with ip
   568→   568→  for (const p of d.ports) if (p.ip) return { ip: p.ip, mask: p.mask, port: p };
   569→   569→  return { ip: d.staticIp, mask: d.staticMask, port: d.ports[0] ?? null };
   570→   570→}

   572→   572→function hostHandle(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
   573→   573→  const forMe = packet.dstMac === BROADCAST_MAC || d.ports.some((p) => p.mac === packet.dstMac);
   574→   574→  const { ip } = hostIp(d);
   575→   575→  const ipForMe = ip !== null && (packet.dstIp === ip || packet.dstIp === BROADCAST_IP);
   576→   576→
   577→   577→  if (packet.proto === "ARP" && packet.kind === "ARP Request") {
   578→   578→    // learn sender
   579→   579→    addArp(d, packet.srcIp, packet.srcMac, inPortId);
   580→   580→    if (ip !== null && packet.dstIp === ip) {
   581→   581→      const out = d.ports.find((p) => p.linkId && p.adminUp) ?? d.ports[0];
   582→   582→      const reply = mkPacket({
   583→   583→        proto: "ARP",
   584→   584→        kind: "ARP Reply",
   585→   585→        kindBi: { ar: "رد ARP", en: "ARP Reply" },
   586→   586→        srcMac: out.mac,
   587→   587→        dstMac: packet.srcMac,
   588→   588→        srcIp: ip,
   589→   589→        dstIp: packet.srcIp,
   590→   590→        ttl: 128,
   591→   591→        payload: { ar: `أنا ${ip} وعنواني ${out.mac}`, en: `I am ${ip}, my MAC is ${out.mac}` },
   592→   592→      });
   593→   593→      step(ctx, d, inPortId, out.id, null, null, reply, info(`${d.name}: أنا ${ip} — أرد على طلب ARP`, `${d.name}: I am ${ip} — replying to ARP request`));
   594→   594→      wireOut(ctx, d, out.id, reply, depth);
   595→   595→    } else {
   596→   596→      step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: طلب ARP ليس له — تجاهل`, `${d.name}: ARP request not for me — ignored`));
   597→   597→    }
   598→   598→    return;
   599→   599→  }
   600→   600→
   601→   601→  if (packet.proto === "ARP" && packet.kind === "ARP Reply") {
   602→   602→    addArp(d, packet.srcIp, packet.srcMac, inPortId);
   603→   603→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: تعلمت ${packet.srcIp} هو ${packet.srcMac} — سُجّل في جدول ARP`, `${d.name}: learned ${packet.srcIp} is ${packet.srcMac} — recorded in ARP table`));
   604→   604→    return;
   605→   605→  }
   606→   606→
   607→   607→  if (!forMe) {
   608→   608→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: ليس عنواني MAC — تجاهل الإطار`, `${d.name}: not my MAC — frame ignored`));
   609→   609→    return;
   610→   610→  }
   611→   611→
   612→   612→  if (packet.proto === "ICMP" && packet.kind === "Echo Request" && ipForMe) {
   613→   613→    // flood saturation: too many echoes from one source exhaust the victim CPU
   614→   614→    const hits = (ctx.hit.get(`${d.id}|${packet.srcIp}`) ?? 0) + 1;
   615→   615→    ctx.hit.set(`${d.id}|${packet.srcIp}`, hits);
   616→   616→    if (hits >= 6) d.overloaded = true;
   617→   617→    if (d.overloaded) {
   618→   618→      step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: مشبع بالطلبات — CPU 100%، الحزمة مُسقطة`, `${d.name}: saturated by flood — CPU 100%, packet dropped`), true);
   619→   619→      return;
   620→   620→    }
   621→   621→    const out = d.ports.find((p) => p.linkId && p.adminUp) ?? d.ports[0];
   622→   622→    const reply = mkPacket({
   623→   623→      proto: "ICMP",
   624→   624→      kind: "Echo Reply",
   625→   625→      kindBi: { ar: "رد ICMP", en: "ICMP Echo Reply" },
   626→   626→      srcMac: out.mac,
   627→   627→      dstMac: packet.srcMac,
   628→   628→      srcIp: ip!,
   629→   629→      dstIp: packet.srcIp,
   630→   630→      ttl: 128,
   631→   631→    });
   632→   632→    step(ctx, d, inPortId, out.id, null, null, reply, info(`${d.name}: وصلني Ping من ${packet.srcIp} — أرد!`, `${d.name}: ping received from ${packet.srcIp} — replying!`));
   633→   633→    wireOut(ctx, d, out.id, reply, depth);
   634→   634→    return;
   635→   635→  }
   636→   636→
   637→   637→  if (packet.proto === "DHCP") {
   638→   638→    dhcpHostHandle(ctx, d, inPortId, packet, depth);
   639→   639→    return;
   640→   640→  }
   641→   641→  if (packet.proto === "DNS" && packet.kind === "DNS Query") {
   642→   642→    // client receiving answer handled below; server:
   643→   643→    dnsHandle(ctx, d, inPortId, packet, depth);
   644→   644→    return;
   645→   645→  }
   646→   646→  if (packet.proto === "DNS" && packet.kind === "DNS Answer") {
   647→   647→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: وصل جواب DNS`, `${d.name}: DNS answer received`));
   648→   648→    return;
   649→   649→  }
   650→   650→  if (packet.proto === "TCP" || packet.proto === "HTTP") {
   651→   651→    httpHandle(ctx, d, inPortId, packet, depth);
   652→   652→    return;
   653→   653→  }
   654→   654→
   655→   655→  step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: استلم الحزمة`, `${d.name}: packet received`));
   656→   656→}

   659→   659→export function connectedRoutes(d: Device): import("./types").Route[] {
   660→   660→  const routes: import("./types").Route[] = [];
   661→   661→  for (const p of d.ports) {
   662→   662→    if (p.ip && p.mask) {
   663→   663→      routes.push({ network: networkOf(p.ip, p.mask), mask: p.mask, nextHop: null, iface: p.id, ad: 0, kind: "connected" });
   664→   664→    }
   665→   665→  }
   666→   666→  return routes;
   667→   667→}

   669→   669→function lookupRoute(d: Device, dstIp: string): import("./types").Route | null {
   670→   670→  const all = [...connectedRoutes(d), ...d.routes];
   671→   671→  let best: import("./types").Route | null = null;
   672→   672→  let bestLen = -1;
   673→   673→  for (const r of all) {
   674→   674→    if (networkOf(dstIp, r.mask) === r.network) {
   675→   675→      const len = maskToPrefix(r.mask);
   676→   676→      if (len > bestLen || (len === bestLen && r.ad < (best?.ad ?? 99))) {
   677→   677→        best = r;
   678→   678→        bestLen = len;
   679→   679→      }
   680→   680→    }
   681→   681→  }
   682→   682→  return best;
   683→   683→}

   685→   685→function routerHandle(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
   686→   686→  const inPort = findPort(d, inPortId);
   687→   687→  if (!inPort) return;
   688→   688→  const myMacs = d.ports.map((p) => p.mac);
   689→   689→  const forMe = packet.dstMac === BROADCAST_MAC || myMacs.includes(packet.dstMac);
   690→   690→
   691→   691→  // ARP for me?
   692→   692→  if (packet.proto === "ARP" && packet.kind === "ARP Request") {
   693→   693→    addArp(d, packet.srcIp, packet.srcMac, inPortId);
   694→   694→    const targetPort = d.ports.find((p) => p.ip === packet.dstIp);
   695→   695→    if (forMe && targetPort) {
   696→   696→      const replyPort = targetPort.linkId && targetPort.adminUp ? targetPort : inPort;
   697→   697→      const reply = mkPacket({
   698→   698→        proto: "ARP",
   699→   699→        kind: "ARP Reply",
   700→   700→        kindBi: { ar: "رد ARP", en: "ARP Reply" },
   701→   701→        srcMac: targetPort.mac,
   702→   702→        dstMac: packet.srcMac,
   703→   703→        srcIp: packet.dstIp,
   704→   704→        dstIp: packet.srcIp,
   705→   705→        ttl: 128,
   706→   706→        payload: { ar: `أنا ${packet.dstIp} عند ${targetPort.mac}`, en: `I am ${packet.dstIp} at ${targetPort.mac}` },
   707→   707→      });
   708→   708→      step(ctx, d, inPortId, replyPort.id, null, null, reply, info(`${d.name}: رد ARP من ${targetPort.name}`, `${d.name}: ARP reply from ${targetPort.name}`));
   709→   709→      wireOut(ctx, d, replyPort.id, reply, depth);
   710→   710→    } else {
   711→   711→      step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: طلب ARP ليس لعناويني`, `${d.name}: ARP request not for my IPs`));
   712→   712→    }
   713→   713→    return;
   714→   714→  }
   715→   715→
   716→   716→  if (packet.proto === "ARP" && packet.kind === "ARP Reply") {
   717→   717→    addArp(d, packet.srcIp, packet.srcMac, inPortId);
   718→   718→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: تعلمت ${packet.srcIp} هو ${packet.srcMac} — سُجّل في ARP`, `${d.name}: learned ${packet.srcIp} is ${packet.srcMac} — recorded in ARP`));
   719→   719→    return;
   720→   720→  }
   721→   721→
   722→   722→  // reverse NAT: a session reply arriving on the WAN addressed to my WAN IP → translate & deliver to the LAN host
   723→   723→  if (d.kind === "wirelessRouter" && d.nat && inPortId === "g0/0") {
   724→   724→    const wanP = findPort(d, "g0/0");
   725→   725→    if (wanP?.ip && packet.dstIp === wanP.ip) {
   726→   726→      const map =
   727→   727→        packet.dstPort != null
   728→   728→          ? d.natTable.find((m) => m.wanPort === packet.dstPort)
   729→   729→          : [...d.natTable].reverse().find((m) => !m.lanPort);
   730→   730→      if (map) {
   731→   731→        const out = d.ports.find((p) => p.linkId && p.adminUp && p.id !== "g0/0");
   732→   732→        if (out) {
   733→   733→          let arpE = d.arp.find((a) => a.ip === map.lanIp);
   734→   734→          if (!arpE) {
   735→   735→            ensureArpFor(ctx, d, out, map.lanIp, depth);
   736→   736→            arpE = d.arp.find((a) => a.ip === map.lanIp);
   737→   737→          }
   738→   738→          if (arpE) {
   739→   739→            const fwd = {
   740→   740→              ...packet,
   741→   741→              dstIp: map.lanIp,
   742→   742→              dstPort: map.lanPort || packet.dstPort,
   743→   743→              dstMac: arpE.mac,
   744→   744→              srcMac: out.mac,
   745→   745→              ttl: packet.ttl - 1,
   746→   746→            };
   747→   747→            step(ctx, d, inPortId, out.id, null, null, fwd, info(`${d.name}: NAT عكسي — إعادة ${packet.dstIp}:${packet.dstPort ?? "-"} إلى ${map.lanIp}:${map.lanPort || "-"}`, `${d.name}: reverse NAT — restoring ${packet.dstIp}:${packet.dstPort ?? "-"} to ${map.lanIp}:${map.lanPort || "-"}`));
   748→   748→            wireOut(ctx, d, out.id, fwd, depth);
   749→   749→            return;
   750→   750→          }
   751→   751→        }
   752→   752→      }
   753→   753→    }
   754→   754→  }
   755→   755→
   756→   756→  if (!forMe) {
   757→   757→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: ليس عنواني MAC — تجاهل`, `${d.name}: not my MAC — ignored`));
   758→   758→    return;
   759→   759→  }
   760→   760→
   761→   761→  // DHCP server on router? (not typical—servers handle) → skip
   762→   762→  // local services only answer packets addressed to one of my own IPs — everything else gets routed/forwarded
   763→   763→  const isMyDstIp = d.ports.some((p) => p.ip === packet.dstIp);
   764→   764→  if (packet.proto === "DNS" && packet.kind === "DNS Query" && isMyDstIp) {
   765→   765→    dnsHandle(ctx, d, inPortId, packet, depth);
   766→   766→    return;
   767→   767→  }
   768→   768→  if ((packet.proto === "TCP" || packet.proto === "HTTP") && isMyDstIp) {
   769→   769→    httpHandle(ctx, d, inPortId, packet, depth);
   770→   770→    return;
   771→   771→  }
   772→   772→  if (packet.proto === "DHCP") {
   773→   773→    dhcpHostHandle(ctx, d, inPortId, packet, depth);
   774→   774→    return;
   775→   775→  }
   776→   776→
   777→   777→  const dstMe = d.ports.find((p) => p.ip === packet.dstIp);
   778→   778→  if (dstMe && packet.proto === "ICMP" && packet.kind === "Echo Request") {
   779→   779→    const replyPort = dstMe.linkId && dstMe.adminUp ? dstMe : inPort;
   780→   780→    const reply = mkPacket({
   781→   781→      proto: "ICMP",
   782→   782→      kind: "Echo Reply",
   783→   783→      kindBi: { ar: "رد ICMP", en: "ICMP Echo Reply" },
   784→   784→      srcMac: dstMe.mac,
   785→   785→      dstMac: packet.srcMac,
   786→   786→      srcIp: packet.dstIp,
   787→   787→      dstIp: packet.srcIp,
   788→   788→      ttl: 128,
   789→   789→    });
   790→   790→    step(ctx, d, inPortId, replyPort.id, null, null, reply, info(`${d.name}: الهدف أنا — رد Pong`, `${d.name}: I am the target — pong reply`));
   791→   791→    wireOut(ctx, d, replyPort.id, reply, depth);
   792→   792→    return;
   793→   793→  }
   794→   794→
   795→   795→  // firewall ACL transit filter (traffic to the firewall itself was handled above)
   796→   796→  if (d.kind === "firewall" || (d.acls?.length ?? 0) > 0 || d.defaultDeny) {
   797→   797→    const verdict = aclCheck(d, packet);
   798→   798→    if (verdict.action === "deny") {
   799→   799→      d.statsDropped = (d.statsDropped ?? 0) + 1;
   800→   800→      if (verdict.rule) verdict.rule.hits++;
   801→   801→      pushAlert(d, "acl-deny", "warn", packet.srcIp, packet.dstIp,
   802→   802→        `ACL أسقط ${packet.proto} من ${packet.srcIp} إلى ${packet.dstIp}${verdict.rule ? ` (قاعدة #${verdict.rule.id})` : " (سياسة الرفض الافتراضية)"}`,
   803→   803→        `ACL dropped ${packet.proto} from ${packet.srcIp} to ${packet.dstIp}${verdict.rule ? ` (rule #${verdict.rule.id})` : " (default deny)"}`);
   804→   804→      step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: ACL — رفض ${packet.proto} ${packet.srcIp} → ${packet.dstIp}`, `${d.name}: ACL — deny ${packet.proto} ${packet.srcIp} → ${packet.dstIp}`), true);
   805→   805→      return;
   806→   806→    }
   807→   807→  }
   808→   808→
   809→   809→  // forwarding
   810→   810→  if (packet.ttl <= 1) {
   811→   811→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: TTL انتهى — إسقاط الحزمة`, `${d.name}: TTL expired — packet dropped`), true);
   812→   812→    return;
   813→   813→  }
   814→   814→  const route = lookupRoute(d, packet.dstIp);
   815→   815→  if (!route) {
   816→   816→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: لا يوجد مسار إلى ${packet.dstIp} — إسقاط (أضف ip route)`, `${d.name}: no route to ${packet.dstIp} — dropping (add ip route)`), true);
   817→   817→    return;
   818→   818→  }
   819→   819→  const nextIp = route.nextHop ?? packet.dstIp;
   820→   820→  // resolve egress: explicit iface, or the port whose subnet contains the next hop (recursive next-hop resolution)
   821→   821→  let outPort = route.iface ? findPort(d, route.iface) : undefined;
   822→   822→  if (!outPort || !outPort.linkId || !outPort.adminUp) {
   823→   823→    const nh = d.ports.find(
   824→   824→      (p) => p.ip && p.mask && p.linkId && p.adminUp && networkOf(nextIp, p.mask) === networkOf(p.ip, p.mask)
   825→   825→    );
   826→   826→    if (nh) outPort = nh;
   827→   827→  }
   828→   828→  if (!outPort || !outPort.linkId || !outPort.adminUp) {
   829→   829→    step(ctx, d, inPortId, route.iface, null, null, packet, info(`${d.name}: منفذ الخروج ${route.iface} غير متصل`, `${d.name}: egress ${route.iface} not connected`), true);
   830→   830→    return;
   831→   831→  }
   832→   832→  const arpEntry = d.arp.find((a) => a.ip === nextIp);
   833→   833→  if (!arpEntry) {
   834→   834→    // trigger ARP resolution first
   835→   835→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: أحتاج MAC لـ ${nextIp} — سأرسل ARP أولاً`, `${d.name}: need MAC for ${nextIp} — sending ARP first`));
   836→   836→    ensureArpFor(ctx, d, outPort, nextIp, depth);
   837→   837→    const after = d.arp.find((a) => a.ip === nextIp);
   838→   838→    if (!after) {
   839→   839→      step(ctx, d, inPortId, route.iface, null, null, packet, info(`${d.name}: لا جواب ARP — الحزمة سقطت`, `${d.name}: no ARP reply — packet dropped`), true);
   840→   840→      return;
   841→   841→    }
   842→   842→    forwardRouted(ctx, d, inPortId, outPort, packet, after.mac, depth);
   843→   843→    return;
   844→   844→  }
   845→   845→  forwardRouted(ctx, d, inPortId, outPort, packet, arpEntry.mac, depth);
   846→   846→}

   848→   848→function forwardRouted(ctx: Ctx, d: Device, inPortId: string, outPort: Port, packet: Packet, dstMac: string, depth: number): void {
   849→   849→  // NAT (wireless router LAN→WAN)
   850→   850→  let fwd = packet;
   851→   851→  if (d.kind === "wirelessRouter" && d.nat && outPort.id === "g0/0" && inPortId !== "g0/0") {
   852→   852→    const wanIp = outPort.ip ?? "203.0.113.2";
   853→   853→    const lanPort = packet.srcPort ?? 0;
   854→   854→    let map = d.natTable.find((m) => m.lanIp === packet.srcIp && m.lanPort === lanPort);
   855→   855→    if (!map) {
   856→   856→      map = { lanIp: packet.srcIp, lanPort, wanPort: ++d.natWanCounter };
   857→   857→      d.natTable.push(map);
   858→   858→    }
   859→   859→    fwd = { ...packet, srcIp: wanIp, srcPort: map.wanPort || packet.srcPort, dstMac, srcMac: outPort.mac, ttl: packet.ttl - 1 };
   860→   860→    step(ctx, d, inPortId, outPort.id, null, null, fwd, info(`${d.name}: NAT — استبدال ${packet.srcIp}:${packet.srcPort ?? "-"} بـ ${wanIp}:${map.wanPort}`, `${d.name}: NAT — replacing ${packet.srcIp}:${packet.srcPort ?? "-"} with ${wanIp}:${map.wanPort}`));
   861→   861→    wireOut(ctx, d, outPort.id, fwd, depth);
   862→   862→    return;
   863→   863→  }
   864→   864→  // NAT reverse (WAN→LAN)
   865→   865→  if (d.kind === "wirelessRouter" && d.nat && inPortId === "g0/0" && outPort.id !== "g0/0") {
   866→   866→    const map = d.natTable.find((m) => m.wanPort === (packet.dstPort ?? 0));
   867→   867→    if (map) {
   868→   868→      fwd = { ...packet, dstIp: map.lanIp, dstPort: map.lanPort || packet.dstPort, srcMac: outPort.mac, ttl: packet.ttl - 1 };
   869→   869→      step(ctx, d, inPortId, outPort.id, null, null, fwd, info(`${d.name}: NAT عكسي — إعادة ${packet.dstIp}:${packet.dstPort ?? "-"} إلى ${map.lanIp}`, `${d.name}: reverse NAT — restoring ${packet.dstIp}:${packet.dstPort ?? "-"} to ${map.lanIp}`));
   870→   870→      wireOut(ctx, d, outPort.id, fwd, depth);
   871→   871→      return;
   872→   872→    }
   873→   873→  }
   874→   874→  const fwdPacket = mkPacket({ ...packet, proto: packet.proto, kind: packet.kind, kindBi: packet.kindBi, srcMac: outPort.mac, dstMac, ttl: packet.ttl - 1, srcPort: packet.srcPort, dstPort: packet.dstPort });
   875→   875→  step(ctx, d, inPortId, outPort.id, null, null, fwdPacket, info(`${d.name}: بحث في جدول التوجيه → ${networkOf(packet.dstIp, "255.255.255.255") ? packet.dstIp : ""} خارج ${outPort.name} (TTL-1)`, `${d.name}: routing table lookup → out ${outPort.name} (TTL-1)`));
   876→   876→  wireOut(ctx, d, outPort.id, fwdPacket, depth);
   877→   877→}

   888→   888→    ttl: 128,
   889→   889→    payload: { ar: `من يعرف ${targetIp}؟`, en: `Who has ${targetIp}?` },
   890→   890→  });

   892→   892→}

   895→   895→function dnsHandle(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
   896→   896→  if (packet.kind !== "DNS Query") return;
   897→   897→  const zone = d.dnsZone;
   898→   898→  const inP = findPort(d, inPortId);
   899→   899→  const out = (inP && inP.linkId && inP.adminUp ? inP : d.ports.find((p) => p.linkId && p.adminUp)) ?? d.ports[0];
   900→   900→  const name = packet.payload?.en ?? "";
   901→   901→  const answer = zone?.a[name];
   902→   902→  const reply = mkPacket({
   903→   903→    proto: "DNS",
   904→   904→    kind: "DNS Answer",
   905→   905→    kindBi: { ar: "جواب DNS", en: "DNS Answer" },
   906→   906→    srcMac: out.mac,
   907→   907→    dstMac: packet.srcMac,
   908→   908→    srcIp: packet.dstIp,
   909→   909→    dstIp: packet.srcIp,
   910→   910→    ttl: 128,
   911→   911→    srcPort: 53,
   912→   912→    dstPort: packet.srcPort,
   913→   913→    payload: answer
   914→   914→      ? { ar: `${name} = ${answer}`, en: `${name} = ${answer}` }
   915→   915→      : { ar: `${name} غير موجود (NXDOMAIN)`, en: `${name} not found (NXDOMAIN)` },
   916→   916→  });
   917→   917→  step(ctx, d, inPortId, out.id, null, null, reply, answer
   918→   918→    ? info(`${d.name}: جواب DNS — ${name} = ${answer}`, `${d.name}: DNS answer — ${name} = ${answer}`)
   919→   919→    : info(`${d.name}: لا يوجد سجل لـ ${name}`, `${d.name}: no record for ${name}`));
   920→   920→  wireOut(ctx, d, out.id, reply, depth);
   921→   921→}

   923→   923→function dhcpHostHandle(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
   924→   924→  const inP = findPort(d, inPortId);
   925→   925→  const out = (inP && inP.linkId && inP.adminUp ? inP : d.ports.find((p) => p.linkId && p.adminUp)) ?? d.ports[0];
   926→   926→  if (packet.kind === "DHCP Discover") {
   927→   927→    // server side
   928→   928→    const pool = d.dhcpPool;
   929→   929→    if (!pool || !pool.ifacePortId) {
   930→   930→      step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: لا أدير DHCP — تجاهل`, `${d.name}: not a DHCP server — ignoring`));
   931→   931→      return;
   932→   932→    }
   933→   933→    const ip = leaseIp(pool, packet.srcMac);
   934→   934→    const offer = mkPacket({
   935→   935→      proto: "DHCP",
   936→   936→      kind: "DHCP Offer",
   937→   937→      kindBi: { ar: "عرض DHCP", en: "DHCP Offer" },
   938→   938→      srcMac: out.mac,
   939→   939→      dstMac: packet.srcMac,
   940→   940→      srcIp: (findPort(d, pool.ifacePortId)?.ip) ?? "0.0.0.0",
   941→   941→      dstIp: ip,
   942→   942→      ttl: 128,
   943→   943→      srcPort: 67,
   944→   944→      dstPort: 68,
   945→   945→      payload: { ar: `أعرض ${ip} بوابة ${pool.gateway} قناع ${pool.mask}`, en: `Offering ${ip} gw ${pool.gateway} mask ${pool.mask}` },
   946→   946→    });
   947→   947→    step(ctx, d, inPortId, out.id, null, null, offer, info(`${d.name}: عرض DHCP بعنوان ${ip}`, `${d.name}: DHCP offer with ${ip}`));
   948→   948→    wireOut(ctx, d, out.id, offer, depth);
   949→   949→    return;
   950→   950→  }
   951→   951→  if (packet.kind === "DHCP Request" && d.dhcpPool) {
   952→   952→    const pool = d.dhcpPool;
   953→   953→    const m = packet.payload?.en.match(/\b(\d{1,3}(?:\.\d{1,3}){3})\b/);
   954→   954→    const ip = m?.[1] ?? leaseIp(pool, packet.srcMac);
   955→   955→    const ack = mkPacket({
   956→   956→      proto: "DHCP",
   957→   957→      kind: "DHCP Ack",
   958→   958→      kindBi: { ar: "تأكيد DHCP", en: "DHCP Ack" },
   959→   959→      srcMac: out.mac,
   960→   960→      dstMac: packet.srcMac,
   961→   961→      srcIp: (findPort(d, pool.ifacePortId)?.ip) ?? "0.0.0.0",
   962→   962→      dstIp: ip,
   963→   963→      ttl: 128,
   964→   964→      srcPort: 67,
   965→   965→      dstPort: 68,
   966→   966→      payload: { ar: `تأكيد ${ip} بوابة ${pool.gateway} DNS ${pool.dns ?? "-"}`, en: `Ack ${ip} gw ${pool.gateway} dns ${pool.dns ?? "-"}` },
   967→   967→    });
   968→   968→    step(ctx, d, inPortId, out.id, null, null, ack, info(`${d.name}: تأكيد DHCP — ${ip} أصبح لعنوان ${packet.srcMac}`, `${d.name}: DHCP ack — ${ip} assigned to ${packet.srcMac}`));
   969→   969→    wireOut(ctx, d, out.id, ack, depth);
   970→   970→    return;
   971→   971→  }
   972→   972→  // client side
   973→   973→  if (packet.kind === "DHCP Offer" && d.dhcpClient) {
   974→   974→    const ip = packet.dstIp;
   975→   975→    const req = mkPacket({
   976→   976→      proto: "DHCP",
   977→   977→      kind: "DHCP Request",
   978→   978→      kindBi: { ar: "طلب DHCP", en: "DHCP Request" },
   979→   979→      srcMac: out.mac,
   980→   980→      dstMac: BROADCAST_MAC,
   981→   981→      srcIp: ZERO_IP,
   982→   982→      dstIp: BROADCAST_IP,
   983→   983→      ttl: 128,
   984→   984→      srcPort: 68,
   985→   985→      dstPort: 67,
   986→   986→      payload: { ar: `أطلب ${ip}`, en: `Requesting ${ip}` },
   987→   987→    });
   988→   988→    step(ctx, d, inPortId, out.id, null, null, req, info(`${d.name}: طلب العنوان ${ip}`, `${d.name}: requesting ${ip}`));
   989→   989→    (d as Device & { _offered?: string })._offered = ip;
   990→   990→    wireOut(ctx, d, out.id, req, depth);
   991→   991→    return;
   992→   992→  }
   993→   993→  if (packet.kind === "DHCP Ack" && d.dhcpClient) {
   994→   994→    const ip = packet.dstIp;
   995→   995→    const port = out;
   996→   996→    port.ip = ip;
   997→   997→    port.mask = "255.255.255.0";
   998→   998→    const gwMatch = packet.payload?.ar.match(/بوابة\s*(\S+)/);
   999→   999→    const dnsMatch = packet.payload?.ar.match(/DNS\s*(\S+)/);
  1000→  1000→    d.gateway = gwMatch?.[1] ?? null;
  1001→  1001→    d.dnsServer = dnsMatch?.[1] && dnsMatch[1] !== "-" ? dnsMatch[1] : null;
  1002→  1002→    delete (d as Device & { _offered?: string })._offered;
  1003→  1003→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: استلمت ${ip} عبر DHCP — تم الإعداد!`, `${d.name}: got ${ip} via DHCP — configured!`));
  1004→  1004→    return;
  1005→  1005→  }
  1006→  1006→}

  1008→  1008→function leaseIp(pool: import("./types").DhcpPool, mac: string): string {
  1009→  1009→  const existing = pool.leases.find((l) => l.mac === mac);
  1010→  1010→  if (existing) return existing.ip;
  1011→  1011→  const from = ipToInt(pool.from);
  1012→  1012→  const to = ipToInt(pool.to);
  1013→  1013→  const used = new Set(pool.leases.map((l) => l.ip));
  1014→  1014→  for (let i = from; i <= to; i++) {
  1015→  1015→    const cand = intToIp(i);
  1016→  1016→    if (!used.has(cand)) {
  1017→  1017→      pool.leases.push({ mac, ip: cand });
  1018→  1018→      return cand;
  1019→  1019→    }
  1020→  1020→  }
  1021→  1021→  return pool.from;
  1022→  1022→}

  1024→  1024→function httpHandle(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
  1025→  1025→  const inP0 = findPort(d, inPortId);

  1235→  1235→export function aclCheck(d: Device, packet: Packet): { action: "permit" | "deny"; rule: import("./types").AclRule | null } {
  1236→  1236→  const rules = d.acls ?? [];

  1243→  1243→function matchAcl(r: import("./types").AclRule, packet: Packet): boolean {
  1244→  1244→  if (r.src !== "any" && r.srcMask !== "any" && !ipInSubnet(packet.srcIp, r.src, r.srcMask)) return false;
  1245→  1245→  if (r.dst !== "any" && r.dstMask !== "any" && !ipInSubnet(packet.dstIp, r.dst, r.dstMask)) return false;
  1246→  1246→  if (r.proto !== "any") {
  1247→  1247→    const protoName =
  1248→  1248→      packet.proto === "ICMP" ? "icmp"
  1249→  1249→        : packet.proto === "TCP" || packet.proto === "HTTP" ? "tcp"
  1250→  1250→          : packet.proto === "DNS" || packet.proto === "DHCP" ? "udp"
  1251→  1251→            : "any";
  1252→  1252→    if (r.proto !== protoName) return false;
  1253→  1253→    if (r.port != null && packet.dstPort !== r.port) return false;
  1254→  1254→  }
  1255→  1255→  return true;
  1256→  1256→}

  1263→  1263→/** IDS sensor inspection: ARP-spoof binding conflicts, floods & scans */
  1264→  1264→function idsInspect(ctx: Ctx, d: Device, inPortId: string, packet: Packet): void {
  1265→  1265→  d.statsIn = (d.statsIn ?? 0) + 1;
  1266→  1266→  // IP↔MAC binding DB: forged ARP replies that contradict a known binding = spoof
  1267→  1267→  if (packet.proto === "ARP" && packet.kind === "ARP Reply" && isValidIp(packet.srcIp) && packet.srcIp !== ZERO_IP) {
  1268→  1268→    const prev = ctx.ipMac.get(packet.srcIp);
  1269→  1269→    if (prev && prev !== packet.srcMac) {
  1270→  1270→      pushAlert(d, "arp-spoof", "critical", packet.srcIp, packet.dstIp,
  1271→  1271→        `تسميم ARP مكتشف: ${packet.srcIp} كان ${prev} والآن يدّعي ${packet.srcMac} — مصدره منفذ ${inPortId}`,
  1272→  1272→        `ARP poisoning detected: ${packet.srcIp} was ${prev} but now claims ${packet.srcMac} — seen on port ${inPortId}`);
  1273→  1273→      return;
  1274→  1274→    }
  1275→  1275→    if (!prev) ctx.ipMac.set(packet.srcIp, packet.srcMac);
  1276→  1276→  }
  1277→  1277→  if (!isValidIp(packet.srcIp) || packet.srcIp === ZERO_IP) return;
  1278→  1278→  // flow statistics per source
  1279→  1279→  const fs = ctx.flow.get(packet.srcIp) ?? { echo: 0, syn: 0, dsts: [] };

  1281→  1281→  if (packet.kind === "TCP SYN") fs.syn += 1;
  1282→  1282→  if (!fs.dsts.includes(packet.dstIp)) fs.dsts.push(packet.dstIp);
  1283→  1283→  ctx.flow.set(packet.srcIp, fs);
  1284→  1284→  if (fs.echo === 8) {

  1286→  1286→      `إغراق ICMP: 8+ طلبات Echo متتالية من ${packet.srcIp} — نمط هجوم DDoS`,
  1287→  1287→      `ICMP flood: 8+ consecutive echo requests from ${packet.srcIp} — DDoS attack pattern`);
  1288→  1288→  }
  1289→  1289→  if (fs.syn === 8) {
  1290→  1290→    pushAlert(d, "syn-flood", "critical", packet.srcIp, packet.dstIp,
  1291→  1291→      `إغراق SYN: 8+ محاولات اتصال نصف مفتوحة من ${packet.srcIp} إلى ${packet.dstIp}`,
  1292→  1292→      `SYN flood: 8+ half-open connection attempts from ${packet.srcIp} to ${packet.dstIp}`);
  1293→  1293→  }
  1294→  1294→  if (fs.dsts.length === 5) {
  1295→  1295→    pushAlert(d, "scan", "warn", packet.srcIp, packet.dstIp,
  1296→  1296→      `مسح شبكة مكتشف: ${packet.srcIp} يرمي حزماً نحو 5 عناوين مختلفة — نمط Port/Host Scan`,
  1297→  1297→      `Network scan detected: ${packet.srcIp} probing 5 distinct addresses — host/port scan pattern`);
  1298→  1298→  }

  1302→  1302→function l3Route(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
  1303→  1303→  if (packet.proto === "ARP" && packet.kind === "ARP Request") {
  1304→  1304→    addArp(d, packet.srcIp, packet.srcMac, inPortId);
  1305→  1305→    const svi = d.ports.find((p) => p.kind === "svi" && p.ip === packet.dstIp);
  1306→  1306→    if (svi && svi.ip) {
  1307→  1307→      const reply = mkPacket({
  1308→  1308→        proto: "ARP", kind: "ARP Reply", kindBi: { ar: "رد ARP (SVI)", en: "ARP Reply (SVI)" },
  1309→  1309→        srcMac: svi.mac, dstMac: packet.srcMac, srcIp: svi.ip, dstIp: packet.srcIp, ttl: 128,
  1310→  1310→        payload: { ar: `بوابة VLAN أنا ${svi.ip}`, en: `I am the VLAN gateway ${svi.ip}` },
  1311→  1311→      });
  1312→  1312→      step(ctx, d, inPortId, inPortId, null, null, reply, info(`${d.name}: SVI ${svi.id} يرد على ARP (${svi.ip})`, `${d.name}: SVI ${svi.id} answers ARP (${svi.ip})`));
  1313→  1313→      wireOut(ctx, d, inPortId, reply, depth);
  1314→  1314→    }
  1315→  1315→    return;
  1316→  1316→  }
  1317→  1317→  if (packet.proto === "ARP" && packet.kind === "ARP Reply") {
  1318→  1318→    addArp(d, packet.srcIp, packet.srcMac, inPortId);
  1319→  1319→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: تعلمت ${packet.srcIp} = ${packet.srcMac} في جدول ARP`, `${d.name}: learned ${packet.srcIp} = ${packet.srcMac} in ARP table`));
  1320→  1320→    return;
  1321→  1321→  }
  1322→  1322→  const sviMe = d.ports.find((p) => p.kind === "svi" && p.ip === packet.dstIp);
  1323→  1323→  if (packet.proto === "ICMP" && packet.kind === "Echo Request" && sviMe && sviMe.ip) {
  1324→  1324→    const reply = mkPacket({
  1325→  1325→      proto: "ICMP", kind: "Echo Reply", kindBi: { ar: "رد ICMP", en: "ICMP Echo Reply" },
  1326→  1326→      srcMac: sviMe.mac, dstMac: packet.srcMac, srcIp: sviMe.ip, dstIp: packet.srcIp, ttl: 128,
  1327→  1327→    });
  1328→  1328→    step(ctx, d, inPortId, inPortId, null, null, reply, info(`${d.name}: SVI ${sviMe.id} هو الهدف — رد Pong`, `${d.name}: SVI ${sviMe.id} is the target — pong reply`));
  1329→  1329→    wireOut(ctx, d, inPortId, reply, depth);
  1330→  1330→    return;
  1331→  1331→  }
  1332→  1332→  if (packet.proto === "DHCP" || packet.proto === "DNS") return; // still bridge-serviced below
  1333→  1333→  if (packet.dstMac !== BROADCAST_MAC) {
  1334→  1334→    l3Forward(ctx, d, inPortId, packet, depth);
  1335→  1335→  }
  1336→  1336→}

  1338→  1338→function l3Forward(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
  1339→  1339→  if (packet.ttl <= 1) {
  1340→  1340→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: TTL انتهى — إسقاط`, `${d.name}: TTL expired — dropped`), true);
  1341→  1341→    return;
  1342→  1342→  }
  1343→  1343→  const route = lookupRoute(d, packet.dstIp);
  1344→  1344→  if (!route) {
  1345→  1345→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: لا مسار إلى ${packet.dstIp} — فعّل ip routing واضبط SVI`, `${d.name}: no route to ${packet.dstIp} — configure an SVI or add a route`), true);
  1346→  1346→    return;
  1347→  1347→  }
  1348→  1348→  // resolve egress SVI by route iface or by subnet match
  1349→  1349→  let svi = route.iface ? d.ports.find((p) => p.id === route.iface && p.kind === "svi") : undefined;
  1350→  1350→  if (!svi) {
  1351→  1351→    svi = d.ports.find((p) => p.kind === "svi" && p.ip && p.mask && networkOf(packet.dstIp, p.mask) === networkOf(p.ip, p.mask));
  1352→  1352→  }
  1353→  1353→  if (!svi || !svi.ip) {
  1354→  1354→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: لا SVI لشبكة ${packet.dstIp}`, `${d.name}: no SVI for ${packet.dstIp} subnet`), true);
  1355→  1355→    return;
  1356→  1356→  }
  1357→  1357→  const vlan = parseInt(svi.id.replace("vlan", ""), 10) || 1;
  1358→  1358→  const nextIp = route.nextHop ?? packet.dstIp;
  1359→  1359→  let arpEntry = d.arp.find((a) => a.ip === nextIp);
  1360→  1360→  if (!arpEntry) {
  1361→  1361→    arpBroadcastOnVlan(ctx, d, vlan, svi, nextIp, depth);
  1362→  1362→    arpEntry = d.arp.find((a) => a.ip === nextIp);
  1363→  1363→    if (!arpEntry) {
  1364→  1364→      step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: لا جواب ARP في VLAN${vlan} — الحزمة سقطت`, `${d.name}: no ARP reply in VLAN${vlan} — packet dropped`), true);
  1365→  1365→      return;
  1366→  1366→    }
  1367→  1367→  }
  1368→  1368→  // pick the physical port where the destination MAC lives (or flood the vlan)
  1369→  1369→  const entry = d.macTable.find((e) => e.mac === arpEntry.mac && e.vlan === vlan);
  1370→  1370→  const phys = d.ports.filter((p) => p.kind === "ethernet" && p.linkId && p.adminUp && p.id !== inPortId && (p.trunk || p.accessVlan === vlan));
  1371→  1371→  const targets = entry ? phys.filter((p) => p.id === entry.portId) : phys;
  1372→  1372→  if (targets.length === 0) {
  1373→  1373→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: لا منافذ في VLAN${vlan} للخروج`, `${d.name}: no egress ports in VLAN${vlan}`), true);
  1374→  1374→    return;
  1375→  1375→  }
  1376→  1376→  for (const p of targets) {
  1377→  1377→    const fwd = mkPacket({ ...packet, srcMac: p.mac, dstMac: arpEntry.mac, ttl: packet.ttl - 1, vlan: p.trunk ? vlan : null });
  1378→  1378→    step(ctx, d, inPortId, p.id, null, null, fwd, info(`${d.name}: توجيه بين VLANs → خارج ${p.name} نحو ${arpEntry.mac} (TTL-1)`, `${d.name}: inter-VLAN route → out ${p.name} toward ${arpEntry.mac} (TTL-1)`));
  1379→  1379→    wireOut(ctx, d, p.id, fwd, depth);
  1380→  1380→  }
  1381→  1381→}

  1383→  1383→function arpBroadcastOnVlan(ctx: Ctx, d: Device, vlan: number, svi: Port, targetIp: string, depth: number): void {
