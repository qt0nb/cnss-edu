   111→  1008→function leaseIp(pool: import("./types").DhcpPool, mac: string): string {
   112→  1009→  const existing = pool.leases.find((l) => l.mac === mac);
   113→  1010→  if (existing) return existing.ip;
   114→  1011→  const from = ipToInt(pool.from);
   115→  1012→  const to = ipToInt(pool.to);
   116→  1013→  const used = new Set(pool.leases.map((l) => l.ip));
   117→  1014→  for (let i = from; i <= to; i++) {
   118→  1015→    const cand = intToIp(i);
   119→  1016→    if (!used.has(cand)) {
   120→  1017→      pool.leases.push({ mac, ip: cand });
   121→  1018→      return cand;
   122→  1019→    }
   123→  1020→  }
   124→  1021→  return pool.from;
   125→  1022→}

   127→  1024→function httpHandle(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
   128→  1025→  const inP0 = findPort(d, inPortId);
   129→  1026→  const out = (inP0 && inP0.linkId && inP0.adminUp ? inP0 : d.ports.find((p) => p.linkId && p.adminUp)) ?? d.ports[0];

   150→  1047→    return;
   151→  1048→  }
   152→  1049→  if (packet.kind === "HTTP GET" && forMeIp) {
   153→  1050→    const page = d.httpRoot;
   154→  1051→    const ok = mkPacket({ proto: "HTTP", kind: "HTTP 200 OK", kindBi: { ar: "رد HTTP 200", en: "HTTP 200 OK" }, srcMac: out.mac, dstMac: packet.srcMac, srcIp: myIp, dstIp: packet.srcIp, ttl: 128, srcPort: 80, dstPort: packet.srcPort, payload: { ar: `${page.title}\n${page.body}`, en: `${page.title}\n${page.body}` } });

   156→  1053→    wireOut(ctx, d, out.id, ok, depth);
   157→  1054→    return;
   158→  1055→  }
   159→  1056→  step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: استلم حزمة ${packet.kind}`, `${d.name}: received ${packet.kind}`));
   160→  1057→}

   199→  1096→    }
   200→  1097→  }
   201→  1098→  const echo = mkPacket({ proto: "ICMP", kind: "Echo Request", kindBi: { ar: "طلب ICMP (Ping)", en: "ICMP Echo Request" }, srcMac: outPort.mac, dstMac: arpEntry.mac, srcIp: ip, dstIp, ttl: 128 });

   205→  1102→  return {
   206→  1103→    steps: ctx.steps,
   207→  1104→    devices: t.devices,
   208→  1105→    success: replied,
   209→  1106→    note: replied
   210→  1107→      ? info("نجح Ping! رأيت الطلب والرد يمران عبر كل الأجهزة", "Ping succeeded! You saw request & reply traverse every device")
   211→  1108→      : info("فشل Ping — راجع الخطوات المظللة بالأحمر", "Ping failed — review the red-marked steps"),
   212→  1109→  };
   213→  1110→}

   255→  1152→export function simulateHttp(topo: Topology, clientId: string, host: string): SimResult & { page: { title: string; body: string } | null } {

   276→  1173→    }

   338→  1235→export function aclCheck(d: Device, packet: Packet): { action: "permit" | "deny"; rule: import("./types").AclRule | null } {
   339→  1236→  const rules = d.acls ?? [];

   341→  1238→    if (matchAcl(r, packet)) return { action: r.action, rule: r };

   346→  1243→function matchAcl(r: import("./types").AclRule, packet: Packet): boolean {
   347→  1244→  if (r.src !== "any" && r.srcMask !== "any" && !ipInSubnet(packet.srcIp, r.src, r.srcMask)) return false;
   348→  1245→  if (r.dst !== "any" && r.dstMask !== "any" && !ipInSubnet(packet.dstIp, r.dst, r.dstMask)) return false;
   349→  1246→  if (r.proto !== "any") {
   350→  1247→    const protoName =
   351→  1248→      packet.proto === "ICMP" ? "icmp"
   352→  1249→        : packet.proto === "TCP" || packet.proto === "HTTP" ? "tcp"
   353→  1250→          : packet.proto === "DNS" || packet.proto === "DHCP" ? "udp"
   354→  1251→            : "any";
   355→  1252→    if (r.proto !== protoName) return false;
   356→  1253→    if (r.port != null && packet.dstPort !== r.port) return false;
   357→  1254→  }
   358→  1255→  return true;
   359→  1256→}
