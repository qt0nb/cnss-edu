    16→export const ipToInt = (ip: string): number =>
    17→  ip.split(".").reduce((s, o) => (s << 8) + (parseInt(o, 10) || 0), 0) >>> 0;
    18→
    19→export const intToIp = (n: number): string =>
    20→  [(n >>> 24) & 255, (n >>> 16) & 255, (n >>> 8) & 255, n & 255].join(".");
    21→
    22→export const prefixToMask = (p: number): string =>
    23→  p === 0 ? "0.0.0.0" : intToIp(0xffffffff << (32 - p) >>> 0);

    25→export const maskToPrefix = (mask: string): number => {
    26→  const bits = ipToInt(mask).toString(2);
    27→  const m = bits.match(/^1*0*$/);
    28→  if (!m || bits.length !== 32) return -1;
    29→  return (bits.match(/1/g) || []).length;
    30→};

    51→  _mac += 1;
    52→  return `00:1B:0C:${((_mac >> 16) & 255).toString(16).padStart(2, "0").toUpperCase()}:${((_mac >> 8) & 255).toString(16).padStart(2, "0").toUpperCase()}:${(_mac & 255).toString(16).padStart(2, "0").toUpperCase()}`;

    54→export function resetMacCounterFor(topology?: Topology) {
    55→  let max = 0;
    56→  for (const d of topology?.devices ?? []) for (const p of d.ports) {
    57→    const parts = p.mac.split(":").map((h) => parseInt(h, 16));
    58→    max = Math.max(max, (parts[3] << 16) | (parts[4] << 8) | parts[5]);
    59→  }
    60→  _mac = Math.max(_mac, max);

   182→export const linkOf = (t: Topology, port: Port | null): { link: Link; otherDev: Device; otherPort: Port } | null => {
   183→  if (!port || !port.linkId) return null;
   184→  const link = t.links.find((l) => l.id === port.linkId);
   185→  if (!link) return null;
   186→  const aDev = findDevice(t, link.a.deviceId);
   187→  const bDev = findDevice(t, link.b.deviceId);
   188→  if (!aDev || !bDev) return null;
   189→  const isA = link.a.deviceId === (findDeviceByPort(t, port) ?? aDev).id;
   190→  const other = isA ? bDev : aDev;
   191→  const otherPort = findPort(other, isA ? link.b.portId : link.a.portId)!;
   192→  return { link, otherDev: other, otherPort };
   193→};

   195→function findDeviceByPort(t: Topology, port: Port): Device | undefined {

   199→// simpler, portability-safe link resolution by (device, port)
   200→export function resolveLink(
   201→  t: Topology,
   202→  deviceId: string,
   203→  portId: string
   204→): { link: Link; otherDevId: string; otherPortId: string } | null {
   205→  const dev = findDevice(t, deviceId);
   206→  const port = dev && findPort(dev, portId);
   207→  if (!dev || !port || !port.linkId) return null;
   208→  const link = t.links.find((l) => l.id === port.linkId);
   209→  if (!link) return null;
   210→  if (link.a.deviceId === deviceId && link.a.portId === portId)
   211→    return { link, otherDevId: link.b.deviceId, otherPortId: link.b.portId };
   212→  if (link.b.deviceId === deviceId && link.b.portId === portId)
   213→    return { link, otherDevId: link.a.deviceId, otherPortId: link.a.portId };
   214→  return null;
   215→}

   217→export function connectPorts(
   218→  t: Topology,
   219→  aDevId: string,
   220→  aPortId: string,
   221→  bDevId: string,
   222→  bPortId: string,
   223→  kind: LinkKind
   224→): Link | null {
   225→  const aDev = findDevice(t, aDevId);
   226→  const bDev = findDevice(t, bDevId);
   227→  const aPort = aDev && findPort(aDev, aPortId);
   228→  const bPort = bDev && findPort(bDev, bPortId);
   229→  if (!aDev || !bDev || !aPort || !bPort) return null;
   230→  if (aPort.linkId || bPort.linkId) return null;
   231→  const link: Link = {
   232→    id: `lk${Date.now().toString(36)}${Math.floor(Math.random() * 999)}`,
   233→    kind,
   234→    a: { deviceId: aDevId, portId: aPortId },
   235→    b: { deviceId: bDevId, portId: bPortId },
   236→  };
   237→  t.links.push(link);
   238→  aPort.linkId = link.id;
   239→  bPort.linkId = link.id;
   240→  return link;
   241→}

   243→const SWITCHY: DeviceKind[] = ["switch", "hub", "cloud", "ap"];
   244→export function autoCable(
   245→  t: Topology,
   246→  aDevId: string,
   247→  bDevId: string
   248→): { aPortId: string; bPortId: string; kind: LinkKind } | null {
   249→  const a = findDevice(t, aDevId);
   250→  const b = findDevice(t, bDevId);
   251→  if (!a || !b || a.id === b.id) return null;
   252→  // wireless: both have free wireless ports + same ssid
   253→  const aw = a.ports.find((p) => p.kind === "wireless" && !p.linkId && p.adminUp);
   254→  const bw = b.ports.find((p) => p.kind === "wireless" && !p.linkId && p.adminUp);
   255→  const aWirelessOnly = a.kind === "smartphone";
   256→  const bWirelessOnly = b.kind === "smartphone";
   257→  if ((aw && bw) && (aWirelessOnly || bWirelessOnly || a.kind === "ap" || b.kind === "ap" || a.kind === "wirelessRouter" || b.kind === "wirelessRouter")) {
   258→    return { aPortId: aw.id, bPortId: bw.id, kind: "wireless" };
   259→  }
   260→  const serial = (d: Device) => d.ports.find((p) => p.kind === "serial" && !p.linkId && p.adminUp);
   261→  if (a.kind === "router" && b.kind === "router") {
   262→    const sa = serial(a);
   263→    const sb = serial(b);
   264→    if (sa && sb) return { aPortId: sa.id, bPortId: sb.id, kind: "serial" };
   265→  }
   266→  const eth = (d: Device) =>
   267→    d.ports.find((p) => (p.kind === "ethernet" || (p.kind === "internet" && d.kind === "wirelessRouter")) && !p.linkId && p.adminUp);
   268→  const ae = eth(a);
   269→  const be = eth(b);
   270→  if (!ae || !be) return null;
   271→  const aIsSwitchy = SWITCHY.includes(a.kind) || (a.kind === "wirelessRouter" && ae.id !== "g0/0");
   272→  const bIsSwitchy = SWITCHY.includes(b.kind) || (b.kind === "wirelessRouter" && be.id !== "g0/0");
   273→  const kind: LinkKind = aIsSwitchy !== bIsSwitchy ? "copper" : "crossover";
   274→  return { aPortId: ae.id, bPortId: be.id, kind };
   275→}

   280→function mkLayers(p: Omit<Packet, "layers" | "id">, vlan?: number | null): PduLayer[] {
   281→  const layers: PduLayer[] = [];
   282→  if (p.proto === "DHCP") {
   283→    layers.push({ n: 7, title: { ar: "طبقة التطبيق — DHCP", en: "Application — DHCP" }, fields: [ { k: { ar: "الرسالة", en: "Message" }, v: p.kind }, { k: { ar: "المحتوى", en: "Payload" }, v: p.payload ? `${p.payload.en}` : "-" } ] });
   284→  }
   285→  if (p.proto === "DNS") {
   286→    layers.push({ n: 7, title: { ar: "طبقة التطبيق — DNS", en: "Application — DNS" }, fields: [ { k: { ar: "الاستعلام", en: "Query" }, v: p.kind }, { k: { ar: "الاسم", en: "Name" }, v: p.payload ? p.payload.en : "-" } ] });
   287→  }
   288→  if (p.proto === "HTTP") {
   289→    layers.push({ n: 7, title: { ar: "طبقة التطبيق — HTTP", en: "Application — HTTP" }, fields: [ { k: { ar: "الرسالة", en: "Message" }, v: p.kind }, { k: { ar: "المحتوى", en: "Content" }, v: p.payload ? p.payload.en : "-" } ] });
   290→  }
   291→  if (p.srcPort !== undefined) {
   292→    layers.push({ n: 4, title: { ar: "طبقة النقل", en: "Transport Layer" }, fields: [ { k: { ar: "البروتوكول", en: "Protocol" }, v: p.proto === "TCP" || p.proto === "HTTP" ? "TCP" : "UDP" }, { k: { ar: "المنفذ المصدر", en: "Src Port" }, v: String(p.srcPort) }, { k: { ar: "المنفذ الهدف", en: "Dst Port" }, v: String(p.dstPort) } ] });
   293→  } else if (p.proto === "ICMP") {
   294→    layers.push({ n: 4, title: { ar: "طبقة النقل — ICMP", en: "Transport — ICMP" }, fields: [ { k: { ar: "النوع", en: "Type" }, v: p.kind === "Echo Request" ? "8 (Echo Request)" : "0 (Echo Reply)" } ] });
   295→  }
   296→  layers.push({ n: 3, title: { ar: "طبقة الشبكة — IP", en: "Network — IP" }, fields: [ { k: { ar: "عنوان المصدر", en: "Source IP" }, v: p.srcIp }, { k: { ar: "عنوان الهدف", en: "Dest IP" }, v: p.dstIp }, { k: { ar: "TTL", en: "TTL" }, v: String(p.ttl) }, ...(p.proto === "ICMP" ? [] : [{ k: { ar: "البروتوكول", en: "Protocol" }, v: p.proto === "ARP" ? "-" : p.srcPort !== undefined ? (p.proto === "TCP" || p.proto === "HTTP" ? "TCP" : "UDP") : "ICMP" }]) ] });
   297→  layers.push({ n: 2, title: { ar: "طبقة وصلة البيانات — إيثرنت", en: "Data Link — Ethernet" }, fields: [ { k: { ar: "MAC المصدر", en: "Src MAC" }, v: p.srcMac }, { k: { ar: "MAC الهدف", en: "Dst MAC" }, v: p.dstMac }, { k: { ar: "EtherType", en: "EtherType" }, v: p.proto === "ARP" ? ETHERTYPE.ARP : ETHERTYPE.IP }, ...(vlan ? [ { k: { ar: "وسم 802.1Q", en: "802.1Q Tag" }, v: `VLAN ${vlan} (+4 bytes)` } ] : []) ] });
   298→  layers.push({ n: 1, title: { ar: "الطبقة الفيزيائية", en: "Physical Layer" }, fields: [ { k: { ar: "الوسيط", en: "Medium" }, v: "bits — إشارات كهربائية/ضوئية" } ] });
   299→  return layers.reverse().slice().sort((a, b) => b.n - a.n); // L7 on top like PT
   300→}

   356→function egressPorts(ctx: Ctx, d: Device, inPortId: string, dstMac: string, ingressVlan: number): { port: Port; vlan: number }[] {
   357→  if (d.kind === "hub") {
   358→    return d.ports
   359→      .filter((p) => p.id !== inPortId && p.linkId && p.adminUp)
   360→      .map((p) => ({ port: p, vlan: ingressVlan }));
   361→  }
   362→  if (isBridge(d) || d.kind === "wirelessRouter") {
   363→    const bridgePorts = d.kind === "wirelessRouter" ? wrLanPorts(d) : d.ports;
   364→    const inPort = d.ports.find((p) => p.id === inPortId)!;
   365→    const inIsTrunk = inPort.trunk;
   366→    const up = (p: Port) => p.linkId && p.adminUp && p.id !== inPortId;
   367→    if (dstMac === BROADCAST_MAC) {
   368→      return bridgePorts
   369→        .filter((p) => up(p) && (inIsTrunk || p.trunk || p.accessVlan === ingressVlan))
   370→        .map((p) => ({ port: p, vlan: p.trunk ? ingressVlan : p.accessVlan === ingressVlan ? ingressVlan : ingressVlan }));
   371→    }
   372→    const entry = d.macTable.find((e) => e.mac === dstMac && (e.vlan === ingressVlan || inIsTrunk));
   373→    if (entry) {
   374→      const p = bridgePorts.find((pp) => pp.id === entry.portId);
   375→      if (p && up(p)) {
   376→        const dstIsTrunk = p.trunk;
   377→        if (!dstIsTrunk && !inIsTrunk && p.accessVlan !== ingressVlan) {
   378→          step(ctx, d, inPortId, p.id, null, null, mkPacket({ proto: "ARP", kind: "Dropped", kindBi: { ar: "إطار مُسقط", en: "Frame dropped" }, srcMac: dstMac, dstMac, srcIp: ZERO_IP, dstIp: ZERO_IP, ttl: 0 }), info("أُسقط الإطار: المنفذ في VLAN مختلف", "Frame dropped: port is in a different VLAN"), true);
   379→          return [];
   380→        }
   381→        return [{ port: p, vlan: dstIsTrunk ? ingressVlan : ingressVlan }];
   382→      }
   383→      return [];
   384→    }
   385→    // unknown unicast → flood
   386→    return bridgePorts
   387→      .filter((p) => up(p) && (inIsTrunk || p.trunk || p.accessVlan === ingressVlan))
   388→      .map((p) => ({ port: p, vlan: p.trunk ? ingressVlan : ingressVlan }));
   389→  }
   390→  return [];
   391→}

   393→// send a frame out of a device port onto the wire
   394→function wireOut(
   395→  ctx: Ctx,
   396→  fromDev: Device,
   397→  fromPortId: string,
   398→  packet: Packet,
   399→  depth = 0
   400→): void {
   401→  if (depth > 40 || ctx.steps.length > 220) {
   402→    step(ctx, fromDev, null, fromPortId, null, null, packet, info("تم إيقاف المحاكاة: حلقة أو عمق كبير — يحتاج STP", "Simulation stopped: loop or excessive depth — STP needed"), true);
   403→    return;
   404→  }
   405→  const res = resolveLink(ctx.t, fromDev.id, fromPortId);
   406→  const port = findPort(fromDev, fromPortId);
   407→  if (!res || !port || !port.adminUp) {
   408→    step(ctx, fromDev, null, fromPortId, null, null, packet, info("الحزمة سقطت: المنفذ غير متصل أو مغلق (shutdown)", "Packet dropped: port not connected or shut down"), true);
   409→    return;
   410→  }
   411→  const outPacket = packet.vlan !== undefined && packet.vlan !== null ? packet : packet;
   412→  step(ctx, fromDev, null, fromPortId, res.otherDevId, res.link.id, outPacket, info("", ""));
   413→  const other = findDevice(ctx.t, res.otherDevId);
   414→  if (!other) return;
   415→  deliver(ctx, other, res.otherPortId, packet, depth + 1);
   416→}

   418→function ingressVlanOf(d: Device, portId: string): number {
   419→  const p = findPort(d, portId);
   420→  if (!p) return 1;
   421→  return p.trunk ? -1 : p.accessVlan; // -1 = came tagged
   422→}

   424→function deliver(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
   425→  const key = `${d.id}|${inPortId}|${packet.id}`;
   426→  if (ctx.seen.has(key)) {
   427→    step(ctx, d, inPortId, null, null, null, packet, info("حزمة تدور في حلقة (Loop) — في الواقع يمنعها STP", "Packet looping — real switches prevent this with STP"), true);
   428→    return;
   429→  }
   430→  ctx.seen.add(key);
   431→
   432→  // VLAN ingress check on access ports (broadcast tagged?)
   433→  const inPort = findPort(d, inPortId);
   434→  if (!inPort) return;
   435→
   436→  // ── bridges (switch/hub/cloud/ap) and wireless-router LAN side
   437→  if (isBridge(d) || (d.kind === "wirelessRouter" && inPortId !== "g0/0")) {
   438→    const vlan = inPort.trunk ? (packet.vlan ?? 1) : inPort.accessVlan;
   439→    if (d.kind !== "hub") learnMac(ctx, d, inPortId, packet.srcMac, vlan);
   440→    const outs = egressPorts(ctx, d, inPortId, packet.dstMac, vlan);
   441→    if (outs.length === 0) {
   442→      // host attached? just no egress; record device entry info
   443→      if (packet.dstMac === BROADCAST_MAC) {
   444→        step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: بثّ الإطار ولا منافذ خروج أخرى`, `${d.name}: broadcast received, no other egress ports`));
   445→      }
   446→    }
   447→    for (const { port, vlan: outVlan } of outs) {
   448→      const tagged = port.trunk ? vlan : null;
   449→      const pkt = tagged !== null ? { ...packet, vlan: tagged } : { ...packet, vlan: null };
   450→      const localInfo =
   451→        packet.dstMac === BROADCAST_MAC
   452→          ? info(`${d.name}: غمر البث من ${inPort.name} إلى ${port.name}`, `${d.name}: flooding broadcast from ${inPort.name} to ${port.name}`)
   453→          : info(`${d.name}: تعلم ${packet.srcMac} على ${inPort.name} ثم التوجيه إلى ${port.name}`, `${d.name}: learned ${packet.srcMac} on ${inPort.name}, forwarding out ${port.name}`);
   454→      step(ctx, d, inPortId, port.id, null, null, pkt, localInfo);
   455→      wireOut(ctx, d, port.id, pkt, depth);
   456→    }
   457→    // wireless-router LAN side: packets for the router itself (broadcast DHCP/DNS/ARP or unicast to its MAC/IP)
   458→    // are also serviced locally — DHCP pool, DNS zone, ARP replies, ICMP, HTTP — like a real home router
   459→    if (d.kind === "wirelessRouter") {
   460→      const toMe =
   461→        packet.dstMac === BROADCAST_MAC ||
   462→        d.ports.some((p) => p.mac === packet.dstMac) ||
   463→        d.ports.some((p) => p.ip && p.ip === packet.dstIp);
   464→      const serviceable =
   465→        packet.proto === "DHCP" ||
   466→        packet.proto === "ARP" ||
   467→        (packet.proto === "DNS" && packet.kind === "DNS Query") ||
   468→        packet.proto === "ICMP" ||
   469→        packet.proto === "HTTP" ||
   470→        packet.proto === "TCP";
   471→      if (toMe && serviceable) {
   472→        routerHandle(ctx, d, inPortId, packet, depth);
   473→      }
   474→    }
   475→    // if a host/server is directly the target of this bridge's delivery, it arrives via wireOut→deliver on the host
   476→    return;
   477→  }
   478→
   479→  // ── router / wireless-router WAN
   480→  if (d.kind === "router" || (d.kind === "wirelessRouter" && inPortId === "g0/0") || d.kind === "wirelessRouter") {
   481→    routerHandle(ctx, d, inPortId, packet, depth);
   482→    return;
   483→  }
   484→
   485→  // ── hosts (pc/server/laptop/smartphone)
   486→  hostHandle(ctx, d, inPortId, packet, depth);
   487→}

   489→function addArp(d: Device, ip: string, mac: string, portId: string) {
   490→  if (!d.arp.find((e) => e.ip === ip)) d.arp.push({ ip, mac, portId });
   491→}

   493→export function hostIp(d: Device): { ip: string | null; mask: string | null; port: Port | null } {
   494→  if (d.kind === "pc" || d.kind === "server") {
   495→    const p = d.ports[0];
   496→    return { ip: p.ip ?? d.staticIp, mask: p.mask ?? d.staticMask, port: p };
   497→  }
   498→  // laptop/smartphone: first port with ip
   499→  for (const p of d.ports) if (p.ip) return { ip: p.ip, mask: p.mask, port: p };
   500→  return { ip: d.staticIp, mask: d.staticMask, port: d.ports[0] ?? null };
   501→}

   503→function hostHandle(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
   504→  const forMe = packet.dstMac === BROADCAST_MAC || d.ports.some((p) => p.mac === packet.dstMac);
   505→  const { ip } = hostIp(d);
   506→  const ipForMe = ip !== null && (packet.dstIp === ip || packet.dstIp === BROADCAST_IP);
   507→
   508→  if (packet.proto === "ARP" && packet.kind === "ARP Request") {
   509→    // learn sender
   510→    addArp(d, packet.srcIp, packet.srcMac, inPortId);
   511→    if (ip !== null && packet.dstIp === ip) {
   512→      const out = d.ports.find((p) => p.linkId && p.adminUp) ?? d.ports[0];
   513→      const reply = mkPacket({
   514→        proto: "ARP",
   515→        kind: "ARP Reply",
   516→        kindBi: { ar: "رد ARP", en: "ARP Reply" },
   517→        srcMac: out.mac,
   518→        dstMac: packet.srcMac,
   519→        srcIp: ip,
   520→        dstIp: packet.srcIp,
   521→        ttl: 128,
   522→        payload: { ar: `أنا ${ip} وعنواني ${out.mac}`, en: `I am ${ip}, my MAC is ${out.mac}` },
   523→      });
   524→      step(ctx, d, inPortId, out.id, null, null, reply, info(`${d.name}: أنا ${ip} — أرد على طلب ARP`, `${d.name}: I am ${ip} — replying to ARP request`));
   525→      wireOut(ctx, d, out.id, reply, depth);
   526→    } else {
   527→      step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: طلب ARP ليس له — تجاهل`, `${d.name}: ARP request not for me — ignored`));
   528→    }
   529→    return;
   530→  }
   531→
   532→  if (packet.proto === "ARP" && packet.kind === "ARP Reply") {
   533→    addArp(d, packet.srcIp, packet.srcMac, inPortId);
   534→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: تعلمت ${packet.srcIp} هو ${packet.srcMac} — سُجّل في جدول ARP`, `${d.name}: learned ${packet.srcIp} is ${packet.srcMac} — recorded in ARP table`));
   535→    return;
   536→  }
   537→
   538→  if (!forMe) {
   539→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: ليس عنواني MAC — تجاهل الإطار`, `${d.name}: not my MAC — frame ignored`));
   540→    return;
   541→  }
   542→
   543→  if (packet.proto === "ICMP" && packet.kind === "Echo Request" && ipForMe) {
   544→    const out = d.ports.find((p) => p.linkId && p.adminUp) ?? d.ports[0];
   545→    const reply = mkPacket({
   546→      proto: "ICMP",
   547→      kind: "Echo Reply",
   548→      kindBi: { ar: "رد ICMP", en: "ICMP Echo Reply" },
   549→      srcMac: out.mac,
   550→      dstMac: packet.srcMac,
   551→      srcIp: ip!,
   552→      dstIp: packet.srcIp,
   553→      ttl: 128,
   554→    });
   555→    step(ctx, d, inPortId, out.id, null, null, reply, info(`${d.name}: وصلني Ping من ${packet.srcIp} — أرد!`, `${d.name}: ping received from ${packet.srcIp} — replying!`));
   556→    wireOut(ctx, d, out.id, reply, depth);
   557→    return;
   558→  }
   559→
   560→  if (packet.proto === "DHCP") {
   561→    dhcpHostHandle(ctx, d, inPortId, packet, depth);
   562→    return;
   563→  }
   564→  if (packet.proto === "DNS" && packet.kind === "DNS Query") {
   565→    // client receiving answer handled below; server:
   566→    dnsHandle(ctx, d, inPortId, packet, depth);
   567→    return;
   568→  }
   569→  if (packet.proto === "DNS" && packet.kind === "DNS Answer") {
   570→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: وصل جواب DNS`, `${d.name}: DNS answer received`));
   571→    return;
   572→  }
   573→  if (packet.proto === "TCP" || packet.proto === "HTTP") {
   574→    httpHandle(ctx, d, inPortId, packet, depth);
   575→    return;
   576→  }
   577→
   578→  step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: استلم الحزمة`, `${d.name}: packet received`));
   579→}

   582→export function connectedRoutes(d: Device): import("./types").Route[] {
   583→  const routes: import("./types").Route[] = [];
   584→  for (const p of d.ports) {
   585→    if (p.ip && p.mask) {
   586→      routes.push({ network: networkOf(p.ip, p.mask), mask: p.mask, nextHop: null, iface: p.id, ad: 0, kind: "connected" });
   587→    }
   588→  }
   589→  return routes;
   590→}
   591→
   592→function lookupRoute(d: Device, dstIp: string): import("./types").Route | null {
   593→  const all = [...connectedRoutes(d), ...d.routes];
   594→  let best: import("./types").Route | null = null;
   595→  let bestLen = -1;
   596→  for (const r of all) {
   597→    if (networkOf(dstIp, r.mask) === r.network) {
   598→      const len = maskToPrefix(r.mask);
   599→      if (len > bestLen || (len === bestLen && r.ad < (best?.ad ?? 99))) {
   600→        best = r;
   601→        bestLen = len;
   602→      }
   603→    }
   604→  }
   605→  return best;
   606→}

   608→function routerHandle(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
   609→  const inPort = findPort(d, inPortId);
   610→  if (!inPort) return;
   611→  const myMacs = d.ports.map((p) => p.mac);
   612→  const forMe = packet.dstMac === BROADCAST_MAC || myMacs.includes(packet.dstMac);
   613→
   614→  // ARP for me?
   615→  if (packet.proto === "ARP" && packet.kind === "ARP Request") {
   616→    addArp(d, packet.srcIp, packet.srcMac, inPortId);
   617→    const targetPort = d.ports.find((p) => p.ip === packet.dstIp);
   618→    if (forMe && targetPort) {
   619→      const replyPort = targetPort.linkId && targetPort.adminUp ? targetPort : inPort;
   620→      const reply = mkPacket({
   621→        proto: "ARP",
   622→        kind: "ARP Reply",
   623→        kindBi: { ar: "رد ARP", en: "ARP Reply" },
   624→        srcMac: targetPort.mac,
   625→        dstMac: packet.srcMac,
   626→        srcIp: packet.dstIp,
   627→        dstIp: packet.srcIp,
   628→        ttl: 128,
   629→        payload: { ar: `أنا ${packet.dstIp} عند ${targetPort.mac}`, en: `I am ${packet.dstIp} at ${targetPort.mac}` },
   630→      });
   631→      step(ctx, d, inPortId, replyPort.id, null, null, reply, info(`${d.name}: رد ARP من ${targetPort.name}`, `${d.name}: ARP reply from ${targetPort.name}`));
   632→      wireOut(ctx, d, replyPort.id, reply, depth);
   633→    } else {
   634→      step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: طلب ARP ليس لعناويني`, `${d.name}: ARP request not for my IPs`));
   635→    }
   636→    return;
   637→  }
   638→
   639→  if (packet.proto === "ARP" && packet.kind === "ARP Reply") {
   640→    addArp(d, packet.srcIp, packet.srcMac, inPortId);
   641→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: تعلمت ${packet.srcIp} هو ${packet.srcMac} — سُجّل في ARP`, `${d.name}: learned ${packet.srcIp} is ${packet.srcMac} — recorded in ARP`));
   642→    return;
   643→  }
   644→
   645→  // reverse NAT: a session reply arriving on the WAN addressed to my WAN IP → translate & deliver to the LAN host
   646→  if (d.kind === "wirelessRouter" && d.nat && inPortId === "g0/0") {
   647→    const wanP = findPort(d, "g0/0");
   648→    if (wanP?.ip && packet.dstIp === wanP.ip) {
   649→      const map =
   650→        packet.dstPort != null
   651→          ? d.natTable.find((m) => m.wanPort === packet.dstPort)
   652→          : [...d.natTable].reverse().find((m) => !m.lanPort);
   653→      if (map) {
   654→        const out = d.ports.find((p) => p.linkId && p.adminUp && p.id !== "g0/0");
   655→        if (out) {
   656→          let arpE = d.arp.find((a) => a.ip === map.lanIp);
   657→          if (!arpE) {
   658→            ensureArpFor(ctx, d, out, map.lanIp, depth);
   659→            arpE = d.arp.find((a) => a.ip === map.lanIp);
   660→          }
   661→          if (arpE) {
   662→            const fwd = {
   663→              ...packet,
   664→              dstIp: map.lanIp,
   665→              dstPort: map.lanPort || packet.dstPort,
   666→              dstMac: arpE.mac,
   667→              srcMac: out.mac,
   668→              ttl: packet.ttl - 1,
   669→            };
   670→            step(ctx, d, inPortId, out.id, null, null, fwd, info(`${d.name}: NAT عكسي — إعادة ${packet.dstIp}:${packet.dstPort ?? "-"} إلى ${map.lanIp}:${map.lanPort || "-"}`, `${d.name}: reverse NAT — restoring ${packet.dstIp}:${packet.dstPort ?? "-"} to ${map.lanIp}:${map.lanPort || "-"}`));
   671→            wireOut(ctx, d, out.id, fwd, depth);
   672→            return;
   673→          }
   674→        }
   675→      }
   676→    }
   677→  }
   678→
   679→  if (!forMe) {
   680→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: ليس عنواني MAC — تجاهل`, `${d.name}: not my MAC — ignored`));
   681→    return;
   682→  }
   683→
   684→  // DHCP server on router? (not typical—servers handle) → skip
   685→  // local services only answer packets addressed to one of my own IPs — everything else gets routed/forwarded
   686→  const isMyDstIp = d.ports.some((p) => p.ip === packet.dstIp);
   687→  if (packet.proto === "DNS" && packet.kind === "DNS Query" && isMyDstIp) {
   688→    dnsHandle(ctx, d, inPortId, packet, depth);
   689→    return;
   690→  }
   691→  if ((packet.proto === "TCP" || packet.proto === "HTTP") && isMyDstIp) {
   692→    httpHandle(ctx, d, inPortId, packet, depth);
   693→    return;
   694→  }
   695→  if (packet.proto === "DHCP") {
   696→    dhcpHostHandle(ctx, d, inPortId, packet, depth);
   697→    return;
   698→  }
   699→
   700→  const dstMe = d.ports.find((p) => p.ip === packet.dstIp);
   701→  if (dstMe && packet.proto === "ICMP" && packet.kind === "Echo Request") {
   702→    const replyPort = dstMe.linkId && dstMe.adminUp ? dstMe : inPort;
   703→    const reply = mkPacket({
   704→      proto: "ICMP",
   705→      kind: "Echo Reply",
   706→      kindBi: { ar: "رد ICMP", en: "ICMP Echo Reply" },
   707→      srcMac: dstMe.mac,
   708→      dstMac: packet.srcMac,
   709→      srcIp: packet.dstIp,
   710→      dstIp: packet.srcIp,
   711→      ttl: 128,
   712→    });
   713→    step(ctx, d, inPortId, replyPort.id, null, null, reply, info(`${d.name}: الهدف أنا — رد Pong`, `${d.name}: I am the target — pong reply`));
   714→    wireOut(ctx, d, replyPort.id, reply, depth);
   715→    return;
   716→  }
   717→
   718→  // forwarding
   719→  if (packet.ttl <= 1) {
   720→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: TTL انتهى — إسقاط الحزمة`, `${d.name}: TTL expired — packet dropped`), true);
   721→    return;
   722→  }
   723→  const route = lookupRoute(d, packet.dstIp);
   724→  if (!route) {
   725→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: لا يوجد مسار إلى ${packet.dstIp} — إسقاط (أضف ip route)`, `${d.name}: no route to ${packet.dstIp} — dropping (add ip route)`), true);
   726→    return;
   727→  }
   728→  const nextIp = route.nextHop ?? packet.dstIp;
   729→  // resolve egress: explicit iface, or the port whose subnet contains the next hop (recursive next-hop resolution)
   730→  let outPort = route.iface ? findPort(d, route.iface) : undefined;
   731→  if (!outPort || !outPort.linkId || !outPort.adminUp) {
   732→    const nh = d.ports.find(
   733→      (p) => p.ip && p.mask && p.linkId && p.adminUp && networkOf(nextIp, p.mask) === networkOf(p.ip, p.mask)
   734→    );
   735→    if (nh) outPort = nh;
   736→  }
   737→  if (!outPort || !outPort.linkId || !outPort.adminUp) {
   738→    step(ctx, d, inPortId, route.iface, null, null, packet, info(`${d.name}: منفذ الخروج ${route.iface} غير متصل`, `${d.name}: egress ${route.iface} not connected`), true);
   739→    return;
   740→  }
   741→  const arpEntry = d.arp.find((a) => a.ip === nextIp);
   742→  if (!arpEntry) {
   743→    // trigger ARP resolution first
   744→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: أحتاج MAC لـ ${nextIp} — سأرسل ARP أولاً`, `${d.name}: need MAC for ${nextIp} — sending ARP first`));
   745→    ensureArpFor(ctx, d, outPort, nextIp, depth);
   746→    const after = d.arp.find((a) => a.ip === nextIp);
   747→    if (!after) {
   748→      step(ctx, d, inPortId, route.iface, null, null, packet, info(`${d.name}: لا جواب ARP — الحزمة سقطت`, `${d.name}: no ARP reply — packet dropped`), true);
   749→      return;
   750→    }
   751→    forwardRouted(ctx, d, inPortId, outPort, packet, after.mac, depth);
   752→    return;
   753→  }
   754→  forwardRouted(ctx, d, inPortId, outPort, packet, arpEntry.mac, depth);
   755→}

   757→function forwardRouted(ctx: Ctx, d: Device, inPortId: string, outPort: Port, packet: Packet, dstMac: string, depth: number): void {
   758→  // NAT (wireless router LAN→WAN)
   759→  let fwd = packet;
   760→  if (d.kind === "wirelessRouter" && d.nat && outPort.id === "g0/0" && inPortId !== "g0/0") {
   761→    const wanIp = outPort.ip ?? "203.0.113.2";
   762→    const lanPort = packet.srcPort ?? 0;
   763→    let map = d.natTable.find((m) => m.lanIp === packet.srcIp && m.lanPort === lanPort);
   764→    if (!map) {
   765→      map = { lanIp: packet.srcIp, lanPort, wanPort: ++d.natWanCounter };
   766→      d.natTable.push(map);
   767→    }
   768→    fwd = { ...packet, srcIp: wanIp, srcPort: map.wanPort || packet.srcPort, dstMac, srcMac: outPort.mac, ttl: packet.ttl - 1 };
   769→    step(ctx, d, inPortId, outPort.id, null, null, fwd, info(`${d.name}: NAT — استبدال ${packet.srcIp}:${packet.srcPort ?? "-"} بـ ${wanIp}:${map.wanPort}`, `${d.name}: NAT — replacing ${packet.srcIp}:${packet.srcPort ?? "-"} with ${wanIp}:${map.wanPort}`));
   770→    wireOut(ctx, d, outPort.id, fwd, depth);
   771→    return;
   772→  }
   773→  // NAT reverse (WAN→LAN)
   774→  if (d.kind === "wirelessRouter" && d.nat && inPortId === "g0/0" && outPort.id !== "g0/0") {
   775→    const map = d.natTable.find((m) => m.wanPort === (packet.dstPort ?? 0));
   776→    if (map) {
   777→      fwd = { ...packet, dstIp: map.lanIp, dstPort: map.lanPort || packet.dstPort, srcMac: outPort.mac, ttl: packet.ttl - 1 };
   778→      step(ctx, d, inPortId, outPort.id, null, null, fwd, info(`${d.name}: NAT عكسي — إعادة ${packet.dstIp}:${packet.dstPort ?? "-"} إلى ${map.lanIp}`, `${d.name}: reverse NAT — restoring ${packet.dstIp}:${packet.dstPort ?? "-"} to ${map.lanIp}`));
   779→      wireOut(ctx, d, outPort.id, fwd, depth);
   780→      return;
   781→    }
   782→  }
   783→  const fwdPacket = mkPacket({ ...packet, proto: packet.proto, kind: packet.kind, kindBi: packet.kindBi, srcMac: outPort.mac, dstMac, ttl: packet.ttl - 1, srcPort: packet.srcPort, dstPort: packet.dstPort });
   784→  step(ctx, d, inPortId, outPort.id, null, null, fwdPacket, info(`${d.name}: بحث في جدول التوجيه → ${networkOf(packet.dstIp, "255.255.255.255") ? packet.dstIp : ""} خارج ${outPort.name} (TTL-1)`, `${d.name}: routing table lookup → out ${outPort.name} (TTL-1)`));
   785→  wireOut(ctx, d, outPort.id, fwdPacket, depth);
   786→}

   788→function ensureArpFor(ctx: Ctx, d: Device, outPort: Port, targetIp: string, depth: number): void {
   789→  const req = mkPacket({
   790→    proto: "ARP",
   791→    kind: "ARP Request",
   792→    kindBi: { ar: "طلب ARP", en: "ARP Request" },
   793→    srcMac: outPort.mac,
   794→    dstMac: BROADCAST_MAC,

   804→function dnsHandle(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {

   816→    dstMac: packet.srcMac,
   817→    srcIp: packet.dstIp,
   818→    dstIp: packet.srcIp,

   832→function dhcpHostHandle(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
   833→  const inP = findPort(d, inPortId);
   834→  const out = (inP && inP.linkId && inP.adminUp ? inP : d.ports.find((p) => p.linkId && p.adminUp)) ?? d.ports[0];
   835→  if (packet.kind === "DHCP Discover") {
   836→    // server side
   837→    const pool = d.dhcpPool;
   838→    if (!pool || !pool.ifacePortId) {
   839→      step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: لا أدير DHCP — تجاهل`, `${d.name}: not a DHCP server — ignoring`));
   840→      return;
   841→    }
   842→    const ip = leaseIp(pool, packet.srcMac);
   843→    const offer = mkPacket({
   844→      proto: "DHCP",
   845→      kind: "DHCP Offer",
   846→      kindBi: { ar: "عرض DHCP", en: "DHCP Offer" },
   847→      srcMac: out.mac,
   848→      dstMac: packet.srcMac,
   849→      srcIp: (findPort(d, pool.ifacePortId)?.ip) ?? "0.0.0.0",
   850→      dstIp: ip,
   851→      ttl: 128,
   852→      srcPort: 67,
   853→      dstPort: 68,
   854→      payload: { ar: `أعرض ${ip} بوابة ${pool.gateway} قناع ${pool.mask}`, en: `Offering ${ip} gw ${pool.gateway} mask ${pool.mask}` },
   855→    });
   856→    step(ctx, d, inPortId, out.id, null, null, offer, info(`${d.name}: عرض DHCP بعنوان ${ip}`, `${d.name}: DHCP offer with ${ip}`));
   857→    wireOut(ctx, d, out.id, offer, depth);
   858→    return;
   859→  }
   860→  if (packet.kind === "DHCP Request" && d.dhcpPool) {
   861→    const pool = d.dhcpPool;
   862→    const m = packet.payload?.en.match(/\b(\d{1,3}(?:\.\d{1,3}){3})\b/);
   863→    const ip = m?.[1] ?? leaseIp(pool, packet.srcMac);
   864→    const ack = mkPacket({
   865→      proto: "DHCP",
   866→      kind: "DHCP Ack",
   867→      kindBi: { ar: "تأكيد DHCP", en: "DHCP Ack" },
   868→      srcMac: out.mac,
   869→      dstMac: packet.srcMac,
   870→      srcIp: (findPort(d, pool.ifacePortId)?.ip) ?? "0.0.0.0",
   871→      dstIp: ip,
   872→      ttl: 128,
   873→      srcPort: 67,
   874→      dstPort: 68,
   875→      payload: { ar: `تأكيد ${ip} بوابة ${pool.gateway} DNS ${pool.dns ?? "-"}`, en: `Ack ${ip} gw ${pool.gateway} dns ${pool.dns ?? "-"}` },
   876→    });
   877→    step(ctx, d, inPortId, out.id, null, null, ack, info(`${d.name}: تأكيد DHCP — ${ip} أصبح لعنوان ${packet.srcMac}`, `${d.name}: DHCP ack — ${ip} assigned to ${packet.srcMac}`));
   878→    wireOut(ctx, d, out.id, ack, depth);
   879→    return;
   880→  }
   881→  // client side
   882→  if (packet.kind === "DHCP Offer" && d.dhcpClient) {
   883→    const ip = packet.dstIp;
   884→    const req = mkPacket({
   885→      proto: "DHCP",
   886→      kind: "DHCP Request",
   887→      kindBi: { ar: "طلب DHCP", en: "DHCP Request" },
   888→      srcMac: out.mac,
   889→      dstMac: BROADCAST_MAC,
   890→      srcIp: ZERO_IP,
   891→      dstIp: BROADCAST_IP,
   892→      ttl: 128,
   893→      srcPort: 68,
   894→      dstPort: 67,
   895→      payload: { ar: `أطلب ${ip}`, en: `Requesting ${ip}` },
   896→    });
   897→    step(ctx, d, inPortId, out.id, null, null, req, info(`${d.name}: طلب العنوان ${ip}`, `${d.name}: requesting ${ip}`));
   898→    (d as Device & { _offered?: string })._offered = ip;
   899→    wireOut(ctx, d, out.id, req, depth);
   900→    return;
   901→  }
   902→  if (packet.kind === "DHCP Ack" && d.dhcpClient) {
   903→    const ip = packet.dstIp;
   904→    const port = out;
   905→    port.ip = ip;
   906→    port.mask = "255.255.255.0";
   907→    const gwMatch = packet.payload?.ar.match(/بوابة\s*(\S+)/);
   908→    const dnsMatch = packet.payload?.ar.match(/DNS\s*(\S+)/);
   909→    d.gateway = gwMatch?.[1] ?? null;
   910→    d.dnsServer = dnsMatch?.[1] && dnsMatch[1] !== "-" ? dnsMatch[1] : null;
   911→    delete (d as Device & { _offered?: string })._offered;
   912→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: استلمت ${ip} عبر DHCP — تم الإعداد!`, `${d.name}: got ${ip} via DHCP — configured!`));
   913→    return;
   914→  }
   915→}

   917→function leaseIp(pool: import("./types").DhcpPool, mac: string): string {
   918→  const existing = pool.leases.find((l) => l.mac === mac);
   919→  if (existing) return existing.ip;
   920→  const from = ipToInt(pool.from);
   921→  const to = ipToInt(pool.to);
   922→  const used = new Set(pool.leases.map((l) => l.ip));
   923→  for (let i = from; i <= to; i++) {
   924→    const cand = intToIp(i);
   925→    if (!used.has(cand)) {
   926→      pool.leases.push({ mac, ip: cand });
   927→      return cand;
   928→    }
   929→  }
   930→  return pool.from;
   931→}

   933→function httpHandle(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
   934→  const inP0 = findPort(d, inPortId);
   935→  const out = (inP0 && inP0.linkId && inP0.adminUp ? inP0 : d.ports.find((p) => p.linkId && p.adminUp)) ?? d.ports[0];
   936→  const myIp = hostIp(d).ip ?? d.ports.find((p) => p.ip)?.ip;
   937→  if (!myIp) return;
   938→  const forMeIp = packet.dstIp === myIp;
   939→
   940→  if (packet.kind === "TCP SYN" && forMeIp) {
   941→    const synack = mkPacket({ proto: "TCP", kind: "TCP SYN-ACK", kindBi: { ar: "مصافحة SYN-ACK", en: "TCP SYN-ACK" }, srcMac: out.mac, dstMac: packet.srcMac, srcIp: myIp, dstIp: packet.srcIp, ttl: 128, srcPort: packet.dstPort, dstPort: packet.srcPort, flags: "SYN,ACK", seq: "200" });
   942→    step(ctx, d, inPortId, out.id, null, null, synack, info(`${d.name}: SYN-ACK — بدء المصافحة الثلاثية`, `${d.name}: SYN-ACK — three-way handshake`));
   943→    wireOut(ctx, d, out.id, synack, depth);
   944→    return;
   945→  }
   946→  if (packet.kind === "TCP ACK") {
   947→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: اكتملت المصافحة — الاتصال مفتوح`, `${d.name}: handshake complete — connection open`));
   948→    return;
   949→  }
   950→  if (packet.kind === "HTTP GET" && forMeIp) {
   951→    const page = d.httpRoot;
   952→    const ok = mkPacket({ proto: "HTTP", kind: "HTTP 200 OK", kindBi: { ar: "رد HTTP 200", en: "HTTP 200 OK" }, srcMac: out.mac, dstMac: packet.srcMac, srcIp: myIp, dstIp: packet.srcIp, ttl: 128, srcPort: 80, dstPort: packet.srcPort, payload: { ar: `${page.title}\n${page.body}`, en: `${page.title}\n${page.body}` } });
   953→    step(ctx, d, inPortId, out.id, null, null, ok, info(`${d.name}: صفحة «${page.title}» جاهزة — 200 OK`, `${d.name}: page "${page.title}" served — 200 OK`));
   954→    wireOut(ctx, d, out.id, ok, depth);
   955→    return;
   956→  }
   957→  step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: استلم حزمة ${packet.kind}`, `${d.name}: received ${packet.kind}`));
   958→}

   965→export function simulatePing(topo: Topology, srcId: string, dstIdOrIp: string): SimResult {
   966→  const t = cloneTopo(topo);
   967→  const ctx = newCtx(t);
   968→  const src = findDevice(t, srcId);
   969→  if (!src) return { steps: [], devices: topo.devices, success: false, note: info("الجهاز غير موجود", "Device not found") };
   970→  const { ip, mask, port } = hostIp(src);
   971→  if (!ip || !mask || !port) {
   972→    return { steps: [], devices: topo.devices, success: false, note: info(`${src.name} بلا عنوان IP — اضبطه أولاً من إعدادات الجهاز`, `${src.name} has no IP — configure it first in device settings`) };
   973→  }
   974→  // resolve destination ip
   975→  let dstIp = dstIdOrIp;
   976→  const dstDev = findDevice(t, dstIdOrIp);
   977→  if (dstDev) {
   978→    const dip = hostIp(dstDev).ip;
   979→    if (!dip) return { steps: [], devices: topo.devices, success: false, note: info(`${dstDev.name} بلا عنوان IP`, `${dstDev.name} has no IP`) };
   980→    dstIp = dip;
   981→  }
   982→  // ARP resolution
   983→  const local = sameSubnet(ip, dstIp, mask);
   984→  const nextIp = local ? dstIp : src.gateway;
   985→  if (!nextIp) {
   986→    return { steps: [], devices: topo.devices, success: false, note: info("الوجهة خارج شبكتك ولا يوجد بوابة افتراضية (Default Gateway)", "Destination is off-subnet and no default gateway is set") };
   987→  }
   988→  let arpEntry = src.arp.find((a) => a.ip === nextIp);
   989→  const outPort = local ? port : (port.linkId ? port : port);
   990→  if (!arpEntry) {
   991→    const req = mkPacket({ proto: "ARP", kind: "ARP Request", kindBi: { ar: "طلب ARP", en: "ARP Request" }, srcMac: outPort.mac, dstMac: BROADCAST_MAC, srcIp: ip, dstIp: nextIp, ttl: 128, payload: { ar: `من يعرف ${nextIp}؟`, en: `Who has ${nextIp}?` } });
   992→    step(ctx, src, null, outPort.id, null, null, req, info(`${src.name}: أريد ${nextIp} — إرسال ARP بثّاً`, `${src.name}: need ${nextIp} — broadcasting ARP`));
   993→    wireOut(ctx, src, outPort.id, req, 0);
   994→    arpEntry = src.arp.find((a) => a.ip === nextIp);
   995→    if (!arpEntry) {
   996→      return { steps: ctx.steps, devices: t.devices, success: false, note: info("لا جواب ARP — تحقق من الكابلات والشبكة الفرعية نفسها", "No ARP reply — check cables and matching subnets") };
   997→    }
   998→  }
   999→  const echo = mkPacket({ proto: "ICMP", kind: "Echo Request", kindBi: { ar: "طلب ICMP (Ping)", en: "ICMP Echo Request" }, srcMac: outPort.mac, dstMac: arpEntry.mac, srcIp: ip, dstIp, ttl: 128 });
  1000→  step(ctx, src, null, outPort.id, null, null, echo, info(`${src.name}: Ping إلى ${dstIp}`, `${src.name}: ping to ${dstIp}`));
  1001→  wireOut(ctx, src, outPort.id, echo, 0);
  1002→  const replied = ctx.steps.some((s) => s.packet.kind === "Echo Reply" && s.deviceId === srcId);
  1003→  return {
  1004→    steps: ctx.steps,
  1005→    devices: t.devices,
  1006→    success: replied,
  1007→    note: replied
  1008→      ? info("نجح Ping! رأيت الطلب والرد يمران عبر كل الأجهزة", "Ping succeeded! You saw request & reply traverse every device")
  1009→      : info("فشل Ping — راجع الخطوات المظللة بالأحمر", "Ping failed — review the red-marked steps"),
  1010→  };
  1011→}

  1027→export function simulateDns(topo: Topology, clientId: string, name: string): SimResult & { resolvedIp: string | null } {
  1028→  const t = cloneTopo(topo);
  1029→  const ctx = newCtx(t);
  1030→  const client = findDevice(t, clientId);
  1031→  if (!client) return { steps: [], devices: topo.devices, success: false, note: info("الجهاز غير موجود", "Device not found"), resolvedIp: null };
  1032→  const { ip, mask, port } = hostIp(client);
  1033→  if (!ip || !mask || !port) return { steps: [], devices: topo.devices, success: false, note: info("اضبط عنوان الجهاز أولاً", "Configure device IP first"), resolvedIp: null };
  1034→  if (!client.dnsServer) return { steps: [], devices: topo.devices, success: false, note: info("لا يوجد خادم DNS مضبوط", "No DNS server configured"), resolvedIp: null };
  1035→  const local = sameSubnet(ip, client.dnsServer, mask);
  1036→  const nextIp = local ? client.dnsServer : client.gateway;
  1037→  if (!nextIp) return { steps: [], devices: topo.devices, success: false, note: info("لا بوابة للوصول لخادم DNS", "No gateway to reach DNS server"), resolvedIp: null };
  1038→  let arpEntry = client.arp.find((a) => a.ip === nextIp);
  1039→  if (!arpEntry) {
  1040→    ensureArpFor(ctx, client, port, nextIp, 0);
  1041→    arpEntry = client.arp.find((a) => a.ip === nextIp);
  1042→    if (!arpEntry) return { steps: ctx.steps, devices: t.devices, success: false, note: info("لا جواب ARP نحو DNS", "No ARP reply toward DNS"), resolvedIp: null };
  1043→  }
  1044→  const query = mkPacket({ proto: "DNS", kind: "DNS Query", kindBi: { ar: "استعلام DNS", en: "DNS Query" }, srcMac: port.mac, dstMac: arpEntry.mac, srcIp: ip, dstIp: client.dnsServer, ttl: 128, srcPort: 1025, dstPort: 53, payload: { ar: name, en: name } });
  1045→  step(ctx, client, null, port.id, null, null, query, info(`${client.name}: استعلام عن ${name}`, `${client.name}: querying ${name}`));
  1046→  wireOut(ctx, client, port.id, query, 0);
  1047→  const answerStep = [...ctx.steps].reverse().find((s) => s.packet.kind === "DNS Answer" && s.deviceId === clientId);
  1048→  const resolved = answerStep?.packet.payload?.ar.split(" = ")[1] ?? null;
  1049→  const ok = !!answerStep && !answerStep.packet.payload?.ar.includes("غير موجود");
  1050→  return { steps: ctx.steps, devices: t.devices, success: ok, note: ok ? info(`تم حل ${name} إلى ${resolved}`, `Resolved ${name} to ${resolved}`) : info(`تعذر حل ${name}`, `Could not resolve ${name}`), resolvedIp: ok ? resolved : null };
  1051→}

  1053→export function simulateHttp(topo: Topology, clientId: string, host: string): SimResult & { page: { title: string; body: string } | null } {
  1054→  const t = cloneTopo(topo);
  1055→  const ctx = newCtx(t);
  1056→  const client = findDevice(t, clientId);
  1057→  if (!client) return { steps: [], devices: topo.devices, success: false, note: info("الجهاز غير موجود", "Device not found"), page: null };
  1058→  let targetIp: string | null = null;
  1059→  let stepsSoFar: SimStep[] = [];
  1060→  if (!isValidIp(host)) {
  1061→    const dns = simulateDns(topo, clientId, host);
  1062→    stepsSoFar = dns.steps;
  1063→    // commit arp/dns state
  1064→    for (const dev of dns.devices) {
  1065→      const mine = findDevice(t, dev.id);
  1066→      if (mine) {
  1067→        mine.arp = dev.arp;
  1068→        mine.macTable = dev.macTable;
  1069→      }
  1070→    }
  1071→    targetIp = dns.resolvedIp;
  1072→    if (!targetIp) {
  1073→      return { steps: stepsSoFar, devices: t.devices, success: false, note: info(`تعذر حل اسم ${host} — أضف سجل A في خادم DNS`, `Cannot resolve ${host} — add an A record on the DNS server`), page: null };
  1074→    }
  1075→  } else {
  1076→    targetIp = host;
  1077→  }
  1078→  // run on a fresh context so steps continue linearly
  1079→  ctx.steps.push(...stepsSoFar);
  1080→  ctx.sid = stepsSoFar.length;
  1081→  const { ip, mask, port } = hostIp(client);
  1082→  if (!ip || !mask || !port) return { steps: ctx.steps, devices: t.devices, success: false, note: info("اضبط عنوان الجهاز أولاً", "Configure device IP first"), page: null };
  1083→  const local = sameSubnet(ip, targetIp, mask);
  1084→  const nextIp = local ? targetIp : client.gateway;
  1085→  if (!nextIp) return { steps: ctx.steps, devices: t.devices, success: false, note: info("لا بوابة مضبوطة", "No gateway set"), page: null };
  1086→  let arpEntry = client.arp.find((a) => a.ip === nextIp);
  1087→  if (!arpEntry) {
  1088→    ensureArpFor(ctx, client, port, nextIp, 0);
  1089→    arpEntry = client.arp.find((a) => a.ip === nextIp);
  1090→    if (!arpEntry) return { steps: ctx.steps, devices: t.devices, success: false, note: info("لا جواب ARP", "No ARP reply"), page: null };
  1091→  }
  1092→  // TCP handshake
  1093→  const sendTcp = (kind: string, kindBi: { ar: string; en: string }, flags: string, seq: string, srcPort: number, dstPort: number, payload?: { ar: string; en: string }) => {
  1094→    const pkt = mkPacket({ proto: kind.startsWith("HTTP") ? "HTTP" : "TCP", kind, kindBi, srcMac: port.mac, dstMac: arpEntry!.mac, srcIp: ip!, dstIp: targetIp!, ttl: 128, srcPort, dstPort, flags, seq, payload });
  1095→    wireOut(ctx, client, port.id, pkt, 0);
  1096→    return pkt;
  1097→  };
  1098→  const syn = sendTcp("TCP SYN", { ar: "SYN", en: "TCP SYN" }, "SYN", "100", 1025, 80);
  1099→  void syn;
  1100→  // SYN-ACK/ACK handled by handlers; send ACK
  1101→  const ack = sendTcp("TCP ACK", { ar: "ACK", en: "TCP ACK" }, "ACK", "101", 1025, 80);
  1102→  void ack;
  1103→  // GET
  1104→  const get = sendTcp("HTTP GET", { ar: "طلب GET", en: "HTTP GET" }, "ACK", "102", 1025, 80, { ar: `GET / HTTP/1.1\nHost: ${host}`, en: `GET / HTTP/1.1\nHost: ${host}` });
  1105→  void get;
  1106→  const okStep = [...ctx.steps].reverse().find((s) => s.packet.kind === "HTTP 200 OK" && s.deviceId === clientId);
  1107→  const page = okStep ? { title: okStep.packet.payload?.en.split("\n")[0] ?? "NetMastery", body: okStep.packet.payload?.en.split("\n").slice(1).join("\n") ?? "" } : null;
  1108→  return { steps: ctx.steps, devices: t.devices, success: !!okStep, note: okStep ? info("تم تحميل الصفحة بنجاح عبر HTTP!", "Page loaded successfully over HTTP!") : info("لم يصل رد HTTP — تأكد من إعداد الخادم والمسار", "No HTTP reply — check server config & route"), page };
