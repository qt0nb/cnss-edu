     1→"use client";
     2→
     3→import React, { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import { motion, AnimatePresence } from "framer-motion";
     5→import {
     6→  MousePointer2, Trash2, Cable, Radar, Globe, HandCoins, Play, Pause, SkipBack,
     7→  SkipForward, RotateCcw, Save, FolderOpen, Download, Upload, FlaskConical, Info,
     8→  MessageSquareText, Eraser, X,
     9→} from "lucide-react";
    10→import { Button } from "@/components/ui/button";
    11→import { Input } from "@/components/ui/input";
    12→import { Badge } from "@/components/ui/badge";
    13→import { Slider } from "@/components/ui/slider";
    14→import { ScrollArea } from "@/components/ui/scroll-area";
    15→import {
    16→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter,
    17→} from "@/components/ui/dialog";
    18→import {
    19→  DropdownMenu, DropdownMenuContent, DropdownMenuItem, DropdownMenuTrigger, DropdownMenuLabel,
    20→} from "@/components/ui/dropdown-menu";
    21→import { useLang } from "@/lib/i18n";
    22→import { useProgress } from "@/lib/store";
    23→import { toast } from "@/hooks/use-toast";
    24→import type { Bi } from "@/lib/types";
    25→import type { Device, SimStep, Topology } from "@/lib/netsim/types";
    26→import { DEVICE_SPECS, LINK_SPECS } from "@/lib/netsim/types";
    27→import {
    28→  createDevice, autoCable, connectPorts, simulatePing, simulateDhcp, simulateDns,
    29→  simulateHttp, findDevice, findPort,
    30→} from "@/lib/netsim/engine";
    31→import { NETSIM_LABS } from "@/lib/netsim/labs";
    32→import { DeviceIcon, linkStyle, protoColor } from "./icons";
    33→import DeviceDialog, { type NetSimActions } from "./DeviceDialog";
    34→import PduDialog from "./PduDialog";
    35→
    36→type Tool = "select" | "delete" | "link" | "ping" | "dhcp" | "http";
    37→
    38→const CATEGORY_LABEL: Record<string, Bi> = {
    39→  end: { ar: "الأجهزة الطرفية", en: "End Devices" },
    40→  network: { "en": "Network Devices", ar: "أجهزة الشبكة" },
    41→  wireless: { ar: "اللاسلكي", en: "Wireless" },
    42→  wan: { ar: "WAN", en: "WAN" },
    43→};
    44→
    45→const CANVAS_W = 1040;
    46→const CANVAS_H = 600;
    47→
    48→export default function NetSim() {
    49→  const { lang, bi, t } = useLang();
    50→  const markUsed = useProgress((s) => s.markPlaygroundUsed);
    51→  useEffect(() => { markUsed("netsim"); }, [markUsed]);
    52→
    53→  const [topo, setTopo] = useState<Topology>({ devices: [], links: [] });
    54→  const topoRef = useRef(topo);
    55→  topoRef.current = topo;
    56→  const [tool, setTool] = useState<Tool>("select");
    57→  const [linkSrc, setLinkSrc] = useState<string | null>(null);
    58→  const [pingSrc, setPingSrc] = useState<string | null>(null);
    59→  const [httpDev, setHttpDev] = useState<string | null>(null);
    60→  const [urlInput, setUrlInput] = useState("");
    61→  const [openDevId, setOpenDevId] = useState<string | null>(null);
    62→  const [sim, setSim] = useState<{ steps: SimStep[]; idx: number; playing: boolean; note: Bi | null; success: boolean } | null>(null);
    63→  const simRef = useRef(sim);
    64→  simRef.current = sim;
    65→  const [pduStep, setPduStep] = useState<SimStep | null>(null);
    66→  const [speed, setSpeed] = useState(1);
    67→  const [paletteOpen, setPaletteOpen] = useState(false);
    68→  const [showPanel, setShowPanel] = useState(false);
    69→  const [drag, setDrag] = useState<{ id: string; dx: number; dy: number; moved: boolean } | null>(null);
    70→  const canvasRef = useRef<HTMLDivElement>(null);
    71→  const fileRef = useRef<HTMLInputElement>(null);
    72→
    73→  // autosave
    74→  useEffect(() => {
    75→    try {
    76→      const saved = localStorage.getItem("nm-netsim");
    77→      if (saved) {
    78→        const parsed = JSON.parse(saved) as Topology;
    79→        if (parsed?.devices?.length) setTopo(parsed);
    80→        else setTopo(NETSIM_LABS[0].build());
    81→      } else {
    82→        setTopo(NETSIM_LABS[0].build());
    83→      }
    84→    } catch {
    85→      setTopo(NETSIM_LABS[0].build());
    86→    }
    87→  }, []);
    88→
    89→  const commitDevices = useCallback((devices: Device[]) => {
    90→    setTopo((t) => ({ devices, links: t.links }));
    91→  }, []);
    92→
    93→  const runSim = useCallback(
    94→    (steps: SimStep[], devices: Device[], note: Bi | null, success: boolean, autoplay = true) => {
    95→      setSim({ steps, idx: 0, playing: autoplay && steps.length > 0, note, success });
    96→      if (steps.length > 0) {
    97→        commitDevices(devices);
    98→        setShowPanel(true);
    99→      }
   100→    },
   101→    [commitDevices]
   102→  );
   103→
   104→  const actions: NetSimActions = useMemo(() => ({
   105→    updateDevice: () => setTopo((t) => ({ devices: [...t.devices], links: t.links })),
   106→    runPing: (srcId, ip) => {
   107→      const r = simulatePing(topoRef.current, srcId, ip);
   108→      runSim(r.steps, r.devices, r.note, r.success);
   109→      const lines = r.success
   110→        ? [
   111→            `Reply from ${ip}: bytes=32 time${lang === "ar" ? "≈" : "≈"}1ms TTL=${128 - Math.max(0, r.steps.filter((s) => s.outPort && s.deviceId !== srcId).length - 1)}`,
   112→            `Reply from ${ip}: bytes=32 time≈1ms`,
   113→            lang === "ar" ? "نجح: 4/4 — Ping يعمل" : "Success: 4/4 — ping working",
   114→          ]
   115→        : [lang === "ar" ? "انتهت المهلة: الطلب لم يجد طريقه — راجع لوحة المحاكاة" : "Request timed out — check the simulation panel"];
   116→      return lines;
   117→    },
   118→    runDhcp: (clientId) => {
   119→      const r = simulateDhcp(topoRef.current, clientId);
   120→      runSim(r.steps, r.devices, r.note, r.success);
   121→      const d = findDevice({ devices: r.devices, links: [] }, clientId);
   122→      return r.success
   123→        ? [lang === "ar" ? `استلمت ${d?.ports.find((p) => p.ip)?.ip ?? ""} بوابة ${d?.gateway ?? "-"} DNS ${d?.dnsServer ?? "-"}` : `Got ${d?.ports.find((p) => p.ip)?.ip ?? ""} gw ${d?.gateway ?? "-"} dns ${d?.dnsServer ?? "-"}`]
   124→        : [lang === "ar" ? "فشل DORA — تأكد من الخادم والكابلات" : "DORA failed — check server & cables"];
   125→    },
   126→    runDns: (srcId, name) => {
   127→      const r = simulateDns(topoRef.current, srcId, name);
   128→      runSim(r.steps, r.devices, r.note, r.success, false);
   129→      return [r.success ? `${name} → ${r.resolvedIp}` : (lang === "ar" ? "تعذر الحل" : "Resolution failed")];
   130→    },
   131→    runHttp: (srcId, host) => {
   132→      const r = simulateHttp(topoRef.current, srcId, host);
   133→      runSim(r.steps, r.devices, r.note, r.success, false);
   134→      return { lines: [bi(r.note)], page: r.page };
   135→    },
   136→  }), [lang, bi, runSim]);
   137→
   138→  // playback
   139→  useEffect(() => {
   140→    if (!sim?.playing) return;
   141→    if (sim.idx >= sim.steps.length - 1) {
   142→      setSim((s) => (s ? { ...s, playing: false } : s));
   143→      return;
   144→    }
   145→    const dur = Math.max(220, 1150 / speed);
   146→    const timer = setTimeout(() => {
   147→      setSim((s) => (s ? { ...s, idx: Math.min(s.steps.length - 1, s.idx + 1) } : s));
   148→    }, dur);
   149→    return () => clearTimeout(timer);
   150→  }, [sim?.playing, sim?.idx, sim?.steps.length, speed]);
   151→
   152→  // ── device ops ──
   153→  const addDevice = (kind: Device["kind"]) => {
   154→    const rect = canvasRef.current?.getBoundingClientRect();
   155→    const x = Math.min(CANVAS_W - 90, Math.max(20, (rect?.width ?? 800) / 2 - 40 + (Math.random() * 120 - 60)));
   156→    const y = Math.min(CANVAS_H - 90, Math.max(20, (rect?.height ?? 500) / 2 - 40 + (Math.random() * 120 - 60)));
   157→    const dev = createDevice(kind, x, y, topoRef.current);
   158→    setTopo((t) => ({ devices: [...t.devices, dev], links: t.links }));
   159→    toast({ title: `${dev.name} ${lang === "ar" ? "أُضيف" : "added"}` });
   160→  };
   161→
   162→  const deleteDevice = (id: string) => {
   163→    setTopo((t) => {
   164→      const links = t.links.filter((l) => l.a.deviceId !== id && l.b.deviceId !== id);
   165→      for (const d of t.devices) {
   166→        if (d.id === id) continue;
   167→        for (const p of d.ports) if (p.linkId && !links.find((l) => l.id === p.linkId)) p.linkId = null;
   168→      }
   169→      return { devices: t.devices.filter((d) => d.id !== id), links };
   170→    });
   171→    setSim(null);
   172→  };
   173→
   174→  const deleteLink = (linkId: string) => {
   175→    setTopo((t) => {
   176→      const links = t.links.filter((l) => l.id !== linkId);
   177→      for (const d of t.devices) for (const p of d.ports) if (p.linkId === linkId) p.linkId = null;
   178→      return { devices: t.devices, links };
   179→    });
   180→    setSim(null);
   181→  };
   182→
   183→  // ── canvas interactions ──
   184→  const onDevicePointerDown = (e: React.PointerEvent, dev: Device) => {
   185→    if (tool !== "select") return;
   186→    const rect = canvasRef.current!.getBoundingClientRect();
   187→    setDrag({ id: dev.id, dx: e.clientX - rect.left - dev.x, dy: e.clientY - rect.top - dev.y, moved: false });
   188→    (e.target as Element).setPointerCapture?.(e.pointerId);
   189→  };
   190→
   191→  const onCanvasPointerMove = (e: React.PointerEvent) => {
   192→    if (!drag) return;
   193→    const rect = canvasRef.current?.getBoundingClientRect();
   194→    if (!rect) return;
   195→    const x = Math.max(8, Math.min(CANVAS_W - 80, e.clientX - rect.left - drag.dx));
   196→    const y = Math.max(8, Math.min(CANVAS_H - 70, e.clientY - rect.top - drag.dy));
   197→    if (Math.abs(x - (findDevice(topo, drag.id)?.x ?? 0)) > 2 || Math.abs(y - (findDevice(topo, drag.id)?.y ?? 0)) > 2) drag.moved = true;
   198→    setTopo((t) => ({
   199→      devices: t.devices.map((d) => (d.id === drag.id ? { ...d, x, y } : d)),
   200→      links: t.links,
   201→    }));
   202→  };
   203→
   204→  const onDeviceClick = (dev: Device) => {
   205→    if (tool === "select") {
   206→      if (drag?.moved) return;
   207→      setOpenDevId(dev.id);
   208→    } else if (tool === "delete") {
   209→      deleteDevice(dev.id);
   210→    } else if (tool === "link") {
   211→      if (!linkSrc) {
   212→        setLinkSrc(dev.id);
   213→        toast({ title: `${dev.name}: ${lang === "ar" ? "اختر الجهاز الثاني للربط" : "pick the second device to connect"}` });
   214→      } else if (linkSrc === dev.id) {
   215→        setLinkSrc(null);
   216→      } else {
   217→        const t = topoRef.current;
   218→        const auto = autoCable(t, linkSrc, dev.id);
   219→        if (!auto) {
   220→          toast({ title: lang === "ar" ? "لا منافذ متوافقة حرة" : "No compatible free ports", variant: "destructive" });
   221→        } else {
   222→          const link = connectPorts(t, linkSrc, auto.aPortId, dev.id, auto.bPortId, auto.kind);
   223→          if (link) {
   224→            setTopo({ devices: [...t.devices], links: [...t.links] });
   225→            toast({ title: `${dev.name} ↔ ${findDevice(t, linkSrc)?.name} — ${bi(LINK_SPECS.find((l) => l.kind === link.kind)!.nameBi)}` });
   226→          }
   227→        }
   228→        setLinkSrc(null);
   229→      }
   230→    } else if (tool === "ping") {
   231→      if (!pingSrc) {
   232→        const hasIp = dev.ports.some((p) => p.ip) || dev.staticIp;
   233→        setPingSrc(dev.id);
   234→        toast({ title: hasIp ? `${dev.name}: ${lang === "ar" ? "اختر الهدف" : "now pick the target"}` : `${dev.name} ${lang === "ar" ? "بلا IP!" : "has no IP!"}`, variant: hasIp ? "default" : "destructive" });
   235→      } else {
   236→        actions.runPing(pingSrc, dev.id);
   237→        setPingSrc(null);
   238→      }
   239→    } else if (tool === "dhcp") {
   240→      actions.runDhcp(dev.id);
   241→    } else if (tool === "http") {
   242→      setHttpDev(dev.id);
   243→    }
   244→  };
   245→
   246→  // ── persistence ──
   247→  const save = () => {
   248→    localStorage.setItem("nm-netsim", JSON.stringify(topo));
   249→    toast({ title: t("copied") + " ✔ " + (lang === "ar" ? "حُفظ المخطط محلياً" : "topology saved locally") });
   250→  };
   251→  const loadSaved = () => {
   252→    const raw = localStorage.getItem("nm-netsim");
   253→    if (raw) {
   254→      setTopo(JSON.parse(raw) as Topology);
   255→      toast({ title: lang === "ar" ? "تم التحميل" : "loaded" });
   256→    } else toast({ title: lang === "ar" ? "لا مخطط محفوظ" : "no saved topology" });
   257→  };
   258→  const exportJson = () => {
   259→    const blob = new Blob([JSON.stringify(topo, null, 2)], { type: "application/json" });
   260→    const a = document.createElement("a");
   261→    a.href = URL.createObjectURL(blob);
   262→    a.download = "netsim-topology.json";
   263→    a.click();
   264→  };
   265→  const importJson = (f: File) => {
   266→    f.text().then((txt) => {
   267→      try {
   268→        const parsed = JSON.parse(txt) as Topology;
   269→        if (Array.isArray(parsed.devices) && Array.isArray(parsed.links)) {
   270→          setTopo(parsed);
   271→          toast({ title: lang === "ar" ? "تم الاستيراد" : "imported" });
   272→        } else throw new Error();
   273→      } catch {
   274→        toast({ title: lang === "ar" ? "ملف غير صالح" : "invalid file", variant: "destructive" });
   275→      }
   276→    });
   277→  };
   278→
   279→  const currentStep = sim && sim.idx < sim.steps.length ? sim.steps[sim.idx] : null;
   280→  const openDevice = openDevId ? findDevice(topo, openDevId) : null;
   281→
   282→  const envelope = useMemo(() => {
   283→    if (!currentStep || !sim?.playing && sim?.idx === 0 && !sim.steps.length) return null;
   284→    if (!currentStep) return null;
   285→    const srcD = findDevice(topo, currentStep.deviceId);
   286→    if (!currentStep.linkId || !currentStep.toDeviceId) return null;
   287→    const dstD = findDevice(topo, currentStep.toDeviceId);
   288→    if (!srcD || !dstD) return null;
   289→    return { x1: srcD.x + 30, y1: srcD.y + 22, x2: dstD.x + 30, y2: dstD.y + 22, proto: currentStep.packet.proto };
   290→     
   291→  }, [currentStep?.id]);
   292→
   293→  const tools: { id: Tool; icon: React.ElementType; label: Bi }[] = [
   294→    { id: "select", icon: MousePointer2, label: { ar: "تحديد/تحريك", en: "Select/Move" } },
   295→    { id: "delete", icon: Trash2, label: { ar: "حذف", en: "Delete" } },
   296→    { id: "link", icon: Cable, label: { ar: "ربط كابل (تلقائي)", en: "Connect cable (auto)" } },
   297→    { id: "ping", icon: Radar, label: { ar: "Ping (PDU بسيط)", en: "Ping (Simple PDU)" } },
   298→    { id: "dhcp", icon: HandCoins, label: { ar: "طلب DHCP", en: "DHCP request" } },
   299→    { id: "http", icon: Globe, label: { ar: "متصفح/HTTP", en: "Browser/HTTP" } },
   300→  ];
   301→
   302→  const palette = (
   303→    <div className="flex gap-1.5 overflow-x-auto lg:overflow-y-auto lg:overflow-x-hidden lg:flex-col lg:gap-0.5 p-1.5">
   304→      {(["end", "network", "wireless", "wan"] as const).map((cat) => (
   305→        <div key={cat} className="shrink-0">
   306→          <div className="text-[9.5px] font-black text-muted-foreground px-1 pt-1.5 pb-0.5 uppercase">{bi(CATEGORY_LABEL[cat])}</div>
   307→          <div className="flex lg:flex-col gap-1">
   308→            {DEVICE_SPECS.filter((s) => s.category === cat).map((spec) => (
   309→              <button
   310→                key={spec.kind}
   311→                onClick={() => addDevice(spec.kind)}
   312→                title={bi(spec.descBi)}
   313→                className="group flex items-center gap-2 rounded-lg border border-border bg-card px-2 py-1.5 hover:border-primary hover:bg-accent transition-colors shrink-0"
   314→              >
   315→                <DeviceIcon kind={spec.kind} size={26} />
   316→                <span className="hidden lg:block text-[10.5px] font-bold text-start leading-tight">{bi(spec.nameBi)}</span>
   317→              </button>
   318→            ))}
   319→          </div>
   320→        </div>
   321→      ))}
   322→    </div>
   323→  );
   324→
   325→  return (
   326→    <div className="flex flex-col gap-3">
   327→      {/* header */}
   328→      <div className="flex flex-wrap items-center gap-2">
   329→        <div className="flex items-center gap-2">
   330→          <div className="grid size-9 place-items-center rounded-xl bg-primary text-primary-foreground glow-primary">
   331→            <FlaskConical className="size-5" />
   332→          </div>
   333→          <div>
   334→            <h2 className="text-sm font-black leading-tight">{lang === "ar" ? "محاكي الشبكات التفاعلي" : "Interactive Network Simulator"}</h2>
   335→            <p className="text-[10.5px] text-muted-foreground">{lang === "ar" ? "مثل Cisco Packet Tracer — عربي بالكامل" : "Like Cisco Packet Tracer"}</p>
   336→          </div>
   337→        </div>
   338→        <div className="ms-auto flex flex-wrap items-center gap-1.5">
   339→          <DropdownMenu>
   340→            <DropdownMenuTrigger asChild>
   341→              <Button variant="outline" size="sm" className="h-8 gap-1.5 text-xs">
   342→                <FlaskConical className="size-3.5" /> {lang === "ar" ? "المعامل الجاهزة" : "Labs"}
   343→              </Button>
   344→            </DropdownMenuTrigger>
   345→            <DropdownMenuContent align="end" className="w-64">
   346→              <DropdownMenuLabel className="text-xs">{lang === "ar" ? "حمّل سيناريو جاهزاً" : "Load a ready scenario"}</DropdownMenuLabel>
   347→              {NETSIM_LABS.map((lab) => (
   348→                <DropdownMenuItem key={lab.id} onClick={() => { setTopo(lab.build()); setSim(null); toast({ title: bi(lab.title), description: bi(lab.goal) }); }}>
   349→                  <div className="flex flex-col">
   350→                    <span className="text-xs font-bold">{bi(lab.title)}</span>
   351→                    <span className="text-[10px] text-muted-foreground">{bi(lab.goal).slice(0, 80)}…</span>
   352→                  </div>
   353→                </DropdownMenuItem>
   354→              ))}
   355→            </DropdownMenuContent>
   356→          </DropdownMenu>
   357→          <Button variant="outline" size="sm" className="h-8 gap-1.5 text-xs" onClick={save}><Save className="size-3.5" />{lang === "ar" ? "حفظ" : "Save"}</Button>
   358→          <Button variant="outline" size="sm" className="h-8 gap-1.5 text-xs" onClick={loadSaved}><FolderOpen className="size-3.5" />{lang === "ar" ? "تحميل" : "Load"}</Button>
   359→          <Button variant="outline" size="sm" className="h-8 gap-1.5 text-xs" onClick={exportJson}><Download className="size-3.5" /></Button>
   360→          <Button variant="outline" size="sm" className="h-8 gap-1.5 text-xs" onClick={() => fileRef.current?.click()}><Upload className="size-3.5" /></Button>
   361→          <input ref={fileRef} type="file" accept="application/json" className="hidden" onChange={(e) => e.target.files?.[0] && importJson(e.target.files[0])} />
   362→          <Button variant="outline" size="sm" className="h-8 gap-1.5 text-xs" onClick={() => { setTopo({ devices: [], links: [] }); setSim(null); }}><Eraser className="size-3.5" /></Button>
   363→        </div>
   364→      </div>
   365→
   366→      {/* toolbar */}
   367→      <div className="flex flex-wrap items-center gap-1.5 rounded-xl border bg-card p-1.5">
   368→        {tools.map((tt) => (
   369→          <button
   370→            key={tt.id}
   371→            onClick={() => { setTool(tt.id); setLinkSrc(null); setPingSrc(null); }}
   372→            className={`flex items-center gap-1.5 rounded-lg px-2.5 py-1.5 text-[11px] font-bold transition-colors
   373→              ${tool === tt.id ? "bg-primary text-primary-foreground glow-primary" : "text-muted-foreground hover:bg-accent hover:text-accent-foreground"}`}
   374→            title={bi(tt.label)}
   375→          >
   376→            <tt.icon className="size-4" />
   377→            <span className="hidden sm:inline">{bi(tt.label)}</span>
   378→          </button>
   379→        ))}
   380→        <div className="ms-auto flex items-center gap-2 text-[11px] text-muted-foreground">
   381→          <span>{topo.devices.length} {lang === "ar" ? "جهاز" : "devices"}</span>
   382→          <span>· {topo.links.length} {lang === "ar" ? "وصلة" : "links"}</span>
   383→        </div>
   384→        <Button variant="outline" size="sm" className="h-7 text-[11px] gap-1" onClick={() => setPaletteOpen((v) => !v)}>
   385→          <Cable className="size-3.5" /> {lang === "ar" ? "الأجهزة" : "Devices"}
   386→        </Button>
   387→        <Button variant="outline" size="sm" className="h-7 text-[11px] gap-1" onClick={() => setShowPanel((v) => !v)}>
   388→          <MessageSquareText className="size-3.5" /> {lang === "ar" ? "المحاكاة" : "Simulation"}
   389→          {sim && <span className="size-1.5 rounded-full bg-primary dot-ping relative" />}
   390→        </Button>
   391→      </div>
   392→
   393→      <div className="flex flex-col lg:flex-row gap-3">
   394→        {/* palette desktop */}
