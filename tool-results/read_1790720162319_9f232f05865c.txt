     1→// ─── NetSim mini IOS-like CLI interpreter (routers, switches, firewall, L3, IDS) ─────────
     2→import type { Device, AttackKind } from "./types";
     3→import { findPort, hostIp, isValidIp, networkOf, prefixToMask, connectedRoutes, maskToPrefix, isHostKind } from "./engine";
     4→
     5→export type CliMode = "exec" | "priv" | "conf" | "iface" | "vlan";
     6→
     7→export interface CliState {
     8→  mode: CliMode;
     9→  iface: string; // current interface in config mode
    10→  vlan: number; // current vlan in vlan mode
    11→}
    12→
    13→export const initialCliState: CliState = { mode: "exec", iface: "", vlan: 0 };
    14→
    15→export interface CliAction {
    16→  type: "ping" | "dhcp-renew" | "dns" | "http" | "attack" | "stop-attack";
    17→  arg: string;
    18→}
    19→
    20→export interface CliResult {
    21→  lines: string[];
    22→  state: CliState;
    23→  action?: CliAction;
    24→}
    25→
    26→const OK = [""];
    27→const isRouterLike = (d: Device) => d.kind === "router" || d.kind === "wirelessRouter" || d.kind === "switch" || d.kind === "hub" || d.kind === "firewall" || d.kind === "l3switch" || d.kind === "ids";
    28→const isSwitchLike = (d: Device) => d.kind === "switch" || d.kind === "hub" || d.kind === "l3switch";
    29→const canFilter = (d: Device) => d.kind === "firewall" || d.kind === "l3switch";
    30→const ATTACK_KINDS: AttackKind[] = ["arpspoof", "ddos", "synflood", "scan"];
    31→
    32→function nextAclId(d: Device): number {
    33→  return (d.acls ?? []).reduce((m, r) => Math.max(m, r.id), 0) + 1;
    34→}
    35→
    36→export function promptOf(d: Device, s: CliState): string {
    37→  const h = d.name;
    38→  switch (s.mode) {
    39→    case "exec": return `${h}>`;
    40→    case "priv": return `${h}#`;
    41→    case "conf": return `${h}(config)#`;
    42→    case "iface": return `${h}(config-if)#`;
    43→    case "vlan": return `${h}(config-vlan)#`;
    44→  }
    45→}
    46→
    47→export function runCliLine(d: Device, rawLine: string, state: CliState): CliResult {
    48→  const line = rawLine.trim();
    49→  const lower = line.toLowerCase();
    50→  const lines: string[] = [];
    51→  let st: CliState = { ...state };
    52→  let action: CliAction | undefined;
    53→
    54→  if (!isRouterLike(d)) {
    55→    lines.push("% الأوامر غير متاحة على هذا الجهاز / Commands not available on this device");
    56→    return { lines, state: st };
    57→  }
    58→  if (line === "" || lower === "cls" || lower === "clear") return { lines, state: st };
    59→  if (line === "?" || lower === "help") {
    60→    lines.push(
    61→      "الأوامر المتاحة / Available commands:",
    62→      "  enable, disable, configure terminal",
    63→      "  interface <name> | interface vlan <n>",
    64→      "  hostname <name>, ip route <net> <mask> <nh>",
    65→      "  access-list <permit|deny> <proto> <src> <dst> [port]  ← جدار ناري",
    66→      "  policy deny-all | no policy deny-all              ← السياسة الافتراضية",
    67→      "  switchport port-security                          ← أمن المنفذ",
    68→      "  show ip interface brief | show arp | show vlan",
    69→      "  show mac address-table | show ip route",
    70→      "  show access-list | show alerts | show running-config",
    71→      "  ping <ip>, attack ... (Kali), exit, end, write"
    72→    );
    73→    return { lines, state: st };
    74→  }
    75→
    76→  // do prefix
    77→  let body = line;
    78→  if (lower.startsWith("do ")) {
    79→    body = line.slice(3);
    80→    st = { ...st, mode: "priv" };
    81→  }
    82→
    83→  const tokens = body.split(/\s+/);
    84→  const cmd = tokens[0]?.toLowerCase() ?? "";
    85→  const arg = tokens.slice(1).join(" ");
    86→
    87→  switch (cmd) {
    88→    case "enable": {
    89→      if (st.mode === "exec") st = { ...st, mode: "priv" };
    90→      return { lines: OK, state: st };
    91→    }
    92→    case "disable": {
    93→      if (st.mode === "priv") st = { ...st, mode: "exec" };
    94→      return { lines: OK, state: st };
    95→    }
    96→    case "configure":
    97→    case "conf": {
    98→      if (st.mode === "priv" && ["terminal", "t"].includes(tokens[1]?.toLowerCase() ?? "")) st = { ...st, mode: "conf" };
    99→      else lines.push("% اكتب: configure terminal");
   100→      return { lines, state: st };
   101→    }
   102→    case "interface":
   103→    case "int": {
   104→      if (st.mode !== "conf") {
   105→        lines.push("% خطأ: يجب أن تكون في وضع الإعداد (configure terminal أولاً)");
   106→        return { lines, state: st };
   107→      }
   108→      const lowerArg = arg.toLowerCase();
   109→      if (lowerArg.startsWith("vlan") && d.kind === "l3switch") {
   110→        const n = parseInt(tokens[2] ?? "", 10);
   111→        const svi = d.ports.find((p) => p.kind === "svi" && p.id === `vlan${n}`);
   112→        if (!svi) {
   113→          lines.push(`% لا توجد SVI للـVLAN ${n} — المتاح: vlan10, vlan20, vlan30`);
   114→          return { lines, state: st };
   115→        }
   116→        st = { ...st, mode: "iface", iface: svi.id };
   117→        return { lines: OK, state: st };
   118→      }
   119→      const name = arg;
   120→      const port = findPortByName(d, name);
   121→      if (!port) {
   122→        lines.push(`% واجهة غير صالحة: ${name}. جرّب: ${d.ports.slice(0, 6).map((p) => p.name).join(" , ")}`);
   123→        return { lines, state: st };
   124→      }
   125→      st = { ...st, mode: "iface", iface: port.id };
   126→      return { lines: OK, state: st };
   127→    }
   128→    case "vlan": {
   129→      if (st.mode === "conf") {
   130→        const n = parseInt(tokens[1] ?? "", 10);
   131→        if (!n || n < 1 || n > 4094) {
   132→          lines.push("% رقم VLAN غير صالح (1-4094)");
   133→        } else {
   134→          st = { ...st, mode: "vlan", vlan: n };
   135→        }
   136→      } else lines.push("% يجب أن تكون في وضع configure terminal");
   137→      return { lines, state: st };
   138→    }
   139→    case "name": {
   140→      if (st.mode === "vlan") {
   141→        lines.push(`VLAN ${st.vlan} named ${arg}`);
   142→      } else if (st.mode === "conf") {
   143→        if (isValidIp(arg) === false && arg.length > 0 && arg.length < 26 && !arg.includes(" ") && !arg.match(/\d+\.\d+\.\d+\.\d+/)) {
   144→          d.name = arg;
   145→        } else if (arg.match(/^\S+$/)) {
   146→          d.name = arg;
   147→        }
   148→      }
   149→      return { lines, state: st };
   150→    }
   151→    case "hostname": {
   152→      if (st.mode === "conf" && /^\S{1,25}$/.test(arg)) d.name = arg;
   153→      else lines.push("% اسم غير صالح");
   154→      return { lines, state: st };
   155→    }
   156→    case "exit": {
   157→      if (st.mode === "iface" || st.mode === "vlan") st = { ...st, mode: "conf" };
   158→      else if (st.mode === "conf") st = { ...st, mode: "priv" };
   159→      else if (st.mode === "priv") st = { ...st, mode: "exec" };
   160→      return { lines: OK, state: st };
   161→    }
   162→    case "end": {
   163→      st = { ...st, mode: "priv" };
   164→      return { lines: OK, state: st };
   165→    }
   166→    case "no": {
   167→      const sub = tokens[1]?.toLowerCase();
   168→      if (st.mode === "iface" && sub === "shutdown") {
   169→        const p = findPort(d, st.iface);
   170→        if (p) p.adminUp = true;
   171→        return { lines: [`%LINK-5-CHANGED: Interface ${p?.name}, changed state to up`], state: st };
   172→      }
   173→      if (st.mode === "iface" && isSwitchLike(d) && sub === "switchport" && tokens[2] === "port-security") {
   174→        const p = findPort(d, st.iface);
   175→        if (p) { p.secure = false; p.stickyMac = null; }
   176→        return { lines: [`${p?.name}: أمن المنفذ معطّل`], state: st };
   177→      }
   178→      if (st.mode === "conf" && sub === "policy") {
   179→        d.defaultDeny = false;
   180→        return { lines: ["السياسة الافتراضية: السماح (permit-all)"], state: st };
   181→      }
   182→      if (st.mode === "conf" && sub === "ip" && tokens[2] === "route") {
   183→        const net = tokens[3];
   184→        const mask = tokens[4];
   185→        d.routes = d.routes.filter((r) => !(r.network === net && r.mask === mask && r.kind === "static"));
   186→        return { lines: OK, state: st };
   187→      }
   188→      if (st.mode === "conf" && (sub === "access-list" || sub === "acl")) {
   189→        const id = parseInt(tokens[2] ?? "", 10);
   190→        d.acls = (d.acls ?? []).filter((r) => r.id !== id);
   191→        return { lines: [`حُذفت القاعدة ${id || ""} `], state: st };
   192→      }
   193→      lines.push("% أمر غير مدعوم");
   194→      return { lines, state: st };
   195→    }
   196→    case "shutdown": {
   197→      if (st.mode === "iface") {
   198→        const p = findPort(d, st.iface);
   199→        if (p) p.adminUp = false;
   200→        return { lines: [`%LINK-5-CHANGED: Interface ${p?.name}, changed state to administratively down`], state: st };
   201→      }
   202→      lines.push("% استخدمه داخل واجهة");
   203→      return { lines, state: st };
   204→    }
   205→    case "ip": {
   206→      const sub = tokens[1]?.toLowerCase();
   207→      if (st.mode === "iface" && sub === "address") {
   208→        const ip = tokens[2];
   209→        const mask = tokens[3] ?? (tokens[2]?.includes("/") ? undefined : "255.255.255.0");
   210→        let m = mask;
   211→        let i = ip;
   212→        if (ip?.includes("/")) {
   213→          const [a, p] = ip.split("/");
   214→          i = a;
   215→          m = prefixToMask(parseInt(p, 10));
   216→        }
   217→        if (!i || !m || !isValidIp(i) || maskToPrefix(m) < 0) {
   218→          lines.push("% عنوان/قناع غير صالح. مثال: ip address 192.168.1.1 255.255.255.0");
   219→          return { lines, state: st };
   220→        }
   221→        const p = findPort(d, st.iface);
   222→        if (p) {
   223→          p.ip = i;
   224→          p.mask = m;
   225→          lines.push(`تم ضبط ${i} ${m} على ${p.name}`);
   226→        }
   227→        return { lines, state: st };
   228→      }
   229→      if (st.mode === "conf" && sub === "route") {
   230→        const net = tokens[2];
   231→        let mask = tokens[3];
   232→        const next = tokens[4];
   233→        let network = net;
   234→        if (net?.includes("/")) {
   235→          const [a, pfx] = net.split("/");
   236→          network = a;
   237→          mask = prefixToMask(parseInt(pfx, 10));
   238→        }
   239→        if (!isValidIp(network) || !mask || !isValidIp(mask)) {
   240→          lines.push("% صيغة: ip route <network> <mask> <next-hop>");
   241→          return { lines, state: st };
   242→        }
   243→        if (!next) {
   244→          lines.push("% حدد next-hop أو منفذ خروج");
   245→          return { lines, state: st };
   246→        }
   247→        d.routes = d.routes.filter((r) => !(r.network === network && r.mask === mask && r.kind === "static"));
   248→        d.routes.push({ network, mask, nextHop: isValidIp(next) ? next : null, iface: isValidIp(next) ? "" : next, ad: 1, kind: "static" });
   249→        if (!isValidIp(next)) {
   250→          const route = d.routes[d.routes.length - 1];
   251→          route.iface = next;
   252→        }
   253→        lines.push(`مسار ثابت أُضيف: ${network} ${mask} عبر ${next}`);
   254→        return { lines, state: st };
   255→      }
   256→      if (st.mode === "conf" && sub === "dhcp") {
   257→        lines.push("% إعداد DHCP للمبدّلات غير مدعوم هنا — استخدم الخادم/الراوتر المنزلي");
   258→        return { lines, state: st };
   259→      }
   260→      lines.push("% أمر ip غير مدعوم هنا");
   261→      return { lines, state: st };
   262→    }
   263→    case "access-list":
   264→    case "acl": {
   265→      if (!canFilter(d)) {
   266→        lines.push("% قوائم ACL تُضبط على الجدار الناري أو مبدّل L3");
   267→        return { lines, state: st };
   268→      }
   269→      if (st.mode !== "conf") {
   270→        lines.push("% يجب أن تكون في configure terminal");
   271→        return { lines, state: st };
   272→      }
   273→      const action = tokens[1]?.toLowerCase();
   274→      if (action !== "permit" && action !== "deny") {
   275→        lines.push("% صيغة: access-list permit|deny any|icmp|tcp|udp <src any|ip> <dst any|ip> [port]");
   276→        return { lines, state: st };
   277→      }
   278→      const proto = (tokens[2]?.toLowerCase() ?? "any") as "any" | "icmp" | "tcp" | "udp";
   279→      if (!["any", "icmp", "tcp", "udp"].includes(proto)) {
   280→        lines.push("% البروتوكول: any أو icmp أو tcp أو udp");
   281→        return { lines, state: st };
   282→      }
   283→      const src = tokens[3] ?? "any";
   284→      const dst = tokens[4] ?? "any";
   285→      const port = tokens[5] ? parseInt(tokens[5], 10) : null;
   286→      if (src !== "any" && !isValidIp(src)) { lines.push("% عنوان المصدر غير صالح (أو any)"); return { lines, state: st }; }
   287→      if (dst !== "any" && !isValidIp(dst)) { lines.push("% عنوان الوجهة غير صالح (أو any)"); return { lines, state: st }; }
   288→      const rule = { id: nextAclId(d), action: action as "permit" | "deny", src, srcMask: "any", dst, dstMask: "any", proto, port: Number.isFinite(port) ? port : null, hits: 0 };
   289→      d.acls = [...(d.acls ?? []), rule];
   290→      return { lines: [`قاعدة #${rule.id}: ${action} ${proto} ${src} → ${dst}${port ? ":" + port : ""}`], state: st };
   291→    }
   292→    case "policy": {
   293→      if (!canFilter(d)) { lines.push("% للجدار الناري فقط"); return { lines, state: st }; }
   294→      if (st.mode !== "conf") { lines.push("% يجب أن تكون في configure terminal"); return { lines, state: st }; }
   295→      if (tokens[1]?.toLowerCase() === "deny-all") {
   296→        d.defaultDeny = true;
   297→        return { lines: ["السياسة الافتراضية: رفض كل ما لا تطابقه قواعد permit"], state: st };
   298→      }
   299→      lines.push("% صيغة: policy deny-all");
   300→      return { lines, state: st };
   301→    }
   302→    case "attack": {
   303→      if (d.kind !== "attacker") {
   304→        lines.push("% أوامر الهجوم لجهاز المهاجم (Kali) فقط");
   305→        return { lines, state: st };
   306→      }
   307→      const sub = tokens[1]?.toLowerCase() ?? "";
   308→      if (sub === "stop") {
   309→        d.attack = null;
   310→        return { lines: ["أُوقف الهجوم وصفّرت الحالة"], state: st, action: { type: "stop-attack", arg: "" } };
   311→      }
   312→      if (!ATTACK_KINDS.includes(sub as AttackKind)) {
   313→        lines.push("% صيغ الهجوم:", "  attack arpspoof <ضحيةIP> <هوية مسروقةIP>", "  attack ddos <هدفIP>", "  attack synflood <هدفIP>", "  attack scan <بدايةIP>", "  attack stop");
   314→        return { lines, state: st };
   315→      }
   316→      const a1 = tokens[2];
   317→      const a2 = tokens[3];
   318→      if (!a1 || !isValidIp(a1)) { lines.push("% حدد عنوان IP صالحاً"); return { lines, state: st }; }
   319→      d.attack = {
   320→        kind: sub as AttackKind,
   321→        targetIp: a1,
   322→        victimIp: sub === "arpspoof" ? (a2 && isValidIp(a2) ? a2 : null) : null,
   323→        active: true,
   324→      };
   325→      const desc = sub === "arpspoof"
   326→        ? `تسميم ARP: سأقنع ${a1} أنني أنا ${a2 ?? "الهوية المسروقة"}`
   327→        : sub === "ddos" ? `إغراق ICMP نحو ${a1}`
   328→          : sub === "synflood" ? `إغراق SYN نحو ${a1}:80`
   329→            : `مسح الشبكة بدءاً من ${a1}`;
   330→      return { lines: [desc, "شغّل «تنفيذ الهجوم» أو اضغط زر الهجوم لمشاهدة المحاكاة"], state: st, action: { type: "attack", arg: "" } };
   331→    }
   332→    case "switchport": {
   333→      if (st.mode !== "iface") {
   334→        lines.push("% استخدمه داخل واجهة");
   335→        return { lines, state: st };
   336→      }
   337→      const p = findPort(d, st.iface);
   338→      if (!p || !isSwitchLike(d)) {
   339→        lines.push("% switchport للمبدّلات فقط");
   340→        return { lines, state: st };
   341→      }
   342→      const sub = tokens[1]?.toLowerCase();
   343→      if (sub === "port-security") {
   344→        p.secure = true;
   345→        p.stickyMac = null;
   346→        return { lines: [`${p.name}: أمن المنفذ مفعّل — سيتثبت أول MAC يتعلمه المنفذ`], state: st };
   347→      }
   348→      if (sub === "mode") {
   349→        const m = tokens[2]?.toLowerCase();
   350→        if (m === "access") { p.trunk = false; lines.push(`منفذ ${p.name} صار access`); }
   351→        else if (m === "trunk") { p.trunk = true; lines.push(`منفذ ${p.name} صار trunk`); }
   352→        else lines.push("% mode: access أو trunk");
   353→      } else if (sub === "access" && tokens[2]?.toLowerCase() === "vlan") {
   354→        const n = parseInt(tokens[3] ?? "", 10);
   355→        if (n >= 1 && n <= 4094) { p.accessVlan = n; lines.push(`${p.name} في VLAN ${n}`); }
   356→        else lines.push("% رقم VLAN غير صالح");
   357→      } else {
   358→        lines.push("% جرّب: switchport mode access|trunk أو switchport access vlan <n>");
   359→      }
   360→      return { lines, state: st };
   361→    }
   362→    case "show": {
   363→      if (st.mode === "exec" || st.mode === "priv") {
   364→        lines.push(...showCommand(d, tokens.slice(1).join(" ")));
   365→      } else {
   366→        lines.push("% استخدم do show ... أو اخرج أولاً");
   367→      }
   368→      return { lines, state: st };
   369→    }
   370→    case "write":
   371→    case "copy": {
   372→      lines.push("Building configuration...", "[OK]");
   373→      return { lines, state: st };
   374→    }
   375→    case "ping": {
   376→      if (!isValidIp(arg)) {
   377→        lines.push("% صيغة: ping <ip>");
   378→        return { lines, state: st };
   379→      }
   380→      action = { type: "ping", arg };
   381→      return { lines: [...lines, `Sending 4 echoes to ${arg}, timeout 2s:`], state: st, action };
   382→    }
   383→    case "clear": {
   384→      return { lines: [], state: st };
   385→    }
   386→    default:
   387→      lines.push(`% أمر غير معروف: ${tokens[0]} — اكتب ? للمساعدة`);
   388→      return { lines, state: st };
   389→  }
   390→}

   392→function findPortByName(d: Device, name: string): ReturnType<typeof findPort> {
   393→  const norm = (s: string) => s.toLowerCase().replace(/[^a-z0-9/]/g, "");
   394→  const target = norm(name);
   395→  return d.ports.find((p) => norm(p.name) === target || p.id.toLowerCase() === target || norm(p.name).endsWith(target) || norm(p.name).replace("fastethernet", "fa").replace("gigabitethernet", "g") === target || p.name.toLowerCase() === name.toLowerCase());
   396→}

   398→export function showCommand(d: Device, sub: string): string[] {
   399→  const out: string[] = [];
   400→  const s = sub.toLowerCase();
   401→  if (s.startsWith("access-list") || s === "acl") {
   402→    out.push(`السياسة الافتراضية: ${d.defaultDeny ? "رفض الكل (deny-all)" : "السماح (permit-all)"}`);
   403→    out.push("#  Action  Proto  Source        Dest          Port  Hits");
   404→    const rules = d.acls ?? [];
   405→    out.push(rules.length ? rules.map((r) => `${String(r.id).padEnd(3)}${r.action.padEnd(8)}${r.proto.padEnd(7)}${(r.src === "any" ? "any" : r.src).padEnd(14)}${(r.dst === "any" ? "any" : r.dst).padEnd(14)}${(r.port ?? "-").toString().padEnd(6)}${r.hits}`).join("\n") : "  (لا قواعد)");
   406→  } else if (s.startsWith("alert")) {
   407→    const alerts = d.idsAlerts ?? [];
   408→    out.push(`تنبيهات ${d.name} (${alerts.length}):`);
   409→    out.push(alerts.length ? alerts.map((a) => `[${a.severity.toUpperCase()}] ${a.kind} ${a.srcIp} → ${a.dstIp} — ${a.detail.ar}`).join("\n") : "  (لا تنبيهات — شغّل حركة أو هجوماً)");
   410→  } else if (s.startsWith("attack")) {
   411→    if (d.kind !== "attacker") { out.push("% لجهاز المهاجم فقط"); return out; }
   412→    out.push(d.attack ? `الهجوم الحالي: ${d.attack.kind} → ${d.attack.targetIp}${d.attack.victimIp ? ` (سرقة هوية ${d.attack.victimIp})` : ""} — ${d.attack.active ? "نشط" : "متوقف"}` : "  (لا هجوم مضبوط — جرّب: attack ddos <ip>)");
   413→  } else if (s.startsWith("port-security")) {
   414→    const secured = d.ports.filter((p) => p.secure);
   415→    out.push(secured.length ? secured.map((p) => `${p.name}: ${p.stickyMac ?? "بانتظار أول MAC"}`).join("\n") : "  (لا منافذ مؤمّنة — استخدم switchport port-security داخل الواجهة)");
   416→  } else if (s.startsWith("ip interface brief") || s === "ip int brief" || s.startsWith("ip int")) {
   417→    out.push("Interface              IP-Address      OK? Method Status  Protocol");
   418→    out.push(d.ports.map((p) => {
   419→      const short = p.name.replace("FastEthernet", "Fa").replace("GigabitEthernet", "Gi").replace("Serial", "Se").replace(" (WAN)", "").padEnd(22).slice(0, 22);
   420→      return `${short}${(p.ip ?? "unassigned").padEnd(16).slice(0, 16)}YES manual ${(p.adminUp ? "up" : "down").padEnd(7)} ${p.linkId ? "up" : "down"}`;
   421→    }).join("\n"));
   422→  } else if (s.startsWith("arp")) {
   423→    out.push("Protocol  Address          Age (min)  Hardware Addr   Type");
   424→    out.push(d.arp.length ? d.arp.map((a) => `Internet  ${a.ip.padEnd(16)} -         ${a.mac}  ARPA`).join("\n") : "  (جدول ARP فارغ)");
   425→  } else if (s.startsWith("mac")) {
   426→    out.push("Mac Address Table", "------------------------------------------", "Vlan    Mac Address       Type        Ports");
   427→    out.push(d.macTable.length ? d.macTable.map((e) => `  ${String(e.vlan).padEnd(5)} ${e.mac.padEnd(17)} DYNAMIC     ${e.portId}`).join("\n") : "  (الجدول فارغ)");
   428→  } else if (s.startsWith("ip route")) {
   429→    const all = [...connectedRoutes(d), ...d.routes];
   430→    out.push("Codes: C - connected, S - static", "");
   431→    out.push(all.length ? all.map((r) => `${r.kind === "connected" ? "C" : "S"}    ${networkOf(r.network, r.mask)}${r.mask !== "255.255.255.255" ? "/" + maskToPrefix(r.mask) : ""} ${r.nextHop ? `is directly via ${r.nextHop}` : `is directly connected, ${r.iface}`}`).join("\n") : "  (لا مسارات)");
   432→  } else if (s.startsWith("vlan")) {
   433→    const vlans = new Map<number, number>();
   434→    for (const p of d.ports) vlans.set(p.trunk ? -1 : p.accessVlan, (vlans.get(p.trunk ? -1 : p.accessVlan) ?? 0) + 1);
   435→    out.push("Vlan Name                Status    Ports", "---- -------------------- --------- -------------------------------");
   436→    for (const [v, c] of vlans) {
   437→      out.push(`${String(v === -1 ? "trunk" : v).padEnd(5)}${`VLAN${v}`.padEnd(21)}active    ${c} منفذ`);
   438→    }
   439→  } else if (s.startsWith("running-config") || s === "run") {
   440→    out.push("! التكوين الحالي لـ " + d.name);
   441→    out.push(d.ports.filter((p) => p.ip).map((p) => `interface ${p.name}\n ip address ${p.ip} ${p.mask}\n ${p.adminUp ? "no shutdown" : "shutdown"}${p.secure ? "\n switchport port-security" : ""}${p.trunk ? "\n switchport mode trunk" : ""}${!p.trunk && p.accessVlan !== 1 && (d.kind === "switch" || d.kind === "l3switch") ? `\n switchport access vlan ${p.accessVlan}` : ""}`).join("\n"));
   442→    out.push(d.routes.filter((r) => r.kind === "static").map((r) => `ip route ${r.network} ${r.mask} ${r.nextHop ?? r.iface}`).join("\n"));
   443→    out.push((d.acls ?? []).map((r) => `access-list ${r.action} ${r.proto} ${r.src} ${r.dst}${r.port ? " " + r.port : ""}`).join("\n"));
   444→    if (d.defaultDeny) out.push("policy deny-all");
   445→  } else if (s.startsWith("version")) {
   446→    out.push(`${d.name} — NetMastery Sim (IOS-like)\nالنموذج: ${d.kind === "router" ? "1941 ISR" : d.kind === "switch" ? "2960" : "عمومي"}\nالذاكرة: 512MB DRAM\nالبوابات: ${d.ports.length}`);
   447→  } else {
   448→    out.push("% اكتب: show ip interface brief | show arp | show mac address-table | show ip route | show vlan | show running-config");
   449→  }
   450→  return out;
   451→}

   453→// ─────────────── PC / host command prompt (like PT) ───────────────
   454→export function runHostLine(d: Device, rawLine: string): { lines: string[]; action?: CliAction } {
   455→  const line = rawLine.trim();
   456→  const lower = line.toLowerCase();
   457→  const lines: string[] = [];
   458→  if (!isHostKind(d.kind)) return { lines: ["ليس جهاز طرفية / not an end device"] };
   459→  if (line === "" || lower === "cls" || lower === "clear") return { lines: [] };
   460→  if (lower === "?" || lower === "help") {
   461→    return { lines: [d.kind === "attacker"
   462→      ? "أوامر Kali: attack arpspoof <ضحية> <هوية> | attack ddos <هدف> | attack synflood <هدف> | attack scan <ip> | attack stop | attack status, ipconfig, arp -a, arp -d, ping <ip>"
   463→      : "أوامر متاحة: ipconfig, ipconfig /ip <ip> <mask>, ipconfig /gw <ip>, ipconfig /dns <ip>, ipconfig /dhcp, arp -a, arp -d, ping <ip>, nslookup <name>, http <host>, cls"] };
   464→  }
   465→  if (d.kind === "attacker" && lower.startsWith("attack")) {
   466→    const toks = line.split(/\s+/);
   467→    const sub = toks[1]?.toLowerCase() ?? "";
   468→    if (sub === "status") {
   469→      return { lines: [d.attack ? `الهجوم: ${d.attack.kind} → ${d.attack.targetIp}${d.attack.victimIp ? ` (هوية ${d.attack.victimIp})` : ""}` : "(لا هجوم مضبوط)"] };
   470→    }
   471→    if (sub === "stop") {
   472→      d.attack = null;
   473→      return { lines: ["أُوقف الهجوم"], action: { type: "stop-attack", arg: "" } };
   474→    }
   475→    if (!ATTACK_KINDS.includes(sub as AttackKind) || !toks[2] || !isValidIp(toks[2])) {
   476→      return { lines: ["% صيغة: attack arpspoof|ddos|synflood|scan <ip> [هوية] — أو attack stop"] };
   477→    }
   478→    d.attack = {
   479→      kind: sub as AttackKind,
   480→      targetIp: toks[2],
   481→      victimIp: sub === "arpspoof" && toks[3] && isValidIp(toks[3]) ? toks[3] : null,
   482→      active: true,
   483→    };
   484→    return { lines: [`هجوم ${sub} جاهز نحو ${toks[2]} — تنفيذ المحاكاة الآن...`], action: { type: "attack", arg: "" } };
   485→  }
   486→  const { ip, mask } = hostIp(d);
   487→  if (lower === "ipconfig" || lower === "ifconfig" || lower === "ipconfig /all") {
   488→    lines.push(`اسم الجهاز . . . . . . . : ${d.name}`);
   489→    for (const p of d.ports) {
   490→      lines.push(`واجهة ${p.name}:`);
   491→      lines.push(`  العنوان الفعلي . . . . : ${p.mac}`);
   492→      lines.push(`  عنوان IPv4. . . . . . : ${p.ip ?? (d.dhcpClient ? "بانتظار DHCP" : "غير مضبوط")}`);
   493→      lines.push(`  القناع الفرعي  . . . . : ${p.mask ?? "-"}`);
   494→    }
   495→    lines.push(`  البوابة الافتراضية. . : ${d.gateway ?? "-"}`);
   496→    lines.push(`  خادم DNS  . . . . . . : ${d.dnsServer ?? "-"}`);
   497→    return { lines };
   498→  }
   499→  if (lower.startsWith("ipconfig")) {
   500→    const rest = line.slice(8).trim();
   501→    const toks = rest.split(/\s+/);
   502→    const flag = toks[0]?.toLowerCase() ?? "";
   503→    if (flag === "/ip") {
   504→      const ipn = toks[1];
   505→      let m = toks[2];
   506→      if (ipn?.includes("/")) {
   507→        const [a, p] = ipn.split("/");
   508→        toks[1] = a;
   509→        m = prefixToMask(parseInt(p, 10));
   510→      }
   511→      if (!isValidIp(toks[1] ?? "") || (m !== undefined && !isValidIp(m))) return { lines: ["% صيغة: ipconfig /ip 192.168.1.10 255.255.255.0"] };
   512→      const port = d.ports.find((p) => p.kind !== "wireless" && p.linkId) ?? d.ports[0];
   513→      port.ip = toks[1];
   514→      port.mask = m ?? "255.255.255.0";
   515→      d.dhcpClient = false;
   516→      return { lines: ["تم ضبط العنوان — يلزم /gw للبوابة إن كانت الوجهة خارج الشبكة"] };
   517→    }
   518→    if (flag === "/gw") {
   519→      if (!isValidIp(toks[1] ?? "")) return { lines: ["% صيغة: ipconfig /gw 192.168.1.1"] };
   520→      d.gateway = toks[1];
   521→      return { lines: [`البوابة: ${toks[1]}`] };
   522→    }
   523→    if (flag === "/dns") {
   524→      if (!isValidIp(toks[1] ?? "")) return { lines: ["% صيغة: ipconfig /dns 8.8.8.8"] };
   525→      d.dnsServer = toks[1];
   526→      return { lines: [`DNS: ${toks[1]}`] };
   527→    }
   528→    if (flag === "/dhcp" || flag === "/renew") {
   529→      d.dhcpClient = true;
   530→      for (const p of d.ports) { p.ip = null; p.mask = null; }
   531→      d.gateway = null;
   532→      d.dnsServer = null;
   533→      return { lines: ["طلب DHCP قيد الإرسال... (شغّل المحاكاة)"], action: { type: "dhcp-renew", arg: "" } };
   534→    }
   535→    return { lines: ["% راجع ?"] };
   536→  }
   537→  if (lower === "arp" || lower === "arp -a") {
   538→    return { lines: d.arp.length ? d.arp.map((a) => `  ${a.ip.padEnd(16)} ${a.mac}`).join("\n").split("\n") : ["  (جدول ARP فارغ)"] };
   539→  }
   540→  if (lower === "arp -d" || lower === "arp /d") {
   541→    d.arp = [];
   542→    return { lines: ["أُفرغ جدول ARP — إعادة التعلم عند أول حزمة (علاج التسميم)"] };
   543→  }
   544→  if (lower.startsWith("ping")) {
   545→    const ipn = line.split(/\s+/)[1];
   546→    if (!ipn) return { lines: ["% صيغة: ping <ip>"] };
   547→    return { lines: [`Pinging ${ipn} ببيانات 32 بايت...`], action: { type: "ping", arg: ipn } };
   548→  }
   549→  if (lower.startsWith("nslookup")) {
   550→    const name = line.split(/\s+/)[1];
   551→    if (!name) return { lines: ["% صيغة: nslookup <name>"] };
   552→    return { lines: [`الاستعلام عن ${name}...`], action: { type: "dns", arg: name } };
   553→  }
   554→  if (lower.startsWith("http") || lower.startsWith("curl")) {
   555→    const name = line.split(/\s+/)[1];
   556→    if (!name) return { lines: ["% صيغة: http <host>"] };
   557→    return { lines: [`جلب http://${name} ...`], action: { type: "http", arg: name } };
   558→  }
   559→  return { lines: [`'${line}' ليس أمراً معروفاً — اكتب ?`] };
   560→}
