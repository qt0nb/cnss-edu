   923→function dhcpHostHandle(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
   924→  const inP = findPort(d, inPortId);
   925→  const out = (inP && inP.linkId && inP.adminUp ? inP : d.ports.find((p) => p.linkId && p.adminUp)) ?? d.ports[0];
   926→  if (packet.kind === "DHCP Discover") {
   927→    // server side
   928→    const pool = d.dhcpPool;
   929→    if (!pool || !pool.ifacePortId) {
   930→      step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: لا أدير DHCP — تجاهل`, `${d.name}: not a DHCP server — ignoring`));
   931→      return;
   932→    }
   933→    const ip = leaseIp(pool, packet.srcMac);
   934→    const offer = mkPacket({
   935→      proto: "DHCP",
   936→      kind: "DHCP Offer",
   937→      kindBi: { ar: "عرض DHCP", en: "DHCP Offer" },
   938→      srcMac: out.mac,
   939→      dstMac: packet.srcMac,
   940→      srcIp: (findPort(d, pool.ifacePortId)?.ip) ?? "0.0.0.0",
   941→      dstIp: ip,
   942→      ttl: 128,
   943→      srcPort: 67,
   944→      dstPort: 68,
   945→      payload: { ar: `أعرض ${ip} بوابة ${pool.gateway} قناع ${pool.mask}`, en: `Offering ${ip} gw ${pool.gateway} mask ${pool.mask}` },
   946→    });
   947→    step(ctx, d, inPortId, out.id, null, null, offer, info(`${d.name}: عرض DHCP بعنوان ${ip}`, `${d.name}: DHCP offer with ${ip}`));
   948→    wireOut(ctx, d, out.id, offer, depth);
   949→    return;
   950→  }
   951→  if (packet.kind === "DHCP Request" && d.dhcpPool) {
   952→    const pool = d.dhcpPool;
   953→    const m = packet.payload?.en.match(/\b(\d{1,3}(?:\.\d{1,3}){3})\b/);
   954→    const ip = m?.[1] ?? leaseIp(pool, packet.srcMac);
   955→    const ack = mkPacket({
   956→      proto: "DHCP",
   957→      kind: "DHCP Ack",
   958→      kindBi: { ar: "تأكيد DHCP", en: "DHCP Ack" },
   959→      srcMac: out.mac,
   960→      dstMac: packet.srcMac,
   961→      srcIp: (findPort(d, pool.ifacePortId)?.ip) ?? "0.0.0.0",
   962→      dstIp: ip,
   963→      ttl: 128,
   964→      srcPort: 67,
   965→      dstPort: 68,
   966→      payload: { ar: `تأكيد ${ip} بوابة ${pool.gateway} DNS ${pool.dns ?? "-"}`, en: `Ack ${ip} gw ${pool.gateway} dns ${pool.dns ?? "-"}` },
   967→    });
   968→    step(ctx, d, inPortId, out.id, null, null, ack, info(`${d.name}: تأكيد DHCP — ${ip} أصبح لعنوان ${packet.srcMac}`, `${d.name}: DHCP ack — ${ip} assigned to ${packet.srcMac}`));
   969→    wireOut(ctx, d, out.id, ack, depth);
   970→    return;
   971→  }
   972→  // client side
   973→  if (packet.kind === "DHCP Offer" && d.dhcpClient) {
   974→    const ip = packet.dstIp;
   975→    const req = mkPacket({
   976→      proto: "DHCP",
   977→      kind: "DHCP Request",
   978→      kindBi: { ar: "طلب DHCP", en: "DHCP Request" },
   979→      srcMac: out.mac,
   980→      dstMac: BROADCAST_MAC,
   981→      srcIp: ZERO_IP,
   982→      dstIp: BROADCAST_IP,
   983→      ttl: 128,
   984→      srcPort: 68,
   985→      dstPort: 67,
   986→      payload: { ar: `أطلب ${ip}`, en: `Requesting ${ip}` },
   987→    });
   988→    step(ctx, d, inPortId, out.id, null, null, req, info(`${d.name}: طلب العنوان ${ip}`, `${d.name}: requesting ${ip}`));
   989→    (d as Device & { _offered?: string })._offered = ip;
   990→    wireOut(ctx, d, out.id, req, depth);
   991→    return;
   992→  }
   993→  if (packet.kind === "DHCP Ack" && d.dhcpClient) {
   994→    const ip = packet.dstIp;
   995→    const port = out;
   996→    port.ip = ip;
   997→    port.mask = "255.255.255.0";
   998→    const gwMatch = packet.payload?.ar.match(/بوابة\s*(\S+)/);
   999→    const dnsMatch = packet.payload?.ar.match(/DNS\s*(\S+)/);
  1000→    d.gateway = gwMatch?.[1] ?? null;
  1001→    d.dnsServer = dnsMatch?.[1] && dnsMatch[1] !== "-" ? dnsMatch[1] : null;
  1002→    delete (d as Device & { _offered?: string })._offered;
  1003→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: استلمت ${ip} عبر DHCP — تم الإعداد!`, `${d.name}: got ${ip} via DHCP — configured!`));
  1004→    return;
  1005→  }

  1008→function leaseIp(pool: import("./types").DhcpPool, mac: string): string {
  1009→  const existing = pool.leases.find((l) => l.mac === mac);
  1010→  if (existing) return existing.ip;
  1011→  const from = ipToInt(pool.from);
  1012→  const to = ipToInt(pool.to);
  1013→  const used = new Set(pool.leases.map((l) => l.ip));
  1014→  for (let i = from; i <= to; i++) {
  1015→    const cand = intToIp(i);
  1016→    if (!used.has(cand)) {
  1017→      pool.leases.push({ mac, ip: cand });
  1018→      return cand;
  1019→    }
  1020→  }
  1021→  return pool.from;
  1022→}

  1024→function httpHandle(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
  1025→  const inP0 = findPort(d, inPortId);
  1026→  const out = (inP0 && inP0.linkId && inP0.adminUp ? inP0 : d.ports.find((p) => p.linkId && p.adminUp)) ?? d.ports[0];
  1027→  const myIp = hostIp(d).ip ?? d.ports.find((p) => p.ip)?.ip;
  1028→  if (!myIp) return;
  1029→  const forMeIp = packet.dstIp === myIp;
  1030→
  1031→  if (packet.kind === "TCP SYN" && forMeIp) {
  1032→    // SYN flood: half-open connections exhaust the backlog
  1033→    const hits = (ctx.hit.get(`${d.id}|${packet.srcIp}`) ?? 0) + 1;
  1034→    ctx.hit.set(`${d.id}|${packet.srcIp}`, hits);
  1035→    if (hits >= 6) d.halfOpen = 6;
  1036→    if ((d.halfOpen ?? 0) >= 6) {
  1037→      step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: طابور SYN ممتلئ — رفض الاتصال (SYN Flood?)`, `${d.name}: SYN backlog full — connection refused (SYN flood?)`), true);
  1038→      return;
  1039→    }
  1040→    const synack = mkPacket({ proto: "TCP", kind: "TCP SYN-ACK", kindBi: { ar: "مصافحة SYN-ACK", en: "TCP SYN-ACK" }, srcMac: out.mac, dstMac: packet.srcMac, srcIp: myIp, dstIp: packet.srcIp, ttl: 128, srcPort: packet.dstPort, dstPort: packet.srcPort, flags: "SYN,ACK", seq: "200" });
  1041→    step(ctx, d, inPortId, out.id, null, null, synack, info(`${d.name}: SYN-ACK — بدء المصافحة الثلاثية`, `${d.name}: SYN-ACK — three-way handshake`));
  1042→    wireOut(ctx, d, out.id, synack, depth);
  1043→    return;
  1044→  }
  1045→  if (packet.kind === "TCP ACK") {
  1046→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: اكتملت المصافحة — الاتصال مفتوح`, `${d.name}: handshake complete — connection open`));
  1047→    return;
  1048→  }
  1049→  if (packet.kind === "HTTP GET" && forMeIp) {
  1050→    const page = d.httpRoot;
  1051→    const ok = mkPacket({ proto: "HTTP", kind: "HTTP 200 OK", kindBi: { ar: "رد HTTP 200", en: "HTTP 200 OK" }, srcMac: out.mac, dstMac: packet.srcMac, srcIp: myIp, dstIp: packet.srcIp, ttl: 128, srcPort: 80, dstPort: packet.srcPort, payload: { ar: `${page.title}\n${page.body}`, en: `${page.title}\n${page.body}` } });
  1052→    step(ctx, d, inPortId, out.id, null, null, ok, info(`${d.name}: صفحة «${page.title}» جاهزة — 200 OK`, `${d.name}: page "${page.title}" served — 200 OK`));
  1053→    wireOut(ctx, d, out.id, ok, depth);
  1054→    return;
  1055→  }
  1056→  step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: استلم حزمة ${packet.kind}`, `${d.name}: received ${packet.kind}`));
  1057→}

  1064→export function simulatePing(topo: Topology, srcId: string, dstIdOrIp: string): SimResult {
  1065→  const t = cloneTopo(topo);
  1066→  const ctx = newCtx(t);
  1067→  const src = findDevice(t, srcId);
  1068→  if (!src) return { steps: [], devices: topo.devices, success: false, note: info("الجهاز غير موجود", "Device not found") };
  1069→  const { ip, mask, port } = hostIp(src);
  1070→  if (!ip || !mask || !port) {
  1071→    return { steps: [], devices: topo.devices, success: false, note: info(`${src.name} بلا عنوان IP — اضبطه أولاً من إعدادات الجهاز`, `${src.name} has no IP — configure it first in device settings`) };
  1072→  }
  1073→  // resolve destination ip
  1074→  let dstIp = dstIdOrIp;
  1075→  const dstDev = findDevice(t, dstIdOrIp);
  1076→  if (dstDev) {
  1077→    const dip = hostIp(dstDev).ip;
  1078→    if (!dip) return { steps: [], devices: topo.devices, success: false, note: info(`${dstDev.name} بلا عنوان IP`, `${dstDev.name} has no IP`) };
  1079→    dstIp = dip;
  1080→  }
  1081→  // ARP resolution
  1082→  const local = sameSubnet(ip, dstIp, mask);
  1083→  const nextIp = local ? dstIp : src.gateway;
  1084→  if (!nextIp) {
  1085→    return { steps: [], devices: topo.devices, success: false, note: info("الوجهة خارج شبكتك ولا يوجد بوابة افتراضية (Default Gateway)", "Destination is off-subnet and no default gateway is set") };
  1086→  }
  1087→  let arpEntry = src.arp.find((a) => a.ip === nextIp);
  1088→  const outPort = local ? port : (port.linkId ? port : port);
  1089→  if (!arpEntry) {
  1090→    const req = mkPacket({ proto: "ARP", kind: "ARP Request", kindBi: { ar: "طلب ARP", en: "ARP Request" }, srcMac: outPort.mac, dstMac: BROADCAST_MAC, srcIp: ip, dstIp: nextIp, ttl: 128, payload: { ar: `من يعرف ${nextIp}؟`, en: `Who has ${nextIp}?` } });
  1091→    step(ctx, src, null, outPort.id, null, null, req, info(`${src.name}: أريد ${nextIp} — إرسال ARP بثّاً`, `${src.name}: need ${nextIp} — broadcasting ARP`));
  1092→    wireOut(ctx, src, outPort.id, req, 0);
  1093→    arpEntry = src.arp.find((a) => a.ip === nextIp);
  1094→    if (!arpEntry) {
  1095→      return { steps: ctx.steps, devices: t.devices, success: false, note: info("لا جواب ARP — تحقق من الكابلات والشبكة الفرعية نفسها", "No ARP reply — check cables and matching subnets") };
  1096→    }
  1097→  }
  1098→  const echo = mkPacket({ proto: "ICMP", kind: "Echo Request", kindBi: { ar: "طلب ICMP (Ping)", en: "ICMP Echo Request" }, srcMac: outPort.mac, dstMac: arpEntry.mac, srcIp: ip, dstIp, ttl: 128 });
  1099→  step(ctx, src, null, outPort.id, null, null, echo, info(`${src.name}: Ping إلى ${dstIp}`, `${src.name}: ping to ${dstIp}`));
  1100→  wireOut(ctx, src, outPort.id, echo, 0);
  1101→  const replied = ctx.steps.some((s) => s.packet.kind === "Echo Reply" && s.deviceId === srcId);
  1102→  return {
  1103→    steps: ctx.steps,
  1104→    devices: t.devices,
  1105→    success: replied,
  1106→    note: replied
  1107→      ? info("نجح Ping! رأيت الطلب والرد يمران عبر كل الأجهزة", "Ping succeeded! You saw request & reply traverse every device")
  1108→      : info("فشل Ping — راجع الخطوات المظللة بالأحمر", "Ping failed — review the red-marked steps"),
  1109→  };
  1110→}

  1126→export function simulateDns(topo: Topology, clientId: string, name: string): SimResult & { resolvedIp: string | null } {

  1152→export function simulateHttp(topo: Topology, clientId: string, host: string): SimResult & { page: { title: string; body: string } | null } {
  1153→  const t = cloneTopo(topo);
  1154→  const ctx = newCtx(t);
  1155→  const client = findDevice(t, clientId);
  1156→  if (!client) return { steps: [], devices: topo.devices, success: false, note: info("الجهاز غير موجود", "Device not found"), page: null };
  1157→  let targetIp: string | null = null;
  1158→  let stepsSoFar: SimStep[] = [];
  1159→  if (!isValidIp(host)) {
  1160→    const dns = simulateDns(topo, clientId, host);
  1161→    stepsSoFar = dns.steps;
  1162→    // commit arp/dns state
  1163→    for (const dev of dns.devices) {
  1164→      const mine = findDevice(t, dev.id);
  1165→      if (mine) {
  1166→        mine.arp = dev.arp;
  1167→        mine.macTable = dev.macTable;
  1168→      }
  1169→    }
  1170→    targetIp = dns.resolvedIp;
  1171→    if (!targetIp) {
  1172→      return { steps: stepsSoFar, devices: t.devices, success: false, note: info(`تعذر حل اسم ${host} — أضف سجل A في خادم DNS`, `Cannot resolve ${host} — add an A record on the DNS server`), page: null };
  1173→    }
  1174→  } else {
  1175→    targetIp = host;
  1176→  }
  1177→  // run on a fresh context so steps continue linearly
  1178→  ctx.steps.push(...stepsSoFar);
  1179→  ctx.sid = stepsSoFar.length;
  1180→  const { ip, mask, port } = hostIp(client);
  1181→  if (!ip || !mask || !port) return { steps: ctx.steps, devices: t.devices, success: false, note: info("اضبط عنوان الجهاز أولاً", "Configure device IP first"), page: null };
  1182→  const local = sameSubnet(ip, targetIp, mask);
  1183→  const nextIp = local ? targetIp : client.gateway;
  1184→  if (!nextIp) return { steps: ctx.steps, devices: t.devices, success: false, note: info("لا بوابة مضبوطة", "No gateway set"), page: null };
  1185→  let arpEntry = client.arp.find((a) => a.ip === nextIp);
  1186→  if (!arpEntry) {
  1187→    ensureArpFor(ctx, client, port, nextIp, 0);
  1188→    arpEntry = client.arp.find((a) => a.ip === nextIp);
  1189→    if (!arpEntry) return { steps: ctx.steps, devices: t.devices, success: false, note: info("لا جواب ARP", "No ARP reply"), page: null };
  1190→  }
  1191→  // TCP handshake
  1192→  const sendTcp = (kind: string, kindBi: { ar: string; en: string }, flags: string, seq: string, srcPort: number, dstPort: number, payload?: { ar: string; en: string }) => {
  1193→    const pkt = mkPacket({ proto: kind.startsWith("HTTP") ? "HTTP" : "TCP", kind, kindBi, srcMac: port.mac, dstMac: arpEntry!.mac, srcIp: ip!, dstIp: targetIp!, ttl: 128, srcPort, dstPort, flags, seq, payload });
  1194→    wireOut(ctx, client, port.id, pkt, 0);
  1195→    return pkt;
  1196→  };
  1197→  const syn = sendTcp("TCP SYN", { ar: "SYN", en: "TCP SYN" }, "SYN", "100", 1025, 80);
  1198→  void syn;
  1199→  // SYN-ACK/ACK handled by handlers; send ACK
  1200→  const ack = sendTcp("TCP ACK", { ar: "ACK", en: "TCP ACK" }, "ACK", "101", 1025, 80);
  1201→  void ack;
  1202→  // GET
  1203→  const get = sendTcp("HTTP GET", { ar: "طلب GET", en: "HTTP GET" }, "ACK", "102", 1025, 80, { ar: `GET / HTTP/1.1\nHost: ${host}`, en: `GET / HTTP/1.1\nHost: ${host}` });
  1204→  void get;
  1205→  const okStep = [...ctx.steps].reverse().find((s) => s.packet.kind === "HTTP 200 OK" && s.deviceId === clientId);

  1234→/** Evaluate a packet against a device ACL rule list (first match wins) */
  1235→export function aclCheck(d: Device, packet: Packet): { action: "permit" | "deny"; rule: import("./types").AclRule | null } {
  1236→  const rules = d.acls ?? [];
  1237→  for (const r of rules) {
  1238→    if (matchAcl(r, packet)) return { action: r.action, rule: r };
  1239→  }
  1240→  return { action: d.defaultDeny ? "deny" : "permit", rule: null };
  1241→}

  1243→function matchAcl(r: import("./types").AclRule, packet: Packet): boolean {
  1244→  if (r.src !== "any" && r.srcMask !== "any" && !ipInSubnet(packet.srcIp, r.src, r.srcMask)) return false;
  1245→  if (r.dst !== "any" && r.dstMask !== "any" && !ipInSubnet(packet.dstIp, r.dst, r.dstMask)) return false;
  1246→  if (r.proto !== "any") {
  1247→    const protoName =
  1248→      packet.proto === "ICMP" ? "icmp"
  1249→        : packet.proto === "TCP" || packet.proto === "HTTP" ? "tcp"
  1250→          : packet.proto === "DNS" || packet.proto === "DHCP" ? "udp"
  1251→            : "any";
  1252→    if (r.proto !== protoName) return false;
  1253→    if (r.port != null && packet.dstPort !== r.port) return false;
  1254→  }
  1255→  return true;
  1256→}

  1258→function ipInSubnet(ip: string, net: string, mask: string): boolean {
  1259→  if (!isValidIp(ip) || !isValidIp(net) || !isValidIp(mask)) return false;
  1260→  return networkOf(ip, mask) === networkOf(net, mask);
  1261→}

  1263→/** IDS sensor inspection: ARP-spoof binding conflicts, floods & scans */
  1264→function idsInspect(ctx: Ctx, d: Device, inPortId: string, packet: Packet): void {
  1265→  d.statsIn = (d.statsIn ?? 0) + 1;
  1266→  // IP↔MAC binding DB: forged ARP replies that contradict a known binding = spoof
  1267→  if (packet.proto === "ARP" && packet.kind === "ARP Reply" && isValidIp(packet.srcIp) && packet.srcIp !== ZERO_IP) {
  1268→    const prev = ctx.ipMac.get(packet.srcIp);
  1269→    if (prev && prev !== packet.srcMac) {
  1270→      pushAlert(d, "arp-spoof", "critical", packet.srcIp, packet.dstIp,
  1271→        `تسميم ARP مكتشف: ${packet.srcIp} كان ${prev} والآن يدّعي ${packet.srcMac} — مصدره منفذ ${inPortId}`,
  1272→        `ARP poisoning detected: ${packet.srcIp} was ${prev} but now claims ${packet.srcMac} — seen on port ${inPortId}`);
  1273→      return;
  1274→    }
  1275→    if (!prev) ctx.ipMac.set(packet.srcIp, packet.srcMac);
  1276→  }
  1277→  if (!isValidIp(packet.srcIp) || packet.srcIp === ZERO_IP) return;
  1278→  // flow statistics per source
  1279→  const fs = ctx.flow.get(packet.srcIp) ?? { echo: 0, syn: 0, dsts: [] };
  1280→  if (packet.proto === "ICMP" && packet.kind === "Echo Request") fs.echo += 1;
  1281→  if (packet.kind === "TCP SYN") fs.syn += 1;
  1282→  if (!fs.dsts.includes(packet.dstIp)) fs.dsts.push(packet.dstIp);
  1283→  ctx.flow.set(packet.srcIp, fs);
  1284→  if (fs.echo === 8) {
  1285→    pushAlert(d, "icmp-flood", "critical", packet.srcIp, packet.dstIp,
  1286→      `إغراق ICMP: 8+ طلبات Echo متتالية من ${packet.srcIp} — نمط هجوم DDoS`,
  1287→      `ICMP flood: 8+ consecutive echo requests from ${packet.srcIp} — DDoS attack pattern`);
  1288→  }
  1289→  if (fs.syn === 8) {
  1290→    pushAlert(d, "syn-flood", "critical", packet.srcIp, packet.dstIp,
  1291→      `إغراق SYN: 8+ محاولات اتصال نصف مفتوحة من ${packet.srcIp} إلى ${packet.dstIp}`,
  1292→      `SYN flood: 8+ half-open connection attempts from ${packet.srcIp} to ${packet.dstIp}`);
  1293→  }
  1294→  if (fs.dsts.length === 5) {
  1295→    pushAlert(d, "scan", "warn", packet.srcIp, packet.dstIp,
  1296→      `مسح شبكة مكتشف: ${packet.srcIp} يرمي حزماً نحو 5 عناوين مختلفة — نمط Port/Host Scan`,
  1297→      `Network scan detected: ${packet.srcIp} probing 5 distinct addresses — host/port scan pattern`);
  1298→  }
  1299→}

  1302→function l3Route(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
  1303→  if (packet.proto === "ARP" && packet.kind === "ARP Request") {
  1304→    addArp(d, packet.srcIp, packet.srcMac, inPortId);
  1305→    const svi = d.ports.find((p) => p.kind === "svi" && p.ip === packet.dstIp);
  1306→    if (svi && svi.ip) {
  1307→      const reply = mkPacket({
  1308→        proto: "ARP", kind: "ARP Reply", kindBi: { ar: "رد ARP (SVI)", en: "ARP Reply (SVI)" },
  1309→        srcMac: svi.mac, dstMac: packet.srcMac, srcIp: svi.ip, dstIp: packet.srcIp, ttl: 128,
  1310→        payload: { ar: `بوابة VLAN أنا ${svi.ip}`, en: `I am the VLAN gateway ${svi.ip}` },
  1311→      });
  1312→      step(ctx, d, inPortId, inPortId, null, null, reply, info(`${d.name}: SVI ${svi.id} يرد على ARP (${svi.ip})`, `${d.name}: SVI ${svi.id} answers ARP (${svi.ip})`));
  1313→      wireOut(ctx, d, inPortId, reply, depth);
  1314→    }
  1315→    return;
  1316→  }
  1317→  if (packet.proto === "ARP" && packet.kind === "ARP Reply") {
  1318→    addArp(d, packet.srcIp, packet.srcMac, inPortId);
  1319→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: تعلمت ${packet.srcIp} = ${packet.srcMac} في جدول ARP`, `${d.name}: learned ${packet.srcIp} = ${packet.srcMac} in ARP table`));
  1320→    return;
  1321→  }
  1322→  const sviMe = d.ports.find((p) => p.kind === "svi" && p.ip === packet.dstIp);
  1323→  if (packet.proto === "ICMP" && packet.kind === "Echo Request" && sviMe && sviMe.ip) {
  1324→    const reply = mkPacket({
  1325→      proto: "ICMP", kind: "Echo Reply", kindBi: { ar: "رد ICMP", en: "ICMP Echo Reply" },
  1326→      srcMac: sviMe.mac, dstMac: packet.srcMac, srcIp: sviMe.ip, dstIp: packet.srcIp, ttl: 128,
  1327→    });
  1328→    step(ctx, d, inPortId, inPortId, null, null, reply, info(`${d.name}: SVI ${sviMe.id} هو الهدف — رد Pong`, `${d.name}: SVI ${sviMe.id} is the target — pong reply`));
  1329→    wireOut(ctx, d, inPortId, reply, depth);
  1330→    return;
  1331→  }
  1332→  if (packet.proto === "DHCP" || packet.proto === "DNS") return; // still bridge-serviced below
  1333→  if (packet.dstMac !== BROADCAST_MAC) {
  1334→    l3Forward(ctx, d, inPortId, packet, depth);
  1335→  }

  1338→function l3Forward(ctx: Ctx, d: Device, inPortId: string, packet: Packet, depth: number): void {
  1339→  if (packet.ttl <= 1) {
  1340→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: TTL انتهى — إسقاط`, `${d.name}: TTL expired — dropped`), true);
  1341→    return;
  1342→  }
  1343→  const route = lookupRoute(d, packet.dstIp);
  1344→  if (!route) {
  1345→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: لا مسار إلى ${packet.dstIp} — فعّل ip routing واضبط SVI`, `${d.name}: no route to ${packet.dstIp} — configure an SVI or add a route`), true);
  1346→    return;
  1347→  }
  1348→  // resolve egress SVI by route iface or by subnet match
  1349→  let svi = route.iface ? d.ports.find((p) => p.id === route.iface && p.kind === "svi") : undefined;
  1350→  if (!svi) {
  1351→    svi = d.ports.find((p) => p.kind === "svi" && p.ip && p.mask && networkOf(packet.dstIp, p.mask) === networkOf(p.ip, p.mask));
  1352→  }
  1353→  if (!svi || !svi.ip) {
  1354→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: لا SVI لشبكة ${packet.dstIp}`, `${d.name}: no SVI for ${packet.dstIp} subnet`), true);
  1355→    return;
  1356→  }
  1357→  const vlan = parseInt(svi.id.replace("vlan", ""), 10) || 1;
  1358→  const nextIp = route.nextHop ?? packet.dstIp;
  1359→  let arpEntry = d.arp.find((a) => a.ip === nextIp);
  1360→  if (!arpEntry) {
  1361→    arpBroadcastOnVlan(ctx, d, vlan, svi, nextIp, depth);
  1362→    arpEntry = d.arp.find((a) => a.ip === nextIp);
  1363→    if (!arpEntry) {
  1364→      step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: لا جواب ARP في VLAN${vlan} — الحزمة سقطت`, `${d.name}: no ARP reply in VLAN${vlan} — packet dropped`), true);
  1365→      return;
  1366→    }
  1367→  }
  1368→  // pick the physical port where the destination MAC lives (or flood the vlan)
  1369→  const entry = d.macTable.find((e) => e.mac === arpEntry.mac && e.vlan === vlan);
  1370→  const phys = d.ports.filter((p) => p.kind === "ethernet" && p.linkId && p.adminUp && p.id !== inPortId && (p.trunk || p.accessVlan === vlan));
  1371→  const targets = entry ? phys.filter((p) => p.id === entry.portId) : phys;
  1372→  if (targets.length === 0) {
  1373→    step(ctx, d, inPortId, null, null, null, packet, info(`${d.name}: لا منافذ في VLAN${vlan} للخروج`, `${d.name}: no egress ports in VLAN${vlan}`), true);
  1374→    return;
  1375→  }
  1376→  for (const p of targets) {
  1377→    const fwd = mkPacket({ ...packet, srcMac: p.mac, dstMac: arpEntry.mac, ttl: packet.ttl - 1, vlan: p.trunk ? vlan : null });
  1378→    step(ctx, d, inPortId, p.id, null, null, fwd, info(`${d.name}: توجيه بين VLANs → خارج ${p.name} نحو ${arpEntry.mac} (TTL-1)`, `${d.name}: inter-VLAN route → out ${p.name} toward ${arpEntry.mac} (TTL-1)`));
  1379→    wireOut(ctx, d, p.id, fwd, depth);
  1380→  }
  1381→}

  1383→function arpBroadcastOnVlan(ctx: Ctx, d: Device, vlan: number, svi: Port, targetIp: string, depth: number): void {
  1384→  const phys = d.ports.filter((p) => p.kind === "ethernet" && p.linkId && p.adminUp && (p.trunk || p.accessVlan === vlan));

  1499→  };
