    27→export interface NetSimActions {
    28→  updateDevice(): void;
    29→  runPing(srcId: string, ip: string): string[];

    34→
    35→export function CliTerminal({
    36→  device,
    37→  actions,
    38→  isHost,
    39→}: {
    40→  device: Device;
    41→  actions: NetSimActions;
    42→  isHost: boolean;
    43→}) {
    44→  const { lang } = useLang();
    45→  const [lines, setLines] = useState<string[]>([
    46→    lang === "ar"
    47→      ? "NetMastery Sim — اكتب ? لعرض الأوامر"
    48→      : "NetMastery Sim — type ? for commands",
    49→  ]);
    50→  const [input, setInput] = useState("");
    51→  const [cliState, setCliState] = useState<CliState>(initialCliState);
    52→  const [hist, setHist] = useState<string[]>([]);
    53→  const [hIdx, setHIdx] = useState(-1);
    54→  const endRef = useRef<HTMLDivElement>(null);
    55→
    56→  useEffect(() => {
    57→    endRef.current?.scrollIntoView({ block: "end" });
    58→  }, [lines]);
    59→
    60→  const prompt = isHost
    61→    ? `${device.name}>`
    62→    : promptOf(device, cliState);
    63→
    64→  const submit = () => {
    65→    const line = input;
    66→    setInput("");
    67→    if (line.trim()) setHist((h) => [...h, line]);
    68→    setHIdx(-1);
    69→    const promptLine = `${prompt} ${line}`;
    70→    let out: string[] = [];
    71→    if (isHost) {
    72→      const r = runHostLine(device, line);
    73→      out = r.lines;
    74→      if (r.action?.type === "ping") out = [...out, ...actions.runPing(device.id, r.action.arg)];
    75→      else if (r.action?.type === "dhcp-renew") out = [...out, ...actions.runDhcp(device.id)];
    76→      else if (r.action?.type === "dns") out = [...out, ...actions.runDns(device.id, r.action.arg)];
    77→      else if (r.action?.type === "http") {
    78→        const r2 = actions.runHttp(device.id, r.action.arg);
    79→        out = [...out, ...r2.lines];
    80→      }
    81→      actions.updateDevice();
    82→    } else {
    83→      const r = runCliLine(device, line, cliState);
    84→      setCliState(r.state);
    85→      out = r.lines;
    86→      if (r.action?.type === "ping") out = [...out, ...actions.runPing(device.id, r.action.arg)];
    87→      actions.updateDevice();
    88→    }
    89→    setLines((ls) => [...ls, promptLine, ...out]);
    90→  };
    91→
    92→  return (
    93→    <div className="flex flex-col gap-2" dir="ltr">
    94→      <div className="rounded-lg bg-zinc-950 border border-zinc-800 p-3 h-72 overflow-y-auto text-[12px] font-mono text-emerald-300 leading-relaxed" style={{ whiteSpace: "pre-wrap" }}>
    95→        {lines.map((l, i) => (
    96→          <div key={i} className={l.startsWith("%") ? "text-rose-400" : undefined}>{l || "\u00A0"}</div>
    97→        ))}
    98→        <div ref={endRef} />
    99→      </div>
   100→      <div className="flex items-center gap-2">
   101→        <span className="text-xs font-mono text-emerald-500 shrink-0">{prompt}</span>
   102→        <Input
   103→          value={input}
   104→          onChange={(e) => setInput(e.target.value)}
   105→          onKeyDown={(e) => {
   106→            if (e.key === "Enter") submit();
   107→            else if (e.key === "ArrowUp") {
   108→              e.preventDefault();
   109→              const idx = hIdx === -1 ? hist.length - 1 : Math.max(0, hIdx - 1);
   110→              if (hist[idx] !== undefined) { setHIdx(idx); setInput(hist[idx]); }
   111→            } else if (e.key === "ArrowDown") {
   112→              e.preventDefault();
   113→              const idx = hIdx === -1 ? -1 : Math.min(hist.length - 1, hIdx + 1);
   114→              if (idx === hist.length - 1) { setHIdx(-1); setInput(""); }
   115→              else if (hist[idx] !== undefined) { setHIdx(idx); setInput(hist[idx]); }
   116→            }
   117→          }}
   118→          className="font-mono text-xs h-8 bg-zinc-950 border-zinc-800 text-emerald-200"
   119→          placeholder="?"
   120→          autoComplete="off"
   121→          spellCheck={false}
   122→        />
   123→        <Button size="sm" onClick={submit} className="h-8">↵</Button>
   124→      </div>
   125→    </div>
   126→  );
   127→}
   128→
   129→function Field({ label, value, onChange, mono = true, placeholder }: { label: string; value: string; onChange: (v: string) => void; mono?: boolean; placeholder?: string }) {
   130→  return (
   131→    <div className="grid grid-cols-[110px_1fr] items-center gap-2">
   132→      <Label className="text-[11px] text-muted-foreground">{label}</Label>
   133→      <Input
   134→        dir="ltr"
   135→        value={value}
   136→        placeholder={placeholder}
   137→        onChange={(e) => onChange(e.target.value)}
   138→        className={`h-8 text-xs ${mono ? "font-mono" : ""}`}
   139→      />
   140→    </div>
   141→  );
   142→}
